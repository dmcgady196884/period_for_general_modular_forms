\title{L-functions for all meromorphic $\SLZ$ modular forms,\\
via manifestly finite contours}
\author{David A.\ McGady}
\date{}

\begin{document}
\maketitle

\begin{abstract}
We give a single manifestly finite, regulator-free functional realizing in one object L-functions
quasi-periods and period polynomials of meromorphic modular forms on $\SLZ$, with generic 
poles throughout the upper half-plane $\HH$ and at the cusp of $\SLZ$. Additionally, our construction
preserves holomorphicity. It is a two-segment contour integral,
$L^*(f,s) = L^*_T(f, s; \gamma^T_{\tau_0}) + L^*_S(f, s; \gamma^S_{\tau_0})$, where $L^*_T$ and $L^*_S$ 
have distinct integration kernels, and 
$\gamma^{S}_{\tau_0}$ is a path in $\HH$ from $S(\tau_0)$ to $\tau_0$, and the same for $\gamma^T_{\tau_0}$. 
For weakly holomorphic forms our L-function is independent of $\tau_0$ and of the path, and reproduces 
classical L-functions for holomorphic modular forms, reproduces the L-functions of weak modular forms 
in~\cite{BFK2011}, which previously needed non-holomorphic regularization to be defined, and reproduces 
Brown's quasi-periods~\cite{Brown2017}. For properly meromorphic modular forms with poles inside the 
interior of $\HH$, our L-function depends on the homotopy classes, defined in terms of pole-crossings. 
There is a natural choice of trivial homotopy class, which renders our L-functions canonical when the 
modular form has a pole in the interior of the fundamental domain, generalizing~\cite{1806}.
\end{abstract}

\section{Introduction and preliminary concepts}\label{sec:intro}

Our primary goal in this note is to put forth a unifying generalization of the L-function 
associated to a given modular form. Our L-function naturally encompasses the classical 
case of L-functions associated to holomorphic modular forms~\cite{01-Apostol}, and 
seamlessly unifies Ref.~\cite{BFK2011}'s extension to L-functions of modular forms with 
poles at the cusp with Ref.~\cite{1806}'s extension to L-functions of modular forms with 
poles at finite locations, exclusively away from cusps. Secondarily, we show that our 
L-functions reproduce the numerical results for quasi-periods associated to weak cusp 
forms~\cite{Brown2017}. 

{\em Note:} Throughout this document, we refer to L-integrals and L-functions. The classical 
L-function-as-Dirichlet-series which has been suitably extended to the complex $s$ plane is 
simply not defined when the modular form that defines the L-function has poles either at the 
cusp or at finite locations within the upper half-plane. So, if there is ever a doubt, we state clearly: 
{\em any L-function in this note is defined by an L-integral}. The focus of this note is to define 
L-integrals for any integer weight $\SLZ$ modular form which may have poles at cusps or away 
from cusps.

\subsection{Modular forms}
Write ${\rm Re}(z)$ and ${\rm Im}(z)$ for the real and imaginary parts of a complex number 
$z\in \CC$. Let $\HH:=\{\tau\in\CC\mid {\rm Im}(\tau)>0\}$ denote the upper half-plane. $\SLZ$ 
is the group generated by $T := \left( \begin{smallmatrix} 1 & 1 \\ 0 & 1 \end{smallmatrix} \right)$ 
and $S := \left( \begin{smallmatrix} 0 & -1 \\ 1 & 0 \end{smallmatrix} \right)$. 

There is a natural $\SLZ$ action on functions $f: \HH \to \CC$, graded by integers $k$: 
$(f|_k \gamma)(\tau) := (c \tau + d)^{-k}f(\gamma\cdot\tau)$ for all $\gamma = \left(\begin{smallmatrix}
a & b \\ c & d \end{smallmatrix}\right) \in \SLZ$. Similarly, $\gamma\cdot \tau := 
\frac{a \tau + b}{c \tau + d}$. 

In this vein, a {\em holomorphic modular form} of weight $k$ for $\SLZ$ is a 
holomorphic function $f:\HH\to \CC$ such that $(f|_k\gamma)(\tau) = f(\tau)$ 
for all $\gamma \in \SLZ$ (aka $f(\tau+1)=f(-\frac1\tau)\tau^{-k}=f(\tau)$) for 
all $\tau\in \HH$, and $f(\tau)={\cal~O}(\tau^k)$ as ${\rm Im}(\tau)\to \infty$, and 
which has no poles at finite locations within $\HH$. By virtue of the periodicity 
$f(\tau) = f(\tau+1)$ and holomorphicity, each holomorphic modular form has 
a q-series decomposition,
\begin{equation}
f(\tau) = \sum_n c_f(n) e^{2 \pi i n \tau} = \sum_n c_f(n) q^n~~,~~q:= e^{2 \pi i \tau}~~.
\end{equation}
In order for $f$ to be bounded as $\tau \to i \infty$, we conclude that $c_f(n) = 0$ 
for all $n < 0$, i.e. that $f = \sum_{n \geq 0} c_f(n) q^n$; for such modular forms $f$ 
we write $f \in M_k$. If a modular form $f$ additionally has a vanishing constant term, 
i.e. if $c_f(0) = 0$, then $f$ is a {\em cusp form}, 
and we write $f \in S_k$. Clearly, $S_k \subset M_k$. Further, $M_k$ is a  
finite-dimensional vector-space, as is $S_k$. Holomorphic $\SLZ$ modular forms must 
have even integer weights, and $k \geq 4$~\cite{01-Apostol}. 

A product of modular forms of weights $k_1$ and $k_2$ is a modular form of weight 
$k_1+k_2$, so $M_*:=\bigoplus_k  M_k$ is naturally a (graded) ring. This graded ring is 
freely generated by the two modular forms $E_4 \in M_4$ and $E_6 \in M_6$, the $k = 4, 6$
specializations of the {\em weight $k$ Eisenstein series} $E_k$, which may be defined as follows:
\begin{align}
E_k := \frac{1}{2}\sum_{(m,n)=1} \frac{1}{(m+n\tau)^k} = 1 - \frac{2k}{B_k}\sum_{n = 1}^{\infty} \sigma_{k-1}(n) q^n~,
\end{align}
where $\sum_{(m,n) = 1}$ denotes summing over all relatively prime $(m,n) \in \ZZ^2$, the Bernoulli 
number $B_n$ is the coefficient of $t^n/n!$ of the Laurent expansion of $t/(e^{t}-1)$ about $t = 0$, 
and $\sigma_m(n) := \sum_{d|n} d^m$ is the Divisor function where $d|n$ means $n/d \in \NN$. 
In short, $M_* = \CC[E_4, E_6]$. We now define the {\em modular discriminant}:
\begin{equation}
\Delta := \frac{E_4^3-E_6^2}{1728} = q + O(q^2) \in S_{12}~.
\end{equation}

Holomorphicity is most canonically relaxed via allowing a modular form to have a pole at the 
cusp of $\SLZ$, i.e. for a weight $k$ modular form for $\SLZ$ to have negative powers of $q$ 
in its q-series expansion. In this case, $f$ diverges as $\tau \to i \infty$ as $f \sim e^{-2 \pi N \tau}$, 
where $-N$ is the largest negative number such that $c_f(n)$ is non-zero. Such a modular form 
$f$ is called a {\em weak modular form}, and we write $f \in M_k^!$. If a weak modular form $f$ 
has a vanishing coefficient of $q^0$, i.e. if $c_f(0) = 0$, it is called a {\em weak cusp form} and 
we write $f \in S_k^!$. Clearly, $S_k^! \subset M_k^!$, $S_k \subset S_k^!$ and $M_k \subset M_k^!$. 
Two important weak cusp forms in this note are as follows: 
\begin{align}
J :=&~ \frac{E_4^3}{\Delta} - 744 = \frac{1}{q} + O(q) \in S_0^!~,~\\
\Dhat =&~ (J^2 + 24 J -393444) \Delta := \frac{1}{q} + O(q^2) \in S_{12}^!~.~
\end{align}
In this work, we further relax from even weakly holomorphic modular forms, and 
consider {\em meromorphic modular forms} of weight $k$ which are quotients: $f=\frac{f_1}{f_2}$ 
where $f_i\in M_{k_i}$, the denominator $f_2$ is not identically zero, and $k=k_1-k_2$. We 
write $F_k$ for the vector space of meromorphic modular forms of weight $k$, and set 
$F_*:=\bigoplus_k F_k$. Then $F_k$ vanishes unless $k$ is an even integer, and 
$F_*$ may be regarded as a ``graded field of fractions'' of $M_*$. 

\subsection{L-functions for holomorphic modular forms and periods for cusp forms}

For any $f \in S_k$, one may define the associated L-function:
\begin{align}
L(f,s) := \frac{(2 \pi)^s}{\Gamma(s)} L^*(f,s) \underset{f \in S_k}{:=} 
\frac{(2 \pi)^s}{\Gamma(s)} \int_0^{\infty} f(it) t^{s-1} dt = \sum_{n \geq 1} \frac{c_f(n)}{n^s}~.
\end{align}
A similar definition applies for $f \in M_k$. (The classical definition of $L^*(f,s)$ for $f \in M_k$ 
is the focus of Remark~\ref{natural-beauty-via-finiteness}, below.) Similarly, one may define a period 
polynomial for $f \in S_k$ in terms of this L-function,
\begin{align}
r_f(X, Y) := (2 \pi i)^{k-1} \sum_{\ell = 0}^{k-2} (-1)^\ell \binom{k-2}{\ell}\, i^{\ell+1}\, L^*(f,\ell+1)\,X^{k-2-\ell}Y^\ell~.
\end{align}
This polynomial, $r_f(X, Y) \in V_{k-2} := \CC[X, Y]_{k-2}$, has a natural action under $\SLZ$, 
$(P|_{\gamma})(X, Y) = P(a X + b Y, c X + d Y)$ for $\gamma = 
\left( \begin{smallmatrix} a & b \\ c & d \end{smallmatrix} \right) \in \SLZ$. This period polynomial 
can be decomposed into a basis of highly constrained polynomials, which are annihilated by $|(1+S)$
and $|(1 + U + U^2)$\cite{KZ1984}, where $U := TS$ generates an order-3 cyclic subgroup of $\SLZ$.
We define the \emph{periods} of a cusp form to be coefficients of $r_f$ within this highly 
constrained sub-space $W$ of $V_{k-2}$, i.e. $W \subset V_{k-2}$.

{\bf Why this work?} As classically computed, these L-functions and associated period polynomials 
(when defined in terms of the special values of the associated L-function) are not defined when $f$ 
has poles either at the cusp of $\SLZ$ or within the interior of $\HH$. As stated above, there 
are extant, but partial and non-overlapping, extensions of the definition of $L(f,s)$ to $f \in M_k^!$~\cite{BFK2011} or 
$f \in F_k$ when $f$ is regular at the cusp of $\SLZ$~\cite{1806}. In this note, we provide a construction that unifies these disparate extensions.
We start with generic L-integrals in section~\ref{sec:Lint}, then specialize to $f \in M_k^!$ in section~\ref{sec:Lfun}, and finally explore the more general context of $f \in F_k$ in section~\ref{sec:mero}.

\section{A finite L-integral}\label{sec:Lint}


\begin{definition}\label{def:segments}
Let $\tau_0\in\HH$ and $f\in F_k$, let
$S=\left(\begin{smallmatrix}0&-1\\1&0\end{smallmatrix}\right)$,
$T=\left(\begin{smallmatrix}1&1\\0&1\end{smallmatrix}\right)$ denote 
the generators of $\Gamma=\SLZ$, and let $f^{-1}(\infty)$ be the locus 
of all poles of $f \in \HH$ (inclusive of the cusp of $\SLZ$).  
The \emph{$S$- and $T$-segments at $\tau_0$} are smooth paths in
$\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$ from $\sigma^{-1}\tau_0$ to $\tau_0$, for
$\sigma=S$ and $\sigma=T$ respectively,
\begin{equation}
\gamma^S_{\tau_0}:\ -1/\tau_0\ \longrightarrow\ \tau_0,\qquad
\gamma^T_{\tau_0}:\ \tau_0-1\ \longrightarrow\ \tau_0.
\end{equation}
\end{definition}

Avoiding poles is automatic for $f \in M_k^!$, as long as the base-point $\tau_0$, 
is not at a cusp of $\SLZ$, i.e. there is only the trivial homotopy class. 
When $f \in F_k$, and $f$ has poles at finite and non-zero imaginary heights, 
integration contours fall into nontrivial homotopy classes.

\begin{definition}\label{def:kernel}
Let $\zeta(s,x)=\sum_{m\ge0}(m+x)^{-s}$ as a function 
$\zeta: \CC_s \times (\CC\setminus\NN)_x \to \CC$ denote the 
Hurwitz zeta function (meromorphic in $s$, holomorphic in $x\in\HH$). 
We define \emph{Hurwitz $T$-kernel} as
\begin{equation}
\tk(\tau,s):=\zeta(1-s,\tau+1)-e^{i\pi(s-1)}\,\zeta\big(1-(k-s),\tau+1\big).
\end{equation}
\end{definition}

\begin{definition}[Generic L-integral]
\label{def:Lint}
Let $f\in F_k$, $\tau_0\in\HH$, and $s\in\CC$. Further, let $\gamma^S_{\tau_0}$,
$\gamma^T_{\tau_0}$ be the $S$- and $T$-segments at $\tau_0$, as in
Definition~\ref{def:segments}, and let $\tk$ be as defined in 
Definition~\ref{def:kernel}. The \emph{L-integral} of $f$ is
\begin{equation}
L^*(f,s;\gamma^S_{\tau_0},\gamma^T_{\tau_0})
:= L^*_S(f,s; \gamma^S_{\tau_0}) + L^*_T(f,s; \gamma^T_{\tau_0})
:=e^{-\pi i s/2}\left(\int_{\gamma^S_{\tau_0}}\!\!\!\! f(\tau)\tau^{s-1}\,d\tau
+\int_{\gamma^T_{\tau_0}}\!\!\!\! f(\tau)\tk(\tau,s)d\tau\right)
\end{equation}
\end{definition}
We use this \emph{L-integral} to define a generalized 
\emph{L-function}:

\begin{definition}[Generic L-function]\label{def:Lfun}
Let $L^*(f,s;\gamma^S_{\tau_0}, \gamma^T_{\tau_0})$ be as defined in Definition~\ref{def:Lint}. 
We define the generalized \emph{L-function} of $f\in F_k$ to be,
\begin{equation}
L(f,s;\gamma^S_{\tau_0},\gamma^T_{\tau_0})
:=\frac{(2\pi)^s}{\Gamma(s)}\,L^*(f,s;\gamma^S_{\tau_0},\gamma^T_{\tau_0}).
\end{equation}
\end{definition}

Having defined the L-integral and L-function, we prove they are manifestly finite for $f \in F_k$: 

\begin{lemma}[Finiteness of L-integrals]\label{lem:finite}
For $f\in F_k$, both $L^*_S(f, s; \gamma^S_{\tau_0})$ and 
$L^*_T(f, s; \gamma^T_{\tau_0})$ in Definition~\ref{def:Lint} are each, 
separately finite for every $s\in\CC$, every $\tau_0\in\HH$, and for 
every path $\gamma^S_{\tau_0},\gamma^T_{\tau_0}$.
\end{lemma}

\begin{proof}
Both segments are compact and lie in $\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$, so $f$ is
continuous on them; $\tau^{s-1}$ is continuous on $\HH$, and $\tk(\tau,s)$ is continuous there
since $\zeta(\nu,\tau+1)$ is holomorphic in $\tau\in\HH$ (Definition~\ref{def:kernel}). A
continuous integrand on a compact path is integrable, for every $s\in\CC$.
\end{proof}

We now turn to path-dependence and $\tau_0$-independence of 
the L-integral and the L-function of $f \in F_k$. 
This holds for all $f \in M_k^!$:

\begin{proposition}[Basepoint \& path independence for $f \in M_k^!$]\label{prop:indep}
For $f\in M^!_k$ the value $L^*(f,s;\gamma^S_{\tau_0},\gamma^T_{\tau_0})$ is independent
of $\tau_0\in\HH$ and of the paths: with no interior poles, $\HH$ is simply connected,
so all admissible contours are homotopic. We write $L^*(f,s)$ for the common value and
$L(f,s):=\frac{(2\pi)^s}{\Gamma(s)}L^*(f,s)$.
\end{proposition}

\begin{proof}
With no interior poles the integrands $f(\tau)\tau^{s-1}$ and $f(\tau)\tk(\tau,s)$ are
holomorphic on $\HH$, which is simply connected; by Cauchy's theorem each integral is unchanged
under any deformation fixing its endpoints, hence depends on those endpoints alone. This is
path-independence.

For independence of $\tau_0$ show that variations in $\tau_0$ for the S-term and T-term in $L^*(f,s)$ 
cancel, i.e. that $\partial_{\tau_0} L^*_S(f,s:\gamma^S_{\tau_0}) = -\partial_{\tau_0} L^*_T(f,s:\gamma^T_{\tau_0})$. 
For $L^*_T$, note that the $T$-segment runs from $T^{-1}\tau_0=\tau_0-1$ to $\tau_0$, 
so by Leibniz's rule and the $T$-periodicity $f(\tau_0-1)=f(\tau_0)$,
\begin{equation}
\partial_{\tau_0} L^*_T(f, s; \gamma^T_{\tau_0})
=f(\tau_0)\,\tk(\tau_0,s)-f(\tau_0-1)\,\tk(\tau_0-1,s)
=f(\tau_0)\,\delta_T\tk(\tau_0-1,s),
\end{equation}
where $\delta_T\tk(\tau,s):=\tk(\tau+1,s)-\tk(\tau,s)$. This is the whole role of $\delta_T$: the
unit gap between the endpoints turns the $\partial_{\tau_0}$ boundary term into a unit shift of
the kernel. That shift is elementary, since $\zeta(\nu,x+1)-\zeta(\nu,x)=-x^{-\nu}$ gives, on 
both terms at $x=\tau+1$,
\begin{align}
& \delta_T\tk(\tau,s)=-(\tau+1)^{s-1}+e^{i\pi(s-1)}(\tau+1)^{k-1-s}
\\
& \implies \partial_{\tau_0} L^*_T(f, s; \gamma^T_{\tau_0}) =f(\tau_0)\big[-\tau_0^{s-1}+e^{i\pi(s-1)}\tau_0^{k-1-s}\big].
\end{align}
The $S$-segment runs $-1/\tau_0\to\tau_0$; its lower endpoint moves with
$\partial_{\tau_0}(-1/\tau_0)=\tau_0^{-2}$, so by Leibniz, the modularity
$f(-1/\tau_0)=\tau_0^k f(\tau_0)$, and the phase $(-1/\tau_0)^{s-1}=e^{i\pi(s-1)}\tau_0^{1-s}$,
\begin{equation}
\partial_{\tau_0} L^*_S(f, s; \gamma^S_{\tau_0}) 
=f(\tau_0)\tau_0^{s-1}-f(-1/\tau_0)\,(-1/\tau_0)^{s-1}\,\tau_0^{-2}
=f(\tau_0)\big[\tau_0^{s-1}-e^{i\pi(s-1)}\tau_0^{k-1-s}\big].
\end{equation}
Thus $\tau_0$-variations are equal and opposite, $\partial_{\tau_0}L^*_S = - \partial_{\tau_0}L^*_T$. Thus 
$\partial_{\tau_0}L^*=\partial_{\tau_0}L^*_S + \partial_{\tau_0}L^*_T=0$.
\end{proof}

\begin{remark}[Recovering the classical L-integral for $f \in S_k$]\label{rem:mellin}
\emph{Let $f\in S_k$. As $f^{-1}(\infty)$ does not include the cusp $i \infty$, 
$\lim_{\tau_0 \to i \infty} L^*(f, s; \gamma^S_{\tau_0}, \gamma^T_{\tau_0})$ exists. 
In this limit, the $S$-segment maps onto the standard 
L-function contour, $[0,i\infty]$. Simultaneously, as $f(it) \sim e^{-2\pi t}$ as 
$t \to \infty$, the $T$-segment vanishes exponentially. Thus, in this limit, $L^*(f,s)$ 
exactly maps onto the classical L-function integral~\cite{01-Apostol}:}
\begin{equation}
\lim_{\tau_0\to i\infty}L^*(f,s)=\int_0^{\infty}\! t^{s-1}f(it)\,dt.
\end{equation}
\end{remark}

\section{L-functions, period polynomials and poles at the cusp}\label{sec:Lfun}

\begin{proposition}\label{prop:HGG}
Let $\Gamma(s, x) := \int_x^{\infty} e^{-t} t^{s-1} dt$ denote the {\em incomplete Gamma function}. For $n\ne0$, $\nu\in\CC$, and $z\in\HH$,
\begin{equation}
\int_{z}^{z+1}\! e^{2\pi inu}\,\zeta(\nu,u)\,du=e^{i \pi (1-\nu)/2} (2\pi n)^{\nu-1}\,\Gamma\big(1-\nu,-2\pi i n z\big).
\end{equation}
\end{proposition}

\begin{proof}
Write $\zeta(\nu,u)=\sum_{m\ge0}(m+u)^{-\nu}$ and shift $u\mapsto u+m$ in the $m$-th term; as
$e^{2\pi in(u+m)}=e^{2\pi inu}$ for integer $m$, the unit segments stack into one horizontal ray,
\begin{equation}
\int_z^{z+1}\! e^{2\pi inu}\zeta(\nu,u)\,du
=\sum_{m\ge0}\int_{z+m}^{z+1+m}\! e^{2\pi inv}v^{-\nu}\,dv
=\int_z^{z+\infty}\! e^{2\pi inv}v^{-\nu}\,dv.
\end{equation}
Now, substitute $w=-2\pi i n v$, so that $v=iw/(2\pi n)$, $dv=i\,dw/(2\pi n)$, and $e^{2\pi inv}=e^{-w}$:
the lower endpoint becomes $w=-2\pi i n z$, and for $n>0$ the ray rotates to $w\to+\infty$ along the
positive reals. Since $e^{-w}w^{-\nu}$ decays in the right half-plane, the closing arc
vanishes. With $v^{-\nu}=i^{-\nu}(2\pi n)^{\nu}w^{-\nu}$,
\begin{equation}
\int_z^{z+\infty}\! e^{2\pi inv}v^{-\nu}\,dv
=i^{1-\nu}(2\pi n)^{\nu-1}\!\int_{-2\pi i n z}^{\infty}\! e^{-w}w^{-\nu}\,dw
=e^{i \pi (1-\nu)/2}(2\pi n)^{\nu-1}\Gamma\big(1-\nu,-2\pi i n z\big).
\end{equation}
For $n<0$ the ray instead grows as ${\rm Im} v\to\infty$ and the rotation is illegal; but both sides are
analytic in $n$ and agree for $n>0$, so the identity continues to $n<0$ with $\Gamma$ on its principal
branch. At $n=0$ the prefactor $(2\pi n)^{\nu-1}$ is singular; that case is handled separately.
\end{proof}

\begin{theorem}[The L-function for $f \in M_k^!$]\label{thm:weakL}
For $f\in M^!_k$, we may set $\tau_0=i$, and prove
\begin{equation}\label{eq:weakL}
L^*(f,s)=-c_f(0)\left(\frac1s+\frac{i^k}{k-s}\right)
+\sum_{n\ne0}c_f(n)\left(\frac{\Gamma(s,2\pi n)}{(2\pi n)^s}+i^k\frac{\Gamma(k-s,2\pi n)}{(2\pi n)^{k-s}}\right)~.
\end{equation}
Here, trivially, $L^*(f,s) = i^k L^*(f,k-s)$. The L-function is $L(f,s) := (2\pi)^s L^*(f,s) /\Gamma(s)$.
\end{theorem}

\begin{proof}
By path independence and base-point independence from Proposition~\ref{prop:indep}, 
we may freely set $\tau_0=i$. Here, $L^*_S(f, s; \gamma^S_{\tau_0}) = 0$ (the $S$-contour 
vanishes in length and there are no nontrivial homotopy classes as $f \in M_k^!$ has no poles 
which a finite contour can wrap), and 
the L-integral simplifies dramatically:
\begin{equation}
L^*(f,s) = L^*_T(f, s; \gamma^T_i) =e^{-\pi i s/2}\int_{i-1}^{i} f(\tau)\tk(\tau, s)d\tau~. 
\end{equation}
We then expand $f$ in terms of its q-series, $\sum_n c_f(n)e^{2\pi in\tau}$,
and substitute the $u$-variable of Proposition~\ref{prop:HGG} as 
$u=\tau+1$, and note that $e^{2\pi i n(u-1)}=e^{2\pi i n u}$. 
When $n\neq0$, straightforward substitutions of 
$\nu, z$ into Proposition~\ref{prop:HGG} suffice: $1-\nu=s$ gives the 
contribution of the first term in $\tk(\tau, s)$, and $1-\nu = k-s$ gives the second: 
\begin{align}
e^{-\pi i s/2}\left(e^{\pi i s/2} \frac{\Gamma(s,2\pi n)}{(2\pi n)^s} - e^{i\pi(s-1)}e^{\pi i (k-s)/2} \frac{\Gamma(k-s,2\pi n)}{(2\pi n)^{k-s}}\right) = \frac{\Gamma(s, 2\pi n)}{(2\pi n)^s} + i^k \frac{\Gamma(k-s, 2\pi n)}{(2\pi n)^{k-s}}~.
\end{align}
We may then evaluate the $n=0$ mode's contribution follows from noting the 
closed form solution for the definite integral, 
\begin{equation}\label{eq:hurwitz-antideriv}
\int_{i - 1}^{i}\! \zeta(\sigma, \tau + 1)\,d\tau
 = \int_i^{i + 1}\!\zeta(\sigma, u)\,du
 = \frac{i^{1 - \sigma}}{\sigma - 1}.
\end{equation}
This result follows from noting $\partial_a(\zeta(\sigma-1, a)/(1-\sigma)) = \zeta(\sigma,a)$ 
and $\zeta(\sigma-1, a+1) - \zeta(\sigma-1, a) = -a^{1-\sigma}$. 
Applying this to the two distinct terms of $\widetilde{k}_T(\tau, s)$ at $\sigma = 1 - s$ and $\sigma = 1 - (k - s)$ gives
\begin{equation}\label{eq:const-mode-integral}
e^{-\pi i s/2}\int_{i - 1}^{i}c_f(0)\widetilde{k}_T(\tau, s)\,d\tau
= -c_f(0)\left(\frac{1}{s}  + \frac{i^k}{k - s}\right).
\end{equation}
Combining the $n = 0$ contributions with the $n \neq 0$ contributions reproduces Eq.~\eqref{eq:weakL}. 
Finally, the functional equation, $L^*(f,s)=i^k L^*(f,k-s)$, follows by direct inspection.
\end{proof}

\subsection{Explicit S- and T-contours for general $\tau_0$}

\begin{lemma}[$L^*_T$ for $M^!_k$]\label{lem:Tclosed-Mk}
For $f\in M^!_k$, and $\tau_0 \in \HH$ away from cusps, 
$L^*_T(f, s; \gamma^T_{\tau_0})$ evaluates to:
\begin{equation}\label{eq:T-segment-closed-Mk}
L_T^*(f, s; \gamma^T_{\tau_0})
= -c_f(0)\!\left(\frac{(\tau_0/i)^{s}}{s} + i^k\frac{(\tau_0/i)^{k-s}}{k-s}\right) + \sum_{n \ne 0}\! c_f(n)\!\left(\frac{\Gamma(s, -2\pi i n\tau_0)}{(2\pi n)^s} + i^k\frac{\Gamma(k - s, -2\pi i n\tau_0)}{(2\pi n)^{k-s}}\right).
\end{equation}
\end{lemma}
\begin{proof}
Expand $f = \sum_n c_f(n)\,q^n$ on the compact segment
$[\tau_0 - 1, \tau_0]$, and restrict attention to the sub-sum, $\sum_{n \neq 0}$. 
For each such $n \neq 0$, apply the Proposition~\ref{prop:HGG} in
$\widetilde{k}_T(\tau, s) = \zeta(1 - s, \tau + 1) - e^{i\pi(s-1)}\zeta(s - (k-1), \tau + 1)$
at $\nu = 1 - s$ and $\nu = s - (k - 1)$. The phases for the second term work-out 
identically as in the proof of Theorem~\ref{thm:weakL}.  

For $n = 0$, the constant $c_f(0)$ multiplies $e^{-\pi i s/2}\!\int_{\tau_0 - 1}^{\tau_0}\!\widetilde k_T(\tau, s)\,d\tau$, 
which we compute as follows. First, the closed-form of the Hurwitz antiderivative 
$\int_{\tau_0 - 1}^{\tau_0}\zeta(\sigma, \tau + 1)\,d\tau = \tau_0^{1 - \sigma}/(\sigma - 1)$
comes from $\partial_\tau\zeta(\sigma - 1, \tau + 1) = -(\sigma - 1)\zeta(\sigma, \tau + 1)$ 
and the shift identity $\zeta(\sigma - 1, \tau_0 + 1) - \zeta(\sigma - 1, \tau_0) = -\tau_0^{1 - \sigma}$. 
Finally, note that the two terms in $\tk$ are simply specializations of $\sigma = 1 - s$ and $\sigma = s - k + 1$, 
which yield $-e^{-\pi i s/2}\tau_0^{\,s}/s = -(\tau_0/i)^s/s$ and 
$-e^{\pi i s/2}\tau_0^{\,k-s}/(k-s) = -i^k(\tau_0/i)^{k-s}/(k-s)$ respectively.
%For $n = 0$, the constant $c_f(0)$ multiplies $e^{-\pi i s/2}\!\int_{\tau_0 - 1}^{\tau_0}\!\widetilde k_T(\tau, s)\,d\tau$;
%the Hurwitz antiderivative $\int_{\tau_0 - 1}^{\tau_0}\zeta(\sigma, \tau + 1)\,d\tau = \tau_0^{1 - \sigma}/(\sigma - 1)$
%--- from $\partial_\tau\zeta(\sigma - 1, \tau + 1) = -(\sigma - 1)\zeta(\sigma, \tau + 1)$ and the shift
%$\zeta(\sigma - 1, \tau_0 + 1) - \zeta(\sigma - 1, \tau_0) = -\tau_0^{1 - \sigma}$ --- applied to the two
%halves of $\widetilde k_T$ at $\sigma = 1 - s$ and $\sigma = s - k + 1$ yields
%$-e^{-\pi i s/2}\tau_0^{\,s}/s = -(\tau_0/i)^s/s$ and $-e^{\pi i s/2}\tau_0^{\,k-s}/(k-s) = -i^k(\tau_0/i)^{k-s}/(k-s)$ respectively.
\end{proof}


\begin{corollary}[$L^*_T$ as $\tau_0 \to i \infty$]\label{lem:cusp-decay}
Let $f \in S^!_k$, where $f = \sum_{n \geq -N} c_f(n)q^n$ for some $-N \in \ZZ$. 
The T-contour's contribution to $L^*(f,s)$ is given by 
Lemma~\ref{lem:Tclosed-Mk}. As $\tau_0 = it$ goes to the cusp $i\infty$,
the $T$-contour scales exactly as $f(it)$ scales:
$L_T^*(f, s; \gamma^T_{\tau_0}) \sim f(it) \sim e^{2\pi N t}$.
\end{corollary}
\begin{proof}
For the asymptotics, $\Gamma(\sigma, z) \sim z^{\sigma - 1}e^{-z}$
as $|z| \to \infty$ for $|\arg z| < \tfrac{3\pi}{2}$ \cite[\S8.11]{DLMF}
gives $|c_f(n)|$-weighted go as $e^{-2\pi n\,{\rm Im}(\tau_0)}$.  The smallest 
mode $n$ in the q-series, i.e. $n = -N$, dominates the scaling of $f(it)$ and of 
$L^*_T(f, s; \gamma^T_{i t})$. Combining these results yields the claimed asymptotics.
\end{proof}

Now, exploiting $\tau_0$-independence of $L^*(f, s)$, its explicit value as given in 
Theorem~\ref{thm:weakL}, and the explicit T-contribution $L^*_T(f, s; \gamma^T_{\tau_0})$ trivially 
implies clean, closed-form expressions for $L^*_S(f, s; \gamma^S_{\tau_0})$:

\begin{corollary}[$L^*_S$ for $M_k^!$]\label{cor:Sclosed}
For $f\in M^!_k$ and $\tau_0\in\HH$, 
$L^*_S(f, s; \gamma^S_{\tau_0})$ evaluates to
\begin{align}
L_S^*(f,s;\gamma^S_{\tau_0})
&= c_f(0)\left(\frac{(\tau_0/i)^{\,s}-1}{s}
+ i^k\frac{(\tau_0/i)^{\,k-s}-1}{k-s}\right) \label{eq:S-segment-explicit}\\
&+ \sum_{n\ne0} c_f(n)\!\left(\frac{\Gamma(s,2\pi n)-\Gamma(s,-2\pi i n\tau_0)}{(2\pi n)^s}
+ i^k\,\frac{\Gamma(k-s,2\pi n)-\Gamma(k-s,-2\pi i n\tau_0)}{(2\pi n)^{k-s}}\right)~. \nonumber
\end{align}
\end{corollary}
\begin{proof}
Recall $L^*(f, s) - L^*_T(f, s; \gamma^T_{\tau_0}) = L^*_S(f, s; \gamma^S_{\tau_0})$. 
Now, $L^*(f,s)$ for $f \in M_k^!$ is given by Theorem~\ref{thm:weakL}, and 
$L^*_T(f, s; \gamma^T_{\tau_0})$ is given by Lemma~\ref{lem:Tclosed-Mk}. 
Their difference yields Eq.~\eqref{eq:S-segment-explicit}.
\end{proof}


\subsection{Classical Mellin recovery and canonical uniqueness.}\label{sec:Connection-w-previous-weak-MF-l-functions}

As discussed in Remark~\ref{rem:mellin}, for $f \in S_k$, taking the cusp-limit
$\tau_0 \to i\infty$ in Theorem~\ref{thm:weakL} collapses the $S$-segment 
onto the classical Mellin contour $[0, i\infty]$ while via Corollary~\ref{lem:cusp-decay} 
the $T$-segment vanishes exponentially as $L^*_T(f, s; \gamma^T_{\tau_0}) \sim f(\tau_0) 
\sim e^{-2\pi {\rm Im}(\tau_0) N}$ (as $f \in S_k \Leftrightarrow f = \sum_{n \geq N} c_f(n) q^n$ with $N \geq 1$). 
I.e., here, one recovers the classical L-integral of $f$ without obstruction:
\begin{equation}\label{eq:Mellin-recovery}
\lim_{\tau_0 \to i\infty}\!
L^*(f, s; \gamma^S_{\tau_0}, \gamma^T_{\tau_0})
\;=\; \int_0^\infty\! t^{s - 1}\,f(it)\,dt~.
\end{equation}  
This story changes when $f = 1/q^N (1 + O(q)) \in S_k^!$ for some $N \geq 1$. 

In short, when $f$ has a polar q-series, it has a pole at the cusp of $\SLZ$. Thus, 
the limit $\tau_0 \to i \infty$, which was crucial to recovering the classical Mellin contour 
in Remark~\ref{rem:mellin} simply does not exist.

However, it can still be instructive to examine $L^*_T(f, s; \gamma^T_{\tau_0})$, 
and $L^*_S(f, s; \gamma^S_{\tau_0})$ as $\tau_0$ goes towards $i \infty$. 
In this context, while of course the sum $L^*_T(f, s; \gamma^T_{\tau_0}) + 
L^*_S(f, s; \gamma^S_{\tau_0}) = L^*(f,s)$ is manifestly invariant under 
changes of the base-point, each of these two terms can indeed change values. Indeed, 
Corollary~\ref{lem:cusp-decay} clearly shows that as $\tau_0 \to i \infty$, when $f \in S_k^!$ 
has negative powers of $q$ in its q-series, then $L^*_T(f, s; \gamma^T_{\tau_0}) \sim 
e^{2 \pi N {\rm Im}(\tau_0)}$. Discarding $\gamma^T_{\tau_0}$ therefore does violence to the result: 
When the $S$-contour is isolated from its T-partner, and $\tau_0 \to i \infty$, it diverges. This is the precise
reason that~\cite{BFK2011} needed to regularize their divergent L-integral representation for properly weak 
cusp forms. 

That said, there is one further aspect of our L-function even in the classical 
case of $f \in M_k$ that is worthy of remark:

\begin{remark}[Recovering the classical L-integral for $f \in M_k$]\label{natural-beauty-via-finiteness}
\emph{In textbook treatments~\cite{01-Apostol}, when $f \in M_k$, its L-integral is defined as}
\begin{align}
L^*(f,s)|_{\rm cl} :=&~ \int_0^\infty \big( f(it) - c_f(0) \big) t^{s-1} dt = \frac{\Gamma(s)}{(2 \pi)^s} \sum_{n = 1}^{\infty} \frac{c_f(n)}{n^s} \\
=& -c_f(0) \bigg(\frac{1}{s} + \frac{i^k}{k-s} \bigg) + \sum_{n = 1}^{\infty} c_f(n)\left(\frac{\Gamma(s,2\pi n)}{(2\pi n)^s} + i^k \frac{\Gamma(k-s, 2\pi n)}{(2\pi n)^{k-s}}\right) \nonumber
\end{align}
\emph{Classically, this constant-term subtraction is needed because $\int_0^\infty t^{s-1} dt$ is a pure 
divergence, so the q-series cannot commute with the integration contour.}

\emph{However, as is clear from our general Theorem~\ref{thm:weakL} for L-integrals for $f \in M_k^!$, 
and from the specific results of the S- and T-contours in Lemma~\ref{lem:Tclosed-Mk} and 
Corollary~\ref{cor:Sclosed}, our L-integral requires no such explicit subtraction of the constant 
term $-c_f(0)$. Thus, not only does our prescription naturally handle divergences at poles when 
$f \in M_k^!$ has polar terms in its q-series, but it also obviates the need to by-hand remove the 
constant mode from the kernel of the L-integral that is needed when defining L-functions in terms 
of the classical Mellin contour, $\int_0^\infty$.}
\end{remark}

\subsection{Period polynomials, periods and quasi-periods}

Because $L^*(f,s)$ is canonical for every $f\in S^!_k$, and because period polynomials 
for a given modular form can be written in terms of their L-functions evaluated at integer-$s$ 
in the critical strip, they need no separate construction.

\begin{definition}[Period polynomial for $S_k^!$]\label{def:rf}
For $f\in S^!_k$ and $n=k-2$, the \emph{period polynomial} is
\begin{equation}
r_f(X,Y):=(2\pi i)^{n+1}\sum_{\ell=0}^{n}(-1)^\ell\binom{n}{\ell}\,i^{\ell+1}\,L^*(f,\ell+1)\,X^{n-\ell}Y^\ell.
\end{equation}
\end{definition}

\begin{proposition}[Period polynomial as a contour integral]\label{prop:periodint}
Let $n:=k-2$. With $(X-\tau Y)^n$ in place of the L-weight $\tau^{s-1}$, and
$\mathbf K_T(\tau;X,Y):=\sum_{\ell=0}^{n}\binom{n}{\ell}(-1)^\ell\,\tk(\tau,\ell+1)\,X^{n-\ell}Y^\ell$
in place of $\tk$, the same two segments assemble $r_f$ directly:
\begin{equation}\label{eq:rf}
r_f(X, Y)=(2\pi i)^{n+1}\Big[\int_{\gamma^S_{\tau_0}} f(\tau)\,(X-\tau Y)^n\,d\tau
+\int_{\gamma^T_{\tau_0}} f(\tau)\,\mathbf K_T(\tau;X,Y)\,d\tau\Big].
\end{equation}
\end{proposition}
\begin{proof}
By Definition~\ref{def:Lint}, the $X^{n-\ell}Y^\ell$ coefficient of the bracket is
\begin{equation}
r_f|_{X^{n-\ell}Y^\ell} = (-1)^\ell\binom{n}{\ell} i^{\ell+1} 
\big[
L^*_S(f, \ell+1) 
+
L^*_T(f, \ell+1)
\big]
=(-1)^\ell\binom{n}{\ell}\,i^{\ell+1}L^*(f,\ell+1)~.
\end{equation}
This matches Definition~\ref{def:rf} term by term, and proves the claim.
\end{proof}

Having now defined the period polynomial associated to a weak cusp form $f \in S_k^!$,
we now seek to find its {\em (quasi-)periods}. First, a number of helpful definitions.

All objects below take values in $V_n:=\CC[X,Y]_n$, $n=k-2$, with the slash action
$(P|_\gamma)(X,Y):=P(aX+bY,cX+dY)$ of \S\ref{sec:intro}. A map $\gamma\mapsto A_\gamma\in V_n$ is a
\emph{$1$-cocycle} for $\SLZ$ if
\begin{equation}\label{eq:cocycle-condition}
A_{\gamma\delta}=A_\gamma|_\delta+A_\delta\qquad\text{for all }\gamma,\delta\in\SLZ ,
\end{equation}
and is \emph{parabolic} if in addition $A_T=0$. For fixed $P\in V_n$ the map
$\gamma\mapsto P|_\gamma-P$ is a \emph{coboundary}, and satisfies~\eqref{eq:cocycle-condition}
identically, since $(P|_\gamma-P)|_\delta+(P|_\delta-P)=P|_{\gamma\delta}-P$. Adding or subtracting
a coboundary from a cocycle therefore returns a cocycle, which is what lets $C_\gamma$ below be
replaced by a representative $\hat C_\gamma$ normalised so that $\hat C_T=0$.

For starters, let $f \in S_k^!$, $\tau_0 \in \HH$ and $\gamma \in \SLZ$, and define
\begin{equation}
C_\gamma(f; \tau_0):= (2\pi i)^{n+1}\int_{\gamma^{-1}\tau_0}^{\tau_0} \omega_f ~~,~~{\rm where}~~\omega_f := f(\tau) (X - \tau Y)^n d\tau~~.\label{cocycleR1}
\end{equation} 
Finally, it is helpful to define two slight abstractions related to $\mathbf K_T(\tau;X,Y)$ and to 
$L_T^*(f, s; \gamma^T_{\tau_0})$:
\begin{align}
\chi(\tau;X,Y)&:=\sum_{\ell=0}^{n}\binom{n}{\ell}(-1)^\ell\,\zeta(-\ell,\tau+1)\,X^{n-\ell}Y^\ell , \label{cocycleR2}\\
E(f; \tau_0)&:=(2\pi i)^{n+1}\!\!\int_{\gamma^T_{\tau_0}}\!\! f(\tau)\,\chi(\tau; X, Y) d\tau ~.\label{cocycleR3}
\end{align}
At non-positive integer first argument the Hurwitz zeta is elementary,
$\zeta(-\ell,x)=-B_{\ell+1}(x)/(\ell+1)$ with $B_m(x)$ the $m$-th Bernoulli
polynomial~\cite[\S25.11]{DLMF}, so $\chi(\tau;X,Y)$ is a polynomial in $\tau$ of degree $n+1$.
Its role is that of a discrete antiderivative of the Eichler kernel: the Hurwitz shift
$\zeta(-\ell,\tau)=\zeta(-\ell,\tau+1)+\tau^{\ell}$ gives, term by term,
\begin{equation}\label{eq:chi-shift}
\chi(\tau-1;X,Y)-\chi(\tau;X,Y)=(X-\tau Y)^{n} .
\end{equation}
Both facts are used in the proof of Lemma~\ref{lem:cocycle}.

For each $\tau$, $\chi(\tau;\cdot,\cdot)\in V_{k-2}=\CC[X,Y]_{k-2}$; the $\SLZ$-slash
$(P|_\gamma)(X,Y):=P(aX+bY,cX+dY)$ acts on the polynomial variables alone, holding $\tau$ fixed:
\begin{equation}
\chi|_\gamma(\tau;X,Y):=\chi\big(\tau;\,aX+bY,\,cX+dY\big),\qquad
\gamma=\left(\begin{smallmatrix}a&b\\c&d\end{smallmatrix}\right)\in\SLZ .
\end{equation}
Thus $|_T$ sends $(X,Y)\mapsto(X+Y,Y)$ with $\tau$ untouched. Being linear and $\tau$-independent,
$|_\gamma$ commutes with $\int\!d\tau$, so slashing the polynomial $E$ agrees with slashing $\chi$
under the integral; the two occurrences of $|_\gamma$ in Lemma~\ref{lem:cocycle} denote the same
operation.

\begin{lemma}\label{lem:cocycle}
Let $f\in S_k^!$, $\tau_0\in\HH$, and let $\omega_f$, $C_\gamma(f; \tau_0)$, $\chi(\tau; X, Y)$, and
$E(f;\tau_0)$ be as defined in Eqs.~\eqref{cocycleR1}, \eqref{cocycleR2} and \eqref{cocycleR3}. 
Then $\hat C_\gamma:=C_\gamma-(E|_\gamma-E)$ is a $1$-cocycle for $\SLZ$ with 
$\hat C_S=r_f$ and $\hat C_T=0$. 
\end{lemma}

\begin{proof}
First, we prove $\hat C_\gamma$ is a cocycle. Path-additivity and covariance of $\omega_f$ give the Eichler--Shimura cocycle relation
$C_{\gamma\delta}=C_\gamma|_\delta+C_\delta$~\cite{BCD2018,KZ1984}; as $\gamma\mapsto E|_\gamma-E$ is a
coboundary, $\hat C$ is a cocycle.

Having established it is a cocyle, we now show that $\hat C_S=r_f$. By
Proposition~\ref{prop:periodint} the period polynomial splits along the two segments of
Definition~\ref{def:segments}, and we identify each contribution in turn. The $S$-segment carries
the kernel $\tau^{s-1}$, so assembling it over $\ell$ returns
$(2\pi i)^{n+1}\int_{\gamma^S_{\tau_0}}\omega_f=C_S$ exactly. For the $T$-segment, at $s=\ell+1$
Definition~\ref{def:kernel} reads
$\tk(\tau,\ell+1)=\zeta(-\ell,\tau+1)-(-1)^\ell\zeta(-(n-\ell),\tau+1)$; assembling $\mathbf K_T$ and
re-indexing $\ell\mapsto n-\ell$ in the second term (with $n$ even) gives $\mathbf K_T=\chi-\chi|_S$,
so the $T$-term of $r_f$ is $E-E|_S$. Hence
\begin{equation}\label{eq:rfsplit}
r_f=C_S+(E-E|_S)=C_S-(E|_S-E)=\hat C_S .
\end{equation}
Neither summand in~\eqref{eq:rfsplit} vanishes in general: the $S$-segment runs from $-1/\tau_0$ to
$\tau_0$ and degenerates only at the fixed point $\tau_0=i$, where $C_S=0$ and $E|_S-E=-r_f$.

Finally, we show that $\hat C_T=0$. Applying the shift identity~\eqref{eq:chi-shift} twice shows
that $A(\tau):=\chi(\tau;X+Y,Y)-\chi(\tau-1;X,Y)$ satisfies $A(\tau-1)=A(\tau)$. Crucially,
$\chi$ is a polynomial in $\tau$, as recorded below Eq.~\eqref{cocycleR3}, so $A$ is a
$1$-periodic polynomial and hence constant:
$\partial_\tau A(\tau) = 0 \Leftrightarrow A(\tau) = A_0$. Thus,
\begin{align}
\hat C_T
&=C_T-(E|_T-E)
\nonumber\\ &
=(2\pi i)^{n+1}\int_{\gamma^T_{\tau_0}} f(\tau) (X - \tau Y)^n d\tau
- (2\pi i)^{n+1}\int_{\gamma^T_{\tau_0}} f(\tau)\big[\chi|_T(\tau; X, Y)-\chi(\tau; X, Y)\big] d\tau \nonumber\\
&=(2\pi i)^{n+1}\int_{\gamma^T_{\tau_0}} f(\tau)\,\bigg(\big[\chi(\tau-1; X, Y)-\chi(\tau; X, Y)\big] - \big[\chi(\tau; X+Y, Y)-\chi(\tau; X, Y)\big] \bigg) d\tau \nonumber\\
&=(2\pi i)^{n+1}\int_{\gamma^T_{\tau_0}} f(\tau)\,\big[\chi(\tau-1; X, Y)-\chi(\tau; X+Y, Y)\big] d\tau 
=-(2\pi i)^{n+1} \int_{\tau_0-1}^{\tau_0} f(\tau)A(\tau)\,d\tau \nonumber\\
&=-(2\pi i)^{n+1}A_0 \int_{\tau_0-1}^{\tau_0} f(\tau)\,d\tau
=-(2\pi i)^{n+1}A_0\,c_f(0)=0,
\end{align}
as all elements $f \in S_k^!$ must have $c_f(0) = 0$.
\end{proof}

\begin{definition}\label{def:w}
$W:=\{w\in\CC[X,Y]_{k-2}: w|_{(1+S)}=w|_{(1+U+U^2)}=0\} \subset V_n$
\end{definition}

\begin{lemma}\label{lem:rfW}
For $f\in S_k^!$, and $W \subset V_n$ from Definition~\ref{def:w}, the period polynomial $r_f(X, Y) \in W$.
\end{lemma}
\begin{proof}
Let $\hat C_\gamma$ be the parabolic cocycle of Lemma~\ref{lem:cocycle}. Thus, $r_f=\hat C_S$ 
and $\hat C_T=0$. Now, note that $\hat C_\gamma(f; \tau_0) = 0$ when $\gamma = \pm I$. Since
$n=k-2$ is even, $\pm I$ act trivially on $V_n$, so the coboundary term $E|_{\pm I}-E$ vanishes and
$\hat C_{\pm I}=C_{\pm I}$; and $C_{\pm I}$ is a closed-contour integral of a function which has no
poles in $\HH$. Finally, direct computation, starting from the identities $-I = U^3 = S^2$
and using the cocycle law $\hat C_{\gamma\delta}=\hat C_\gamma|_\delta+\hat C_\delta$ and clever 
use of $\hat C_T = 0$, shows $r_f|_{(1+S)}=r_f|_{(1+U+U^2)}=0$:
\begin{align}
&\hat C_U = \hat C_{TS} = \hat C_T|_S + \hat C_S = \hat C_S = r_f~,  \\
&0 = \hat C_{S^2} = \hat C_S|_S + \hat C_S = \hat C_S|_{(1+S)} = r_f|_{(1+S)}~,\\
&0 = \hat C_{U^3} = \hat C_U|_{UU}+\hat C_{UU}=\hat C_U|_{U^2}+\big(\hat C_U|_U+\hat C_U\big)=\hat C_U|_{(1+U+U^2)}=r_f|_{(1+U+U^2)}~.
\end{align}
Thus, $r_f \in W \subset V_n$, and finishes the proof. %
\end{proof}

To define periods and quasi-periods, we recall the standard fact
that $\dim(W)=2\dim(S_k)+1$ (see, e.g.~\cite{KZ1984}), and define 
the inner-product:
\begin{align}\label{inner}
\langle P,Q\rangle:=\sum_a(-1)^a\binom{n}{a}^{-1}p_a\,q_{n-a}~.
\end{align}
Now for the basis elements for $k$ where $\dim(S_k) = 1$:
\begin{definition}\label{def:Wpm}
Let $\dim S_k=1$, $n:=k-2$, $p_0(X):=X^n-1$, $B_\ell := \tfrac{\ell!}{2\pi i}\oint \tfrac{t^{-\ell}}{e^{t}-1}\,dt$, and
$B^0_m(X):=\sum_{i\ne1}\binom{m}{i}B_i X^{m-i}$. Motivated by~\cite[Thm.~1$'$]{KZ1984}, we define:
\begin{align}
W^{(k)}_-(X)&:={\rm pp}\left(\tfrac{1}{n-1}\big(B^0_{n-1}(X)+X^nB^0_{n-1}(\tfrac1X)\big)
-\tfrac13\big(B^0_3(X)+X^nB^0_3(\tfrac1X)\big)\right),\\
W^{(k)}_+(X)&:={\rm pp}\left(\tfrac12\big(B^0_2(X)-X^nB^0_2(\tfrac1X)\big)
+\tfrac1n\big(B^0_n(X)-X^nB^0_n(\tfrac1X)\big)-c\,p_0(X)\right),\\
v_0^{(k)}(X)&:= p_0(X)~,
\end{align}
where ${\rm pp}$ acts on any polynomial with rational coefficients by multiplying by all
coefficient-denominators and dividing by the gcd, and $\langle P, Q\rangle$ is as in
Eq.~\eqref{inner}. The bracketed polynomial defining $W^{(k)}_+$ has coefficients $a_i=-a_{n-i}$,
so the single constant $c:=a_n$ eliminates its $X^n$ and constant terms at once; this is the
$p_0$-component, and removing it is what makes $\langle W^{(k)}_+,p_0\rangle=0$ below.
\end{definition}

\begin{lemma}\label{lem:Wbasis}
For $k$ with $\dim S_k=1$, the space $W$ of Lemma~\ref{lem:rfW} has $\dim W=3$, and $\{
W^{(k)}_+$, $W^{(k)}_-$, $v_0^{(k)}\}$ is an orthogonal basis of $W$ under $\langle\cdot,\cdot\rangle$ from Eq.~\eqref{inner}.
\end{lemma}
\begin{proof}
First, let $W_\pm^{(k)}(X, Y) := Y^{k-2} W_\pm^{(k)}(X/Y)$, and let $v_0^{(k)}(X, Y) := Y^{k-2} v_0^{(k)}(X/Y)$.

Membership in $W$ is the substantive point. Theorem~1$'$ of~\cite{KZ1984} gives the odd and even
period polynomials of the cusp form $E_{k-12}\Delta$ (where $E_0:=1$) as modified-Bernoulli
combinations and places them in $W$; the $W^{(k)}_\pm$ of Definition~\ref{def:Wpm} are
specializations of those formulae, $W^{(k)}_-$ odd and $W^{(k)}_+$ even under $X\mapsto-X$, so both
lie in $W$. So does $v_0^{(k)}=p_0=X^{n}-Y^{n}$, directly: $p_0|_S=-p_0$ since $n$ is even, and
$p_0|_{(1+U+U^2)}=0$.

By Eichler--Shimura $\dim W=2\dim S_k+1=3$, so it is enough that the three be independent. Each
satisfies $\langle P,P\rangle\neq0$~\cite{VerifyScripts}, and mutually orthogonal vectors with
$\langle P,P\rangle\neq0$ are independent: pairing $\sum_ic_iv_i=0$ against $v_j$ leaves
$c_j\langle v_j,v_j\rangle=0$.

For orthogonality, note first that since $n$ is even, $X\mapsto-X$ multiplies the coefficient of
$X^{n-a}Y^a$ by $(-1)^a$, so an odd polynomial has $p_a=0$ for even $a$ and an even one has $q_b=0$
for odd $b$. As Eq.~\eqref{inner} pairs $p_a$ against $q_{n-a}$, and $a$ odd forces $n-a$ odd,
opposite parities pair to zero: the odd $W^{(k)}_-$ is orthogonal to the even $W^{(k)}_+$ and
$v_0^{(k)}$. For the remaining pair, $p_0$ is supported on its two extreme coefficients, so
$\langle P,p_0\rangle$ is the coefficient of $Y^n$ in $P$ less that of $X^n$; $c$ annihilates both
of these in $W^{(k)}_+$, whence $\langle W^{(k)}_+,v_0^{(k)}\rangle=0$. Hence
$\{W^{(k)}_+,W^{(k)}_-,v_0^{(k)}\}$ is an orthogonal basis, with
$\langle v_0^{(k)},v_0^{(k)}\rangle=-2$ at every such $k$.
\end{proof}

\section{Meromorphic modular forms}\label{sec:mero}

For $f\in F_k$ with poles in $\HH$, the two segment integrals of
Definition~\ref{def:Lint} depend on their contours; we show their sum depends only on the
homotopy class of the pair.

\begin{proposition}\label{prop:indepHomotopy}
For $f\in F_k$ and $\tau_0\in\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$, the L-integral
$L^*(f,s;\gamma^S_{\tau_0},\gamma^T_{\tau_0})$ depends only on the homotopy class of its
two segments in $\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$, and not on the basepoint
$\tau_0$. At fixed $\tau_0$ each of $L^*_S,L^*_T$ is separately path-independent.
\end{proposition}

\begin{proof}
At fixed $\tau_0$ the integrands $f(\tau)\tau^{s-1}$ and $f(\tau)\tk(\tau,s)$ are
holomorphic on $\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$, so Cauchy's theorem makes each
of $L^*_S,L^*_T$ constant on the homotopy class of its own segment. Under a motion of the
basepoint, $\partial_{\tau_0}L^*$ is evaluated at the moving endpoints alone; the
computation of Proposition~\ref{prop:indep}, which uses only $f(\tau_0-1)=f(\tau_0)$ and
$f(-1/\tau_0)=\tau_0^{k}f(\tau_0)$, cancels the boundary terms of $L^*_S$ against those
of $L^*_T$, the interior poles lying off the contours. Hence $\partial_{\tau_0}L^*=0$ along
any motion in $\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$, while each summand may 
separately vary.
\end{proof}

\begin{definition}\label{def:class}
For $f\in F_k$ and $\tau_0\in\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$, let
$[\gamma^S_{\tau_0}]$, $[\gamma^T_{\tau_0}]$ denote the homotopy classes of the two
segments in $\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$.
\end{definition}

Thus, with $\gamma^S_{\tau_0} \in [\gamma^S_{\tau_0}]$ and 
$\gamma^T_{\tau_0} \in [\gamma^T_{\tau_0}]$, by Proposition~\ref{prop:indepHomotopy} 
the value
\begin{equation}\label{eq:Lclass}
L^*\big(f,s;[\gamma^S_{\tau_0}],[\gamma^T_{\tau_0}]\big) 
:= L^*\big(f,s;\gamma^S_{\tau_0} , \gamma^T_{\tau_0}\big)
= L^*_S(f,s;\gamma^S_{\tau_0}) + L^*_T(f,s;\gamma^T_{\tau_0})
\end{equation}
is independent of the chosen representative paths and independent of $\tau_0$ (subject to 
natural constraints of $\tau_0$ monodromies around singular points, which move between 
homotopy classes).

We would like to speak of ``the'' L-integral. Homotopy classes of course split this into 
L-integrals for contours in different homotopy classes. And, so, in what follows, we define 
the reference homotopy classes: 
\begin{definition}[Reference homotopy class for $f \in F_k$]\label{def:null_homotopy}
Let $f\in F_k$, let $\tau_0$ be the base-point of $L^*(f,s)$, and let $f^{-1}(\infty)$ be the
poles of $f$. If $f$ has no poles on the line ${\rm Im}(\tau) = 1$, choose $\delta = 0$. 
Else, choose $\delta>0$ such that no pole lies in the strip $1<{\rm Im}(\tau)\leq 1+\delta$. 
Set $\hat\tau_0^f:=i(1+\delta)$. The \emph{reference homotopy class}
$[\hat\gamma_f]=[\hat\gamma_f^T]+[\hat\gamma_f^S]$ is:
\begin{itemize}
\item $[\hat\gamma_f^T]$: contains the horizontal segment from $T^{-1}\hat\tau_0^f$ to $\hat\tau_0^f$;
\item $[\hat\gamma_f^S]$: contains the near-vertical path from $S^{-1}\hat\tau_0^f$ to $\hat\tau_0^f$
along the imaginary axis, taking a pole at $i$ by principal value.
\end{itemize}
For $\delta=0$, the reference paths match the paths $\gamma^S_i\!:\! i \!\to\! i$ and $\gamma^T_i\!:\! (i\!-\!1)\!\to\! i$ for $L^*(f,s)$ when $f \in M_k^!$.
\end{definition}

%\smallskip\noindent For $f\in F_k$ with real $q$-coefficients, the reference class yields
%$\omega^+(f)\in\mathbb{R}$ and $\omega^-(f)\in i\mathbb{R}$: the axis-symmetric contour respects the
%reflection $\tau\mapsto-\bar\tau$, and the one on-axis pole (at $i$) reaches $\omega^\pm$ only through
%$(X-iY)^{k-2}$ --- which the period pairing annihilates when $k\equiv2\bmod4$ (there $W_\pm(i,1)=0$),
%or through $\widehat P_i$ with the real weights $(-2\pi i)^m a_{-m}$ when $4\mid k$.

With this definition, we may thus unambiguously write
\begin{align}
L^*(f,s) := L^*(f, s; [\hat\gamma_f^S], [\hat\gamma_f^T])~.
\end{align} 
Now, having fixed the reference value $L^*(f,s)$ in the reference
homotopy class of Definition~\ref{def:null_homotopy}, consider
two paths $\gamma^S_{\tau_0}$ and $\gamma^T_{\tau_0}$ which may 
be in any homotopy class. Further, let $X_S(p),X_T(p)\in\ZZ$ count 
the windings $\gamma^S_{\tau_0}$, $\gamma^T_{\tau_0}$
has about each pole $\tau_p\in f^{-1}(\infty)$ relative to the reference-class.

\begin{lemma}\label{lem:wall}
Let $f\in F_k$, $r_S(f, \tau_p) := \Res_{\tau=\tau_p} (f(\tau) \tau^{s-1})$ 
and $r_T(f, \tau_p) := \Res_{\tau=\tau_p} (f(\tau) \tk(\tau, s))$. Then, 
\begin{equation}\label{eq:wall}
L^*\big(f,s;[\gamma^S_{\tau_0}],[\gamma^T_{\tau_0}]\big)
= L^*(f,s)
+ 2\pi i\,e^{-\pi i s/2}\sum_{\tau_p\in f^{-1}(\infty)}\Big[
X_S(\tau_p) r_S(f, \tau_p)
+ X_T(\tau_p) r_T(f, \tau_p) \Big].
\end{equation}
\end{lemma}

\begin{proof}
Two pairs whose winding data differ by one unit at a single pole $p$, on one segment,
bound a loop encircling $p$ once; the residue theorem adds $2\pi i$ times the residue of
that segment's integrand, $f(\tau) \tau^{s-1}$ for $\gamma^S$ or $f(\tau)\tk(\tau, s)$ for $\gamma^T$,
and the prefactor $e^{-\pi i s/2}$ of Definition~\ref{def:Lint} multiplies through.
Summing over crossings gives \eqref{eq:wall}.
\end{proof}

This content provides the basics to discuss our finite L-functions for $f \in F_k$.
However, we wish to present these L-functions in terms of their q-series coefficients,
just as we have done for L-functions for $f \in M_k^!$ in the previous section.

The obstruction to doing so directly is not a technical one. For $f\in M_k^!$ the series
$\sum_n c_f(n)q^n$ converges on all of $\HH$, as $f$ is holomorphic there, and the term-by-term
evaluation of \S\ref{sec:Lfun} is legitimate throughout. A pole at finite height ends this. The
expansion about the cusp then represents $f$ only above the highest pole, on
${\rm Im}\,\tau>y_f:=\sup\{{\rm Im}\,\tau_p\}$, with coefficients growing as
$|c_f(n)|\gtrsim n^{m-1}e^{2\pi n\,{\rm Im}\,\tau_p}$ for a pole of order $m$~\cite{1806}. The
reference contour of Definition~\ref{def:null_homotopy} runs along ${\rm Im}\,\tau=1+\delta$ (with $\delta = 0$ if $f$ has no poles with ${\rm Im}(\tau_p)=1$), so as soon as
$f$ has a pole at or above that height the $q$-series diverges at every point of the segment. There
is then no series to integrate term by term, rather than a slowly convergent one to be handled with
care. To do this, we must introduce polylogarithmic projections of Ref.~\cite{1806}.

\subsection{Explicit formulae for $L^*(f,s)$ for general $f \in F_k$}

The polylog projection isolates the q-series contribution of a given pole of an $f \in F_k$
into a known special function, an integral of a polylogarithm of negative index, and leaves 
behind an object $(I-\widehat P)f$ with that one pole removed. Since the $q$-series of $f$ diverges
on the reference contour as soon as a pole sits at or above ${\rm Im}\,\tau=1$, it is essential for
our purposes to project out every such pole: only then does what remains admit a $q$-series there.

Within this general context, we lay-out the starting definitions and facts:

\begin{definition}\label{def:Finfty}
$F(\infty)$ is the space of meromorphic functions $g:\HH\to\CC$ with $g(\tau+1)=g(\tau)$, finitely many
poles in each compact subset of the strip $|{\rm Re}\,\tau|\le\tfrac12$, and bounded pole order at $q = 0$. 
\end{definition}

\begin{definition}\label{def:proj}
Let $f\in F_k$ have a pole of order $m_p$ at $\tau_p\in\HH$. On a punctured $\varepsilon$-ball about
$\tau_p$ small enough to exclude the other poles of $f$, we have
\begin{equation}\label{eq:laurent}
f(\tau)=\sum_{m=1}^{m_p}\frac{r^*_{f,\tau_p}(m)}{(\tau-\tau_p)^{m}}+g(\tau),
\qquad
r^*_{f,\tau_p}(m)=\frac{1}{2\pi i}\oint_{|\tau-\tau_p|=\varepsilon}
f(\tau)\,(\tau-\tau_p)^{m-1}\,d\tau ,
\end{equation}
with $g$ holomorphic at $\tau_p$. We call $r^*_{f,\tau_p}(1),\dots,r^*_{f,\tau_p}(m_p)$ the
\emph{pole weights} of $f$ at $\tau_p$; the first is the residue, and the last is nonzero, which is
what fixes the order. We may then define the \emph{polylog projection} of $f$ at $\tau_p$ to be
\begin{equation}\label{eq:polyproj}
\widehat P_{\tau_p}f(\tau):=\sum_{m=1}^{m_p} r^*_{f,\tau_p}(m)\,
\frac{(-2\pi i)^m}{(m-1)!}\,\Li_{1-m}\big(e^{2\pi i(\tau-\tau_p)}\big)~,~{\rm where} ~\Li_s(x):=\sum_{n\ge1}\frac{x^{n}}{n^{s}}~.
\end{equation}
\end{definition}

\begin{definition}\label{def:pole_split}
For every $f \in F_k$, there are a finite set of $T$-inequivalent finite order poles
$\{ \tau_p \}$ of order $\{ m_p \}$. We may use the polylogarithm projector of
Definition~\ref{def:proj} to define two functions $\widehat{P}(f)$ and
$(1-\widehat{P})(f) \in F(\infty)$, via the composite projection operator:
\begin{equation}\label{eq:polesplit}
\widehat P:=\sum_{{\rm Im}\,\tau_p\ge1}\widehat P_{\tau_p}~.
\end{equation}
Their respective singularity structures are integral to our explicit L-function expressions,
below.
\end{definition}

\begin{lemma}\label{lem:proj}
%The weights $r^*_{f,\tau_p}(m)$, $1\le m\le m_p$, exist and are unique, and
%$(I-\widehat P_{\tau_p})f$ is holomorphic at $\tau_p$.

The normalized block $\frac{(-2\pi i)^m}{(m-1)!}\Li_{1-m}\big(e^{2\pi i(\tau-\tau_p)}\big)$ has
principal part exactly $(\tau-\tau_p)^{-m}$, so $\widehat P_{\tau_p}f$ matches the principal part of
$f$ and $(I-\widehat P_{\tau_p})f$ is holomorphic at $\tau_p$.
\end{lemma}
\begin{proof}
%$\Li_{1-m}\big(e^{2\pi i(\tau-\tau_p)}\big)$ has a pole of order exactly $m$ at $\tau_p$, so
%matching the principal part of $f$ is a triangular---hence uniquely solvable---linear system
%relating $r^*_{f,\tau_p}(1),\dots,r^*_{f,\tau_p}(m)$ to $a^{(p)}_{-1},\dots,a^{(p)}_{-m}$; equal
%principal parts leave $(I-\widehat P_{\tau_p})f$ regular at $\tau_p$.
%By the small-$t$ expansion $\Li_{1-m}(e^{-t})=(m-1)!\,t^{-m}+O(1)$ with $t=-2\pi i(\tau-\tau_p)$, each

%$\tfrac{(m-1)!}{(-2\pi i)^m}(\tau-\tau_p)^{-m}$ with no lower-order poles. Matching the principal part
%of $f$ is therefore diagonal, giving $r^*_{f,\tau_p}(m)=(-2\pi i)^m a^{(p)}_{-m}/(m-1)!$ and leaving

By the small-$t$ expansion $\Li_{1-m}(e^{-t})=(m-1)!\,t^{-m}+O(t^0)$ with $t=-2\pi i(\tau-\tau_p)$, the
block $\frac{(-2\pi i)^m}{(m-1)!}\Li_{1-m}\big(e^{2\pi i(\tau-\tau_p)}\big)$ has a \emph{pure} pole

\begin{definition}\label{def:cRf}
Let $f \in F_k$ and $\widehat P$ as in Definition~\ref{def:pole_split}. We denote the Fourier
coefficients of $(I-\widehat P)f$ as follows:
\begin{equation}
(I-\widehat P)f(\tau)=\sum_n c_{Rf}(n)\,q^n,\qquad q=e^{2\pi i\tau}.
\end{equation}
\end{definition}

\begin{lemma}\label{lem:projspace}
$F_k\subset F(\infty)$ and $(I-\widehat P)f\in F(\infty)$. Every pole of $(I-\widehat P)f$ has
${\rm Im}\,\tau<1$, with a maximal height $y_b<1$ (set $y_b=0$ if $(I-\widehat P)f$ is
holomorphic).
\end{lemma}
\begin{proof}
A weakly holomorphic form is $T$-periodic, meromorphic, and $O(e^{C{\rm Im}\tau})$ at the cusp,
so $F_k\subset F(\infty)$; the projected polylogs $\Li_{1-m}(e^{2\pi i(\tau-\tau_p)})$ share
these properties, so $(I-\widehat P)f\in F(\infty)$ (and leaves $F_k$, being non-modular). By
Lemma~\ref{lem:proj}, $\widehat P$ cancels the principal part of every pole with
${\rm Im}\tau\ge1$, so the poles of $(I-\widehat P)f$ lie strictly below ${\rm Im}\tau=1$;
finite in each compact set and accumulating only at ${\rm Im}\tau=0$, they have a maximal
height $y_b<1$.
\end{proof}

The Fourier coefficients of a $T$-periodic meromorphic function are controlled by its highest pole:
$|c_{Rf}(n)|\le e^{2\pi n(y_b+\varepsilon)}$ for every $\varepsilon>0$ and all large
$n$~\cite{1806}. Since $y_b<1$, the series $\sum_n c_{Rf}(n)q^n$ converges geometrically down to
${\rm Im}\,\tau=1$, at base $e^{-2\pi(1-y_b-\varepsilon)}$, and so on $\hat\gamma^T_f$ itself. This
is the point of the projection: convergence merely for large ${\rm Im}\,\tau$ would be automatic and
of no use, as the contour sits at height one.

For $n\ne0$, define
\begin{align}\label{eq:gc}
G_{s,k}(n):=\frac{\Gamma(s,2\pi n)}{(2\pi n)^s}
+i^k\,\frac{\Gamma(k-s,2\pi n)}{(2\pi n)^{k-s}}.
\end{align}
Each pole with ${\rm Im}\,\tau_p\ge1$ contributes one projected polylog to $L^*(f,s)$, whose
L-integral we now record.

\begin{definition}\label{def:IN}
For a pole of order $N+1$ projected at $z$ with ${\rm Im}\,z>1$, let $I_N(s,z)$ be the L-integral of the
projected polylog $\lambda_{N,z}$ in the minimal class $[\hat\gamma_f^S],[\hat\gamma_f^T]$
of Definition~\ref{def:null_homotopy}:
\begin{equation}\label{eq:IN}
I_N(s,z):=L^*\big(\lambda_{N,z},\,s;\,[\hat\gamma_f^S],[\hat\gamma_f^T]\big),
\end{equation}
where $\lambda_{N,z}$ denotes the function $\Li_{-N}(e^{2\pi i(\tau-z)})$, fed into the linear
$L^*$-functional.
\end{definition}

\begin{lemma}\label{lem:polylogcf}
Let $f\in F_k$ have a pole of order $p$ at $z$, ${\rm Im}\,z>1$, with projection weights
$r^*_{f,z}(m)$ (Definition~\ref{def:proj}). Polylog inversion
$\Li_{-N}(w)=(-1)^{N+1}\Li_{-N}(1/w)-\delta_{N,0}$ ($N\in\NN_0$) resums \eqref{eq:IN} into
(each exponentially convergent, base $e^{2\pi(1-{\rm Im}(z))}$)
\begin{align}
L^*(\widehat P_z f,\,s;\,[\hat\gamma_f^S],[\hat\gamma_f^T])&=\sum_{m=1}^{p} r^*_{f,z}(m)\,\frac{(-2\pi i)^m}{(m-1)!}\,I_{m-1}(s,z),\label{eq:lem4.12.whole.z}\\
I_N(s,z)&=\delta_{N,0}\Big(\tfrac1s+\tfrac{i^k}{k-s}\Big)
+(-1)^{N+1}\sum_{n=1}^{\infty} n^N e^{2\pi inz}\,G_{s,k}(-n).
\end{align}
\end{lemma}

\begin{proof}
By Proposition~\ref{prop:indepHomotopy} we may evaluate $I_N(s,z)$ in the $\delta\to0$
representative of the minimal class; there the projected polylog, which is regular at $i$ since
${\rm Im}\,z>1$, has vanishing $\gamma^S$-integral and $\hat\gamma_f^T\to[i-1,i]$. On
$[i-1,i]$ one has $|e^{2\pi i(\tau-z)}|>1$, so the defining series of $\Li_{-N}$ diverges;
inversion rewrites it as $(-1)^{N+1}\sum_{n\ge1}n^Ne^{2\pi in(z-\tau)}$, convergent since
$|e^{2\pi i(z-\tau)}|<1$, and term-by-term integration against $\tk$ gives
$\sum_n n^N e^{2\pi inz}G_{s,k}(-n)$, the mode-$(-n)$ block of Theorem~\ref{thm:weakL}; for
$N=0$ the leftover $-1$ in $\Li_0(w)=-1-\Li_0(1/w)$ contributes $\tfrac1s+\tfrac{i^k}{k-s}$.
The rate is $e^{-2\pi({\rm Im}\,z-1)}$, from $\Gamma(s,-2\pi n)\sim(2\pi n)^{s-1}e^{2\pi n}$
against $|e^{2\pi inz}|=e^{-2\pi n\,{\rm Im}\,z}$.
\end{proof}

Write $(a)_{(\ell)}:=a(a-1)\cdots(a-\ell+1)$ for the falling factorial, with $(a)_{(0)}:=1$. For any
integer $J\ge1$, the incomplete gamma satisfies
\begin{equation}\label{eq:gammarec}
\Gamma(s,X)=e^{-X}\sum_{\ell=0}^{J-1}(s-1)_{(\ell)}\,X^{s-1-\ell}+(s-1)_{(J)}\,\Gamma(s-J,X).
\end{equation}
It peels the leading $J$ terms of the large-$X$ expansion of $\Gamma(s,X)$ into elementary powers
$X^{s-1-\ell}$ and collects the rest into an incomplete gamma of lower order $s-J$ and hence
faster-decaying. Importantly, the peeled powers integrate and resum against our kernel in closed
form. Choosing $J$ large enough therefore quarantines the entire singular structure at
${\rm Im}(z)=1$ into that finite, explicit sum, while the remainder's mode-sum is left manifestly
convergent. We use~\eqref{eq:gammarec} first for the poles on the line ${\rm Im}(z)=1$ with $z\neq i$,
and then precisely at $z=i$.

\Pi_N(s,z)&:=\sum_{\ell=0}^{N}\frac{\Li_{\ell+1-N}\big(e^{2\pi i(z-i)}\big)}{(-2\pi)^{\ell+1}} \big\{ (s-1)_{(\ell)}+i^k(k-s-1)_{(\ell)} \big\},\label{eq:tailpiece1}\\
R_N(s,z)&:=\sum_{n=1}^{\infty}n^N e^{2\pi inz}\Big[(s{-}1)_{(N{+}1)}\tfrac{\Gamma(s{-}N{-}1,{-}2\pi n)}{(-2\pi n)^{s}}
+i^k(k{-}s{-}1)_{(N{+}1)}\tfrac{\Gamma(k{-}s{-}N{-}1,{-}2\pi n)}{(-2\pi n)^{k-s}}\Big], \label{eq:tailpiece2}
\end{align}
with $\Li_s$ as in Definition~\ref{def:proj}, and $\Gamma(a,x)$ the upper incomplete gamma taken on
its principal branch~\cite{DLMF}.

\begin{lemma}\label{lem:polylogedge}

%Eqs.~\eqref{eq:tailpiece1} and~\eqref{eq:tailpiece2}, for ${\rm Im}\,z>1$ and every $s\in\CC$,
%\begin{equation}\label{eq:twolayer}
%%I_N(s,z)=\delta_{N,0}\Big(\tfrac1s+\tfrac{i^k}{k-s}\Big)
%L^*\big(\Li_{-N}(e^{2\pi i(\cdot-z)}),\,s\big)=I_N(s,z)=\delta_{N,0}\Big(\tfrac1s+\tfrac{i^k}{k-s}\Big)

%\end{equation}
%and the right side extends $I_N(s,\cdot)$ continuously to the edge ${\rm Im}\,z=1$ for
%$z\notin i+\ZZ$.
Let $f\in F_k$ have a pole of order $p$ at $z$ with projection weights $r^*_{f,z}(m)$
(Definition~\ref{def:proj}). With $\Pi_N,R_N$ as in \eqref{eq:tailpiece1}--\eqref{eq:tailpiece2},
for ${\rm Im}\,z>1$ and every $s\in\CC$,
\begin{align}
L^*(\widehat P_z f,\,s;\,[\hat\gamma_f^S],[\hat\gamma_f^T])&=\sum_{m=1}^{p} r^*_{f,z}(m)\,\frac{(-2\pi i)^m}{(m-1)!}\,I_{m-1}(s,z),\label{eq:lem4.13.whole.z}\\
I_N(s,z)&=\delta_{N,0}\Big(\tfrac1s+\tfrac{i^k}{k-s}\Big)
+(-1)^{N+1}\big[\Pi_N(s, z)+R_N(s, z)\big],\label{eq:twolayer}
\end{align}
and the second line's right side extends $I_N(s,\cdot)$ continuously to ${\rm Im}\,z=1$ for
$z\notin i+\ZZ$.
\end{lemma}

The crux of this Lemma is to show the existence of the smooth continuation of $I_N(s, z)$ 
from Lemma~\ref{lem:polylogcf}, derived for ${\rm Im}(z) > 1$, to the ${\rm Im}(z) = 1$-line 
away from the elliptic point $z = i$. This elliptic point requires special treatment and care,
involving specific relations between Laurent coefficients at the special elliptic point $\tau = i$.
We shall deal with this special case separately.

\begin{proof}
%Write $w:=e^{2\pi i(z-i)}$ and $Q:=e^{2\pi iz}=e^{-2\pi}w$, so $|Q|\le e^{-2\pi}$ and $|w|\le1$, with
%$|w|=1$ exactly on the edge $\mathrm{Im}\,z=1$. By Lemma~\ref{lem:polylogcf},
We manipulate the main result of Lemma~\ref{lem:polylogcf},
\begin{align}
I_N(s,z)=\delta_{N,0}\Big(\tfrac1s+\tfrac{i^k}{k-s}\Big)+(-1)^{N+1}\sum_{n=1}^{\infty}n^N e^{2\pi inz} \left(
\frac{\Gamma(s,-2\pi n)}{(-2\pi n)^{s}}+i^k\frac{\Gamma(k-s,-2\pi n)}{(-2\pi n)^{k-s}} \right)~,
\end{align}
Our proof proceeds in two steps. First, we isolate the possible sources of divergence on the line ${\rm Im}(z) = 1$ into a finite sum of polylogarithms $\Li_{\ell+1-N}(e^{2 \pi i (z-i)})$ for $\ell = 0, \ldots, N$. To do this, we apply the recursion~\eqref{eq:gammarec} to each gamma in $G_{s,k}(-n)$ at $X=-2\pi n$, $J=N+1$. Dividing each incomplete-Gamma term by $(-2\pi n)^{s}$
and $(-2\pi n)^{k-s}$ yields, then, a common factor of $1/(-2\pi n)^{\ell+1}$ for each term in the finite sum.
Further, this finite-sum is proportional to $e^{2\pi n}$; we leave the trailing incomplete gammas as-is:
\begin{align}
G_{s,k}(-n) =
\left\{
e^{2\pi n}\sum_{\ell = 0}^{N} \frac{(s-1)_{(\ell)}}{(-2\pi n)^{\ell+1}}
{+} \frac{(s{-}1)_{(N{+}1)} \Gamma(s{-}N{-}1,{-}2\pi n)}{(-2\pi n)^s}
\right\} {+}i^k(s{\to}k{-}s)~.
\end{align}
Now, we consider the $n$-sum $G_{s,k}(-n)$, starting with the falling factorial terms:
\begin{align}
\sum_{n=1}^{\infty} n^N e^{2\pi i n z} e^{2 \pi n} \left(  \sum_{\ell = 0}^{N} \frac{(s-1)_{(\ell)}}{(-2\pi n)^{\ell+1}}{+}i^k(s{\to}k{-}s) \right) = \sum_{\ell = 0}^{N} \frac{\Li_{\ell+1-N}(e^{2 \pi i (z -i)})}{(-2\pi)^{\ell+1}} \big\{ (s-1)_{(\ell)} + i^k (k- s-1)_{(\ell)}\big\}~,
\end{align}
i.e., $\Pi_N(s,z)$ of \eqref{eq:tailpiece1}, while the reduced-gamma terms remaining are 
$R_N(s, z)$ of Eq.~\eqref{eq:tailpiece2}, which together establish the lemma's
identity~\eqref{eq:twolayer}.

The second main step of our proof is to show that $\Pi_N(s, z)$ is regular as 
${\rm Im}(z) \to 1$ for $z \neq i$, while $R_N(s, z)$ is regular for all $z$ with 
${\rm Im}(z) = 1$. We treat each piece in turn, beginning with $R_N$.


Each term of $R_N$ is internal to the $G_{s,k}(-n)$ in the mode $\sum e^{2\pi inz} G_{s,k}(-n)$ 
from Lemma~\ref{lem:polylogcf}. Clearly, $|e^{2\pi inz}|=e^{-2\pi n\,{\rm Im}\,z}$. Thus, the 
incomplete gamma's exponential growth cancels against the exponentially 
falling $e^{2 \pi i n z}$ factors. As $n \to \infty$, we have:
\begin{align}
\left|n^N e^{2\pi i n z} \frac{\Gamma(s{-}N{-}1,{-}2\pi n)}{({-}2\pi n)^s}\right| &
{\sim}\left| \frac{n^N e^{2\pi i n z} ({-}2\pi n)^{s{-}N{-}2}e^{2\pi n}}{({-}2\pi n)^s}\right|
{\sim}\left| \frac{e^{2\pi i n z}e^{2\pi n}}{ ({-}2\pi)^{2{+}N} n^2}\right|
{\sim} \frac{e^{2\pi n (1{-}{\rm Im}(z))}}{(2\pi)^{N{+}2} n^2}%~,
\end{align}
where the detailed asymptotics of $\Gamma(s, x)$ in the first scaling relation follow 
from~\cite[\S8.11]{DLMF}. (The $i^k (s \rightarrow k-s)$-partner follows identically.)
Hence the $n$-th term of $R_N$ is $O\!\big(n^{-2}e^{-2\pi n({\rm Im}\,z-1)}\big)$, uniformly for
${\rm Im}\,z\ge1$. There $e^{-2\pi({\rm Im}\,z-1)}\le1$, so
$\sum_{n}n^{-2}e^{-2\pi n({\rm Im}\,z-1)}\le\zeta(2)$: the series converges absolutely and uniformly,
up to and including the edge ${\rm Im}\,z=1$, so $R_N$ is continuous there.

For $\Pi_N$, recall that it is a sum over $N+1$ polylogarithms of integer index $\le1$,
$\Pi_N(s, z) = c_1(s) \Li_1(w) + c_0(s) \Li_0(w) + ... + c_{1{-}N}(s) \Li_{1{-}N}(w)$, where $w := e^{2\pi i (z-i)}$.
Each $\Li_a$ ($a\le1$) is single-valued and analytic on the cut plane
$\CC\setminus[1,\infty)$: for $a\le0$ it is rational with a pole of order $1-a$ at $w=1$, while
$\Li_1(w)=-\log(1-w)$ has a logarithmic branch point there. Every singularity sits at the endpoint
$w=1$, and the cut of $\Li_1$ runs along the ray $[1,\infty)$. The edge is the unit circle
$|w|=1$, which meets that radial cut only at $w=1$; approaching it from inside the disc
(${\rm Im}\,z>1$) gives the analytic boundary value of each $\Li_a$ at every $w\ne1$. Hence
$\Pi_N$ is analytic along the whole edge except at $z\in i+\ZZ$, where each term
diverges. With the continuity of $R_N$, \eqref{eq:twolayer} seamlessly extends
$I_N(s,\cdot)$ continuously to ${\rm Im}\,z=1$ for $z\notin i+\ZZ$.
\end{proof}

The above result is insufficient to treat with the Laurent series singularity patterns of a modular form 
$f \in F_k$ which has poles at $\tau = i$, where $\Pi_N$ is singular ($w=1$). However, 
as we will presently show, even though each individual polylogarithm of index at most one diverges
at the point $z = i$, the special elliptic nature of that point imposes constraints on the sum of naively 
divergent terms such that the divergences exactly cancel. Before doing this, we find it helpful to present 
an important definition, and an associated remark: 


\begin{definition}\label{def:FP}
For $N\ge0$, with $H_N:=\sum_{r=1}^N\tfrac1r$ and $c^+_\ell(s):=(s-1)_{(\ell)}+i^k(k-s-1)_{(\ell)}$, define:
\begin{align}\label{eq:Eplus}
\mathcal E^+_N(s)&:=\sum_{n\ge1}n^N e^{2\pi n}\Big[(s-1)_{(N+1)}\tfrac{\Gamma(s-N-1,2\pi n)}{(2\pi n)^s}
+i^k(k-s-1)_{(N+1)}\tfrac{\Gamma(k-s-N-1,2\pi n)}{(2\pi n)^{k-s}}\Big], \\
\label{eq:FP}
\mathrm{FP}_N(s,k)&:=\frac{c^+_N(s)\,H_N}{(2\pi)^{N+1}}
+\sum_{\ell=0}^{N-1}\frac{c^+_\ell(s)\,\zeta(\ell+1-N)}{(2\pi)^{\ell+1}}
+\frac{(-1)^N N!}{(2\pi)^{N+1}}\sum_{t=0}^{N-1}\frac{(-1)^t(s-1)_{(t)}}{t!\,(N-t)}
+\mathcal E^+_N(s).
\end{align}
\end{definition}
Note: $\mathcal E^+_N(s)$ and $R_N(s,i)$ of~\eqref{eq:tailpiece2} resemble one another because both
come from expanding $\Li_{-N}(e^{2\pi i(\tau-z)})$ and integrating term by term; they are not equal,
differing by the inversion $e^{2\pi i(\tau-z)}\mapsto e^{-2\pi i(\tau-z)}$. On the contour that
argument has modulus $e^{2\pi({\rm Im}\,z-1)}$: for ${\rm Im}\,z>1$ the un-inverted series diverges,
so Lemma~\ref{lem:polylogcf} inverts and $R_N$ inherits the arguments $-2\pi n$, whereas at $z=i$ the
modulus is $1$, inversion buys nothing, and $\mathcal E^+_N$ resums the un-inverted series
with $+2\pi n$.

\begin{remark}\label{rem:cutoff}
\emph{The remainders $R_N$ of Lemma~\ref{lem:polylogedge}~\eqref{eq:tailpiece2} and $\mathcal E^+_N$ of
Definition~\ref{def:FP}~\eqref{eq:Eplus} are what is left after unfolding $J$ terms of the
incomplete-gamma recursion~\eqref{eq:gammarec}; both use $J=N+1$. The unfolding
isolates the terms that could diverge as ${\rm Im}(z)\to1^+$ into an explicit finite set of special
functions, and $J=N+1$ is the least $J$ that isolates all of them, and leaves a remainder that
converges for every $z$ on ${\rm Im}(z)=1$. Any larger $J$ isolates the same terms, leaving $I_N$ and
$\mathrm{FP}_N$ unchanged and merely quickening the remainder's decay ($\sim n^{-(J-N+1)}$) at the
cost of carrying further positive-index polylogs $\Li_2,\dots,\Li_{J-N}$ into the finite sum
(contributing values up to $\zeta(J-N)$ at $z=i$).}
\end{remark}

\begin{lemma}\label{lem:Laurent}
Let $f\in F_k$ have a pole of order $P$ at $i$, with pole weights $r^*_{f,i}(m)$
(Definition~\ref{def:proj}), i.e.\ principal part $\sum_{m=1}^{P}r^*_{f,i}(m)(\tau-i)^{-m}$.
Modularity $f(-1/\tau)=\tau^k f(\tau)$ is equivalent to one relation for each principal-part
order $\mu$,
\begin{equation}\label{eq:Laurent}
%(-1)^r\sum_{m=r}^P a_{-m}\binom{m}{r}i^{m-r}
%=i^k\sum_{m=r}^P a_{-m}\binom{k}{m-r}(-i)^{m-r},\qquad r=1,\dots,P.
%\sum_{m=r}^{P} a_{-m}\binom{m}{r}\,i^{m}
%=i^{k}\sum_{m=r}^{P} a_{-m}\binom{k}{m-r}\,i^{-m},\qquad r=1,\dots,P.
\sum_{m=\mu}^{P} r^*_{f,i}(m)\binom{m}{\mu}\,i^{m}
=i^{k}\sum_{m=\mu}^{P} r^*_{f,i}(m)\binom{k}{m-\mu}\,i^{-m},\qquad \forall \mu=1,\dots,P.
\end{equation}
The leading relation ($\mu=P$) is $\big((-1)^P-i^k\big)r^*_{f,i}(P)=0$, so the pole order matches
$k\bmod4$ ($P$ odd $\Leftrightarrow k\equiv2$, $P$ even $\Leftrightarrow 4\mid k$).
\end{lemma}

Note: at $P=1$, the Laurent-series relations claimed in this Lemma work out to be this is 
$(1+i^k)r^*_{f,i}(1)=0$; for a double pole the $\mu=1$ relations are $r^*_{f,i}(1)=\tfrac{i(k-2)}{2}r^*_{f,i}(2)$.

\begin{proof}
Write $\tau=i+\varepsilon$. Then $-1/\tau-i=-\varepsilon/(1-i\varepsilon)$, so the principal
part of $f(-1/\tau)$ is $\sum_m r^*_{f,i}(m)(i-\varepsilon^{-1})^m$ and that of $\tau^k f(\tau)$ is
$i^k(1-i\varepsilon)^k\sum_m r^*_{f,i}(m)\varepsilon^{-m}$. Matching the coefficient of
$\varepsilon^{-\mu}$ through
$(i-\varepsilon^{-1})^m=\sum_j\binom mj i^{m-j}(-1)^j\varepsilon^{-j}$ and
$(1-i\varepsilon)^k=\sum_l\binom kl(-i)^l\varepsilon^l$ gives
$(-1)^\mu\sum_{m=\mu}^P r^*_{f,i}(m)\binom{m}{\mu}i^{m-\mu}=i^k\sum_{m=\mu}^P r^*_{f,i}(m)\binom{k}{m-\mu}(-i)^{m-\mu}$;
dividing through by $i^\mu$ (with $-1=i^2$ and $-i=i^{-1}$) yields~\eqref{eq:Laurent}.
%$(1-i\varepsilon)^k=\sum_l\binom kl(-i)^l\varepsilon^l$ gives \eqref{eq:Laurent}.
\end{proof}

\begin{lemma}\label{lem:wallvalue}
Let $f\in F_k$ have a pole of order $P$ at $i$ with pole weights $r^*_{f,i}(m)$
(Definition~\ref{def:proj}), and $\widehat P_i f$ its projection. Evaluated in the minimal class at
basepoint $i(1+\delta)$, the pole's contribution to $L^*(f,s)$ has the finite $\delta\to0$ limit
\begin{equation}\label{eq:wallvalue}
L^*(\widehat P_i f,\,s):=\lim_{\delta\to0}L^*\big(\widehat P_i f,\,s;\,i(1+\delta)\big)
=\sum_{m=1}^P \frac{(-2\pi i)^m r^*_{f,i}(m)}{(m-1)!}\,\mathrm{FP}_{m-1}(s,k),
\end{equation}
entire in $s$. The limit exists precisely because the pole weights obey the relations of
Lemma~\ref{lem:Laurent} at the $S$-fixed point $i$: they cancel the individually divergent
contour pieces.
\end{lemma}

\begin{proof}
We begin by noting that, at %Fix $s$; evaluate in the minimal class at basepoint $i(1+\delta)$, $\delta>0$, and set
%$u:=2\pi i(\tau-i)$.
%
%\emph{Block expansion.} At 
its branch point $w = e^{2 \pi i (\tau-i)} \sim 1$ the polylogarithm expands as
$\Li_{1-m}(e^{2 \pi i (\tau-i)})=(m-1)!\,(-2 \pi i (\tau-i))^{-m}+\sum_{p\ge0}\tfrac{\zeta(1-m-p)}{p!}(2 \pi i (\tau-i))^{p}$
(\cite[\S25.12]{DLMF}). Thus, %; multiplying by $\tfrac{(-2\pi i)^m}{(m-1)!}$ and using $-u=-2\pi i(\tau-i)$,
\begin{equation}\label{eq:blockexp}
\frac{(-2\pi i)^m}{(m-1)!}\,\Li_{1-m}\!\big(e^{2\pi i(\tau-i)}\big)
=(\tau-i)^{-m}+\frac{(-2\pi i)^m}{(m-1)!}\sum_{p=0}^{\infty}\frac{\zeta(1-m-p)}{p!}\,(2\pi i(\tau-i))^{p},
\end{equation}
is a pure order-$m$ pole, as in Lemma~\ref{lem:proj}, plus a tail holomorphic on $|\tau-i|<1$. Hence
$\widehat P_i f=\sum_{m=1}^P r^*_{f,i}(m)(\tau-i)^{-m}+(\text{regular at }i)$. 
As the contour ${\rm Im}(\hat\gamma_f^T)$ has height $1+\delta > 1$, the analogous step of 
$w$-inversion, $\Li_m(w)=-\delta_{m,0}-(-1)^m \Li_m(1/w)$ from the proof of Lemma~\ref{lem:polylogcf} 
does not need to be invoked. Thus there is no $\tfrac1s+\tfrac{i^k}{k-s}$ in the final result. 
%Because the pole lies
%below the $T$-segment (height $1+\delta$), $|e^{u}|<1$ there and the block is used \emph{without}
%inversion: no constant ($n{=}0$) mode appears, so no $\tfrac1s+\tfrac{i^k}{k-s}$ term arises and
%$L^*(\widehat P_i f,s)$ is entire in $s$. %(A pole \emph{above} the contour would invert
%$\Li_0(w)=-1-\Li_0(1/w)$, and the leftover $-1$ is the source of that pole in
%Lemma~\ref{lem:polylogcf}; here it is absent.)

We now extract their finite parts of this polylogarithm on the $\hat\gamma_f^T$ and $\hat\gamma_f^S$ contour-integrals.

%\emph{Finite part is $\mathrm{FP}$.} Write $N:=m-1$; the finite part collects a $T$-segment piece
%(the $\zeta$-sum, harmonic term, and $\mathcal E^+$) and an $S$-segment piece (the polynomial term).

On $\hat\gamma_f^T$ the block is the convergent series
$\Li_{1-m}(e^{u})=\sum_{n\ge1}n^{m-1}e^{2\pi n}e^{2\pi in\tau}$; integrating against $\tk$ from the
basepoint $\tau_0=i(1+\delta)$ gives, by Theorem~\ref{thm:weakL},
$\sum_{n\ge1}n^{m{-}1} e^{2\pi n}\big[\tfrac{\Gamma(s,X_n)}{(2\pi n)^{s}}+i^k\tfrac{\Gamma(k-s,X_n)}{(2\pi n)^{k-s}}\big]$
with $X_n:=-2\pi in\tau_0=2\pi n(1+\delta)$. Expanding by the recursion~\eqref{eq:gammarec} at $J=m$, and using
$e^{2\pi n}e^{-X_n}=e^{-2\pi n\delta}$ and $X_n^{s-1-\ell}=(2\pi n)^{s-1-\ell}(1+\delta)^{s-1-\ell}$, the
peeled sum over $0\le\ell\le m-1$ becomes (with $c^+_\ell(s)$ collecting the $s$- and $i^k(k{-}s)$-parts)
\begin{equation}\label{eq:Tsum}
\sum_{\ell=0}^{m-1}\frac{c^+_\ell(s)\,(1+\delta)^{s-1-\ell}}{(2\pi)^{\ell+1}}\,\Li_{\ell+2-m}\!\big(e^{-2\pi\delta}\big)
\;+\;\mathcal E^+_{m-1}(s),
\end{equation}
$\mathcal E^+_{m-1}$ being the reduced-$\Gamma$ remainder of~\eqref{eq:Eplus}. % (any deeper cutoff $J>m$
%moves the $j$-terms $m,\dots,J-1$ out of $\mathcal E^+_{m-1}$ into the $\zeta$-sum and leaves
%$\mathrm{FP}_{m-1}$ unchanged). 
%
With
$\Li_{\ell+2-m}(e^{-2\pi\delta})=\Gamma(m-\ell-1)(2\pi\delta)^{\ell-m+1}+\sum_{p\ge0}\tfrac{\zeta(\ell+2-m-p)}{p!}(-2\pi\delta)^p$,
two finite ($\delta^0$) parts survive:
\begin{enumerate}
\item The regular part at $\delta^0$ gives, for $\ell\ne m-1$, the $\zeta$-value
$\tfrac{c^+_\ell(s)}{(2\pi)^{\ell+1}}\zeta(\ell+2-m)$: the $\zeta$-sum of~\eqref{eq:FP}.
\item For $\ell<m-1$ the singular $\Gamma(m-\ell-1)(2\pi\delta)^{\ell-m+1}$ crosses the $\delta^{m-\ell-1}$ coefficient
$\binom{s-1-\ell}{m-\ell-1}$ of $(1+\delta)^{s-1-\ell}$; by the falling-factorial identity
$(s-1)_{(\ell)}(m-\ell-2)!\binom{s-1-\ell}{m-\ell-1}=(s-1)_{(m-1)}/(m-\ell-1)$ (and its $k{-}s$ partner) these sum to
\begin{equation}\label{eq:harm}
\frac{c^+_{m-1}(s)}{(2\pi)^{m}}\sum_{\ell=0}^{m-2}\frac{1}{m-\ell-1}=\frac{c^+_{m-1}(s)\,H_{m-1}}{(2\pi)^{m}},
\end{equation}
the harmonic term of~\eqref{eq:FP}.
\end{enumerate}
The remaining $\ell=m-1$ term is just $\log(2\pi\delta)$, which cancels in the
$\delta\to0$ limit of the S-contour. We now turn to this S-contour's contribution. 
%The remaining $\ell=m-1$ term yields no finite part --- only a $\log(2\pi\delta)$, which cancels in the
%$\delta\to0$ limit of the S-contour, which we now turn to. 

On $\hat\gamma_f^S$ the pure pole $\sum_m r^*_{f,i}(m)(\tau-i)^{-m}$, integrated by principal value
against $\tau^{s-1}$ along the axis ($\tau=i(1+v)$, running from $\tau_0$ to $S\tau_0=i/(1+\delta)$,
so that $-\delta/(1+\delta)\le v\le\delta$), is elementary. Expanding
$(1+v)^{s-1}=\sum_{t}\tfrac{(s-1)_{(t)}}{t!}v^{t}$, the terms $t\ge m$ are regular and $O(\delta)$, the
term $t=m-1$ integrates to $\log(1+\delta)=O(\delta)$, and each $t\le m-2$ contributes
\begin{equation}\label{eq:Sexp}
\mathrm{PV}\!\int_{-\delta/(1+\delta)}^{\delta}\!\frac{v^{t}}{v^{m}}\,dv
=\frac{\delta^{t-m+1}}{t-m+1}\Big[1-(-1)^{t-m+1}(1+\delta)^{m-t-1}\Big].
\end{equation}
Expanding $(1+\delta)^{m-t-1}$, its terms $\delta^{j}$ with $j<m-t-1$ carry the endpoint poles
$\delta^{-(m-1)},\dots,\delta^{-1}$, while the top term $j=m-t-1$ leaves the $\delta^0$ finite part
\begin{equation}\label{eq:Spoly}
i^{-m}\,\mathrm{PV}\!\int_{-\delta/(1+\delta)}^{\delta}\frac{(1+v)^{s-1}}{v^{m}}\,dv\;\Big|_{\delta^0}
=i^{-m}(-1)^{m-1}\sum_{t=0}^{m-2}\frac{(-1)^{t}(s-1)_{(t)}}{t!\,(m-t-1)},
\end{equation}
the polynomial term of~\eqref{eq:FP}. Collecting, the finite part of block $m$ is
$\tfrac{(-2\pi i)^m}{(m-1)!}r^*_{f,i}(m)\,\mathrm{FP}_{m-1}(s,k)$.

\emph{Divergences cancel by~\eqref{eq:Laurent}.} The residual pieces are the $\delta\to0$ divergences
of the pole part $\sum_m r^*_{f,i}(m)(\tau-i)^{-m}$: the powers $\delta^{-(m-1)},\dots,\delta^{-1}$,
carried by the endpoints of $\hat\gamma_f^S$ straddling $i$ as in~\eqref{eq:Sexp} and by the
$\ell<m-1$ terms of~\eqref{eq:Tsum}, together with a $\log\delta$ carried by
$\Li_1(e^{-2\pi\delta})$ at $\ell=m-1$ alone. At order $\mu$ the coefficient of the powers is
\begin{equation}\label{eq:walldiv}
\sum_{m=\mu}^{P} r^*_{f,i}(m)\binom{m}{\mu}i^{m}\;-\;i^{k}\sum_{m=\mu}^{P} r^*_{f,i}(m)\binom{k}{m-\mu}i^{-m},
\end{equation}
precisely the difference of the two sides of the modularity relation~\eqref{eq:Laurent}, hence $0$ for
$f\in F_k$. The $\log\delta$ coefficient, summed over blocks with the normalisation
$\tfrac{(-2\pi i)^m}{(m-1)!}$, is
$-\sum_{m=1}^{P}(-i)^m r^*_{f,i}(m)\big[\binom{s-1}{m-1}+i^k\binom{k-s-1}{m-1}\big]$, which the same
relations annihilate. Every $\delta^{-\mu}$ and $\log\delta$ term therefore cancels and the $\delta\to0$ limit
exists. Conversely, weights violating~\eqref{eq:Laurent} at order $\mu$ leave that piece uncancelled (at $\mu=1$, a $\log\delta$), so the limit exists only for $f\in F_k$.% break~\eqref{eq:Laurent} at order $r$ and that divergence survives (at $r=1$, a $\log\delta$).

The finite part and the (vanishing) divergences give~\eqref{eq:wallvalue}.
\end{proof}


\begin{theorem}\label{thm:mero}
Let $f\in F_k$ have poles $\tau_p$ of order $P_p$ with ${\rm Im}\,\tau_p\ge1$ in the strip
$|{\rm Re}\,\tau|\le\tfrac12$, pole-projected Fourier coefficients $c_{Rf}(n)$
(Definition~\ref{def:cRf}), and projection weights $r^*_{f,\tau_p}(m)$
(Definition~\ref{def:proj}). In the minimal class,
\begin{equation}\label{eq:mero}
L^*(f,s)=-c_{Rf}(0)\Big(\tfrac1s+\tfrac{i^k}{k-s}\Big)
+\sum_{n\ne0}c_{Rf}(n)\,G_{s,k}(n)
+\sum_{\tau_p\ne i}\ \sum_{m=1}^{P_p}r^*_{f,\tau_p}(m)\,\frac{(-2\pi i)^m}{(m-1)!}\,I_{m-1}(s,\tau_p)
+L^*(\widehat P_i f,\,s),
\end{equation}
with $I_{m-1}$ from Lemmas~\ref{lem:polylogcf}--\ref{lem:polylogedge} and the last term
(present only for a pole at $i$) from Lemma~\ref{lem:wallvalue}. The right side is entire in
$s$ apart from at most simple poles at $s=0$ and $s=k$, and reduces to the weakly-holomorphic
$L^*$ of Theorem~\ref{thm:weakL} when $f$ is holomorphic on $\HH$.
\end{theorem}

\begin{proof}
Split $f=(I-\widehat P)f+\widehat Pf$. By Lemma~\ref{lem:projspace} the Fourier series of the
pole-projected part converges on $\hat\gamma_f^T$; Theorem~\ref{thm:weakL} applied term by term
gives its $L^*$ as the first two sums. The projection
$\widehat Pf=\sum_{\tau_p}\sum_m r^*_{f,\tau_p}(m)\,\frac{(-2\pi i)^m}{(m-1)!}\,\Li_{1-m}(e^{2\pi i(\tau-\tau_p)})$
contributes, for each $\tau_p\ne i$, the L-integrals $I_{m-1}(s,\tau_p)$ of
Definition~\ref{def:IN}, evaluated by Lemmas~\ref{lem:polylogcf}--\ref{lem:polylogedge}; the
pole at $i$ contributes $L^*(\widehat P_i f,s)$ of Lemma~\ref{lem:wallvalue}. Entirety and the
pole locations follow from those of $G_{s,k}$ and the $\delta_{N,0}$ terms; when $f$ is
holomorphic on $\HH$, $\widehat Pf=0$ and $c_{Rf}=c_f$, recovering Theorem~\ref{thm:weakL}.
\end{proof}

\subsection{Relations of this work to L-functions for $f \in F_k$ in~\cite{1806}}

We briefly pause to make two remarks that connect the L-functions for $f \in F_k$ defined in this note 
to the slightly different definition in Ref.~\cite{1806}.

\begin{remark}\label{rem:residues}
\emph{Casual inspection of the structure of the $I_N(s, \tau_p)$-functions entering into Theorem~\ref{thm:mero} 
would lead one to conclude that they substantively differ from those in~\cite{1806}, where the author first

However, our L-function has $1/s$-residue contributions from all \emph{simple} poles located above the 
${\rm Im}(\tau_p) > 1$-line. This naive difference is immaterial, and does not impact the results of the 
residue theorems for the $1/s$-poles in~\cite{1806}, for the simple reason that in Ref.~\cite{1806}, the 
reference contour was implicitly at $\tau_0 \to i \infty$. This was valid in the restricted context of~\cite{1806}, 
as that paper only focused on $f \in F_k$ where $f = \sum_{n \geq 0} c_f(n) q^n$; poles at the $\SLZ$ cusp 
were explicitly excluded, which allowed the limit $\tau_0 \to i \infty$ to be smooth and safe. However, in this 
work, we make a different reference contour --- that in Definition~\ref{def:null_homotopy}. }
%The poles at $s=0$ and $s=k$ belong to the class, not to $f$ alone. Here the $T$-segment sits at
%height $1+\delta$, so each pole above it is reached through the inversion of
%Lemma~\ref{lem:polylogcf}, contributing
%$-2\pi i\,r^*_{f,\tau_p}(1)\big(\tfrac1s+\tfrac{i^k}{k-s}\big)$; the L-function of~\cite{1806}

\end{remark}

\begin{remark}\label{rem:raised}
\emph{The $1/s$ residue is carried entirely by the $T$-segment: $L^*_S$ is a finite integral of
$f\,\tau^{s-1}$ and is entire in $s$, while $\tk$ has a simple pole at $s=0$ whose coefficient
is $\int_{\hat\gamma_f^T}f\,d\tau$, the constant Fourier mode of $f$ in the strip containing
that segment. Deforming $\hat\gamma_f^T$ to pass above every pole of $f$ therefore returns

\end{remark}

\subsection{Period polynomials for generic $f \in F_k$}

In this section, we use our L-functions to assign period polynomials $r_f \in W \subset V_{k-2}$ to
general elements $f \in F_k$. The route is straightforward when $f$ has poles exclusively outside a
certain region $\mathcal{T}\subset\HH$, lying in the strip $0\le{\rm Re}\,\tau\le1$ directly beneath
the $T$-segment. We denote the set of such functions as $F_k^\circ \subset F_k$. For every
$f \in F_k^\circ$ the period polynomial constructed from the special L-values of $L^*(f,s)$ of
Theorem~\ref{thm:mero} is immediately an element of $W$. However, when $f$ has poles within
$\mathcal{T}$, more care is needed. Ref.~\cite{BrownFonseca2025} gives a parallel construction of
$r_f$ for $f \in F_k$, which we re-present alongside one of our own. It is not obvious to us
which construction should be preferred.

To begin, we define $\mathcal{T}$, then $F_k^\circ$, and then $r_f$ for $f \in F_k^\circ$:

\begin{definition}\label{def:Tri}
Let $\mathcal{T}\subset\HH$ be the region
\begin{equation}\label{eq:Tri}
\mathcal{T}:=\Big\{\tau\in\HH:\ 0\le{\rm Re}\,\tau\le1,\ \tfrac12\le{\rm Im}\,\tau\le1,\
\big|\tau-\tfrac{i}{2}\big|\ge\tfrac12,\ \big|\tau-1-\tfrac{i}{2}\big|\ge\tfrac12\Big\}.
\end{equation}
\end{definition}

$\mathcal{T}$ is the closed curvilinear triangle with vertices $i$, $1+i$ and $\tfrac{1+i}{2}$,
bounded above by the line ${\rm Im}\,\tau=1$ and below by its $\SLZ$-translates through the cusps
$0$ and $1$, the circles $|\tau-\tfrac{i}{2}|=\tfrac12$ and $|\tau-1-\tfrac{i}{2}|=\tfrac12$, which
meet each other tangentially at $\tfrac{1+i}{2}$. Its three sides are the $T$-, $UT$- and
$U^2T$-translates of $\hat\gamma^T_f$, meeting at those vertices; since $(TS)^3=1$ the last of these
is $S^{-1}$, so $\partial\mathcal{T}$ is the $U$-orbit of $T\hat\gamma^T_f$. Accordingly $U$ cycles
$i\to1+i\to\tfrac{1+i}{2}\to i$ and $\mathcal{T}$ is $U$-invariant. Note that $\hat\gamma^T_f$ runs
from $i-1$ to $i$, so it is $T\hat\gamma^T_f$, not the segment itself, that caps $\mathcal{T}$; the
region directly beneath the contour is $T^{-1}\mathcal{T}$. The bound
${\rm Im}\,\tau\ge\tfrac12$ is not redundant: dropping it leaves a second region below the point of
tangency, widening to the full interval $0\le{\rm Re}\,\tau\le1$ as ${\rm Im}\,\tau\to0$ and so of
infinite hyperbolic area. Figure~\ref{fig:Tregion} draws $\mathcal{T}$, and locates it against the
fundamental domain and its translates.

\begin{definition}\label{def:Fcirc}
Define
\begin{equation}\label{eq:Fcirc}
F_k^{\circ}:=\Big\{f\in F_k\ :\ f^{-1}(\infty)\cap\mathcal{T}=\emptyset\ \ \text{and}\ \
\int_{\hat\gamma^T_f}f(\tau)\,d\tau=0\Big\}.
\end{equation}
\end{definition}

The second condition is the vanishing of the constant Fourier mode of $f$ in the strip containing
$\hat\gamma^T_f$, and reduces to $c_f(0)=0$ when $f$ has no poles above that segment.

\begin{definition}\label{def:rfmero}
For $f\in F_k^{\circ}$, let $r_f$ be given by Definition~\ref{def:rf}, with $L^*(f,s)$ evaluated in
the reference class of Definition~\ref{def:null_homotopy}.
\end{definition}

Our definitions of $\mathcal{T}$ and $F_k^\circ$ are designed so that the proof of
Lemma~\ref{lem:rfW}, which used $f\in S^!_k$ in exactly two places, extends to $f\in F_k^\circ$:

\begin{lemma}\label{lem:rfWmero}
For $f\in F_k^{\circ}$, the period polynomial $r_f\in W\subset V_n$.
\end{lemma}
\begin{proof}
Lemma~\ref{lem:rfW} invokes $f\in S^!_k$ only through $\hat C_T=0$ and $\hat C_{\pm I}=0$, and the
two conditions of~\eqref{eq:Fcirc} supply exactly these.

For $\hat C_T$, the computation in the proof of Lemma~\ref{lem:cocycle} gives
$\hat C_T=-(2\pi i)^{n+1}A_0\int_{\hat\gamma^T_f}f(\tau)\,d\tau$, which vanishes by the second
condition of~\eqref{eq:Fcirc}.

For $\hat C_{\pm I}$, $n$ is even, so $-I$ acts trivially on $V_n$ and the coboundary drops out,
leaving $\hat C_{-I}=C_{-I}$. Since $U^3=-I$, the relation at $-I$ is the $(1+U+U^2)$ loop, and by
Definition~\ref{def:Tri} its geometric realisation is $\partial\mathcal{T}$: thus $C_{-I}$ is the
integral of $f(\tau)(X-\tau Y)^{n}$ around the closed contour $\partial\mathcal{T}$. This is the
step the region $\mathcal{T}$ exists to serve. The integrand is holomorphic wherever $f$ is, and the
first condition of~\eqref{eq:Fcirc} says precisely that $f$ has no pole on or within $\mathcal{T}$,
so Cauchy's theorem gives $C_{-I}=0$. Were even one pole of $f$ to lie in $\mathcal{T}$, the loop
would enclose it and $C_{-I}$ would collect its residue instead; that residue is the defect computed
in Lemma~\ref{lem:rfWFk} below, and it is what obstructs $r_f\in W$ for general $f\in F_k$.

The remaining steps are formal consequences of the cocycle law and apply verbatim.
\end{proof}

The two conditions in~\eqref{eq:Fcirc} are not independent. An orbit meets $\mathcal{T}$ exactly
when its representative $p$ in the standard fundamental domain has ${\rm Im}\,p\le1$: for such a
$p$, $|p|\ge1$ gives $|p-\tfrac{i}{2}|^2=|p|^2-{\rm Im}\,p+\tfrac14\ge\tfrac14$ and likewise at the
cusp $1$, so $p$ lies outside both circles, while ${\rm Im}\,p\ge\tfrac{\sqrt3}{2}>\tfrac12$ holds
throughout the standard fundamental domain; hence $p$ lies in $\mathcal{T}$ after a translation.
Conversely no point of $\mathcal{T}$ can be raised above the contour, since
${\rm Im}(\gamma\tau)={\rm Im}\,\tau/|c\tau+d|^{2}>1$ would need $|c\tau+d|^{2}<{\rm Im}\,\tau\le1$:
for $c=0$ this asks $|d|<1$; for $|c|\ge2$ it fails because
$|c\tau+d|^{2}\ge c^{2}({\rm Im}\,\tau)^{2}\ge1$ once ${\rm Im}\,\tau\ge\tfrac12$; and for $|c|=1$ it
asks exactly that $\tau$ lie inside $|\tau-m-\tfrac{i}{2}|<\tfrac12$ for some $m\in\ZZ$, which is
empty on $0\le{\rm Re}\,\tau\le1$ unless $m\in\{0,1\}$. The last two bounds of~\eqref{eq:Tri} are
therefore not ad hoc: ${\rm Im}\,\tau\ge\tfrac12$ disposes of $|c|\ge2$, and the two excluded disks
are the $|c|=1$ isometric circles. So $\mathcal{T}$ is precisely the part of the strip
$0\le{\rm Re}\,\tau\le1$ at or below the contour that no $\gamma\in\SLZ$ can lift above it, and
\begin{equation}\label{eq:foldT}
\mathcal{T}_{\mathcal{F}}\;:=\;\SLZ\!\cdot\!\mathcal{T}\,\cap\,\mathcal{F}
\;=\;\{\tau\in\mathcal{F}:{\rm Im}\,\tau\le1\}.
\end{equation}
The first
condition therefore says that every pole of $f$ lies above $\hat\gamma^T_f$. This holds for a
fraction $3/\pi\approx0.955$ of pole positions in hyperbolic measure, the region
${\rm Im}\,\tau>1$ having area $1$ against $\pi/3$ for the fundamental domain. But it also
separates $\hat\gamma^T_f$ from the cusp, so the strip constant is $c_f(0)$ less $2\pi i$ times the
residues in one period strip, and the second condition is then a further linear condition rather
than a consequence of $c_f(0)=0$. A form vanishing at the cusp whose poles form a single orbit of
simple poles has strip constant exactly $2\pi i\,r^*_{f,p}(1)$, and so lies in $F_k^{\circ}$ only
if that residue vanishes; membership requires $c_f(0)$ tuned against the residues, as for
$E_{12}\big(j-j_1\big)/\big(j-j(2i)\big)$ with $j_1$ chosen
suitably~\cite{VerifyScripts}. Figure~\ref{fig:TF} draws $\mathcal{T}_{\mathcal{F}}$, whose area is
$\int_{-1/2}^{1/2}\big((1-x^2)^{-1/2}-1\big)\,dx=\tfrac{\pi}{3}-1$.

Poles inside $\mathcal{T}$ do not destroy this structure; they shift $r_f$ by an explicit
polynomial. As the sides of $\mathcal{T}$ are $U$-transports of one path, the enclosed pole set
is a union of $U$-orbits $\{p,\,Up,\,U^2p\}$, and each enclosed pole contributes its residue.

\begin{definition}\label{def:Qf}
Let $f\in F_k$ have a pole of order $m_p$ at each $p\in f^{-1}(\infty)\cap\mathcal{T}^{\circ}$,
with projection weights $r^*_{f,p}(\alpha)$ (Definition~\ref{def:proj}). The \emph{polar defect
polynomial} of $f$ is
\begin{align}\label{eq:Qf}
&Q_f(X,Y):=(2\pi i)^{n+2}\!\!\sum_{p\,\in\,f^{-1}(\infty)\cap\mathcal{T}^{\circ}}\ \sum_{\alpha=1}^{m_p}
r^*_{f,p}(\alpha)\,B_\alpha(p;X,Y),
\\
&B_\alpha(p;X,Y):=\binom{n}{\alpha-1}(-Y)^{\alpha-1}(X-pY)^{\,n-\alpha+1}.
\end{align}
Simple poles contribute the $\alpha=1$ term alone, $r^*_{f,p}(1)(X-pY)^n$.
\end{definition}

\begin{lemma}\label{lem:rfWFk}
Let $f\in F_k$ satisfy $\int_{\hat\gamma^T_f}f(\tau)\,d\tau=0$. Then
\begin{equation}\label{eq:rfWFk}
r_f\big|_{(1+S)}=0
\qquad\text{and}\qquad
r_f\big|_{(1+U+U^2)}=-\,Q_f .
\end{equation}
In particular $r_f\in W$ if and only if $Q_f=0$, and Lemma~\ref{lem:rfWmero} is the case
$f^{-1}(\infty)\cap\mathcal{T}^{\circ}=\emptyset$.
\end{lemma}

\begin{proof}
Both relations are values of the cocycle of Lemma~\ref{lem:cocycle} on the two relators of
$\SLZ$, evaluated on the loops their words induce; the loops, not the group elements, carry
the content. For the first, $\hat C_{S^2}=\hat C_S|_{(1+S)}=r_f|_{(1+S)}$; as $n$ is even the
coboundary drops out, so this is the integral of $(2\pi i)^{n+1}f(\tau)(X-\tau Y)^n$ over the
$S^2$-loop, the concatenation of $\hat\gamma^S_f$ with its $S$-image. At $\delta=0$ that loop is
degenerate. For $\delta>0$ it retraces itself: the reference $S$-segment runs up the imaginary axis
from $i/(1+\delta)$ to $i(1+\delta)$, and $S$ carries $iy$ to $i/y$, mapping the segment onto itself
with reversed orientation, so the two legs cancel. Definition~\ref{def:null_homotopy} admits no pole
in $1<{\rm Im}\,\tau\le1+\delta$, and applying $S$ excludes them from
$1/(1+\delta)\le{\rm Im}\,\tau<1$ as well, so the only pole that can meet the segment is $\tau=i$.
There $S$ acts with derivative $-1$, preserving the symmetric excision, and the two principal values
cancel with the rest. Either way $r_f|_{(1+S)}=0$~\cite{VerifyScripts}. For the second, the
hypothesis
$\int_{\hat\gamma^T_f}f\,d\tau=0$ gives $\hat C_T=0$ as in the proof of
Lemma~\ref{lem:rfWmero}, hence $\hat C_U=r_f$ and
$r_f|_{(1+U+U^2)}=\hat C_{U^3}=C_{-I}$, the integral of the same $1$-form over the
$(TS)^3$-loop: the three $U$-transports of the $U$-path traverse $\partial\mathcal{T}$ once
clockwise, with a common integrand by modular covariance. The residue theorem then gives
$C_{-I}=-(2\pi i)^{n+1}\cdot2\pi i\sum_{p\in\mathcal{T}^{\circ}}
\mathop{\rm Res}_{\tau=p}\big[f(\tau)(X-\tau Y)^n\big]$, and since
$\tfrac{1}{(\alpha-1)!}\partial_\tau^{\,\alpha-1}(X-\tau Y)^n=B_\alpha(\tau;X,Y)$,
that residue is $\sum_{\alpha\le m_p}r^*_{f,p}(\alpha)B_\alpha(p;X,Y)$; hence
$C_{-I}=-Q_f$.
\end{proof}

A pole of $f$ can meet $\partial\mathcal{T}$ only at the elliptic point $i$. The sides of
$\mathcal{T}$ are $\Gamma$-translates of $\hat\gamma^T_f$ and the poles of $f$ are
$\Gamma$-invariant, so a pole on $\partial\mathcal{T}$ would lie on $\hat\gamma^T_f$ itself, which
Definition~\ref{def:null_homotopy} excludes by lifting the base-point to $i(1+\delta)$; when
$\delta>0$ the $S$-segment no longer degenerates and $\partial\mathcal{T}$ acquires three further
sides translating $\hat\gamma^S_f$, so~\eqref{eq:Tri} is the $\delta=0$ form, and $i$ is taken
there by principal value.

The defect is removed by subtracting from $f$ reference forms carrying the same poles. Write
$\SLZ\!\cdot\!p:=\{\gamma\cdot p:\gamma\in\SLZ\}$ for the orbit of $p\in\HH$.

Write $\Gamma_p\le\SLZ$ for the stabiliser of $p$, and let
$A_p:=\{\alpha:1\le\alpha\le m,\ B_\alpha(p;\cdot,\cdot)\ \text{has nonzero
}\Gamma_p\text{-average}\}$. At a non-elliptic $p$ one has $\Gamma_p=\{\pm I\}$ and $A_p$ is all of
$\{1,\dots,m\}$; at $\tau=i$ and $\tau=e^{i\pi/3}$ it is a proper subset, namely the
$\alpha\equiv k/2\bmod2$ and the $\alpha\equiv-k\bmod3$ respectively~\cite{VerifyScripts}, which are
exactly the pole orders the valence formula permits there.

\begin{definition}\label{def:reffam}
A \emph{reference family} at $p$ of depth $m$ is a set $\{\Psi^{(\alpha)}_p\}_{\alpha\in A_p}$ of
weight-$k$ meromorphic modular forms, each holomorphic on $\HH\smallsetminus\SLZ\!\cdot\!p$ with
constant Fourier coefficient zero, such that
\begin{equation}\label{eq:psiform}
\mathop{\rm Res}_{\tau=p}\big[\Psi^{(\alpha)}_p(\tau)(X-\tau Y)^n\,d\tau\big]
=\sum_{\beta\in A_p,\ \beta\le\alpha}\lambda_{\alpha\beta}\,B_\beta(p;X,Y),
\qquad \lambda_{\alpha\alpha}\neq0 ,
\end{equation}
that is, triangular in the $B_\beta$ with nonvanishing diagonal.
\end{definition}

Two constructions. Taking $\Psi^{(\alpha)}_p:=g_\alpha/(j-j(p))^{\alpha}$ with $g_\alpha\in M_k$ not
vanishing at $p$ gives a reference family whenever such $g_\alpha$ exist, since a pole of exact
order $\alpha$ contributes $\lambda_{\alpha\alpha}=r^*_{\Psi^{(\alpha)},p}(\alpha)\neq0$ and nothing
beyond. Alternatively the Poincar\'e series
$\Psi^{\,\alpha-1,\,n-\alpha+1}_{\SLZ}(\,\cdot\,,p)$ of~\cite[Def.\ 3.1]{BrownFonseca2025} form one
at every $p$: they have weight $k$, are holomorphic off $\SLZ\!\cdot\!p$ with poles there of order
at most $\alpha$, and vanish at the cusp~\cite[Prop.\ 3.3]{BrownFonseca2025}, while
by~\cite[Cor.\ 3.11]{BrownFonseca2025} their residue is a multiple of
$(X-\bar pY)^{\alpha-1}(X-pY)^{n-\alpha+1}$, which on expanding
$X-\bar pY=(X-pY)-(p-\bar p)(-Y)$ is triangular in the $B_\beta$ with diagonal entry proportional to
$(p-\bar p)^{\alpha-1}$. That frame agrees with ours, their $(X,Y)$ being dual to $(1,-\tau)$ so
that their $X-\tau Y$ is the class of $du$ on $\CC/(\ZZ+\ZZ\tau)$; the multiple is a local character
sum $\kappa^{\alpha-1,n-\alpha+1}$, equal by~\cite[Rem.\ 3.12]{BrownFonseca2025} to $|\Gamma_p|$ for
$\alpha\in A_p$ and to zero otherwise.

\begin{theorem}\label{thm:period}
Let $f\in F_k$ have $c_f(0)=0$, with poles in $\HH$ on the orbits
$\SLZ\!\cdot\!p_1,\dots,\SLZ\!\cdot\!p_N$ of orders $m_1,\dots,m_N$, elliptic or not; a pole at the
cusp is allowed. Fix a reference family $\Psi^{(\alpha)}_{p_a}$ at each $p_a$ of depth $m_a$, and
evaluate every period polynomial below in the reference class of $f$. This is available to the
$\Psi^{(\alpha)}_{p_a}$ as well: their poles lie among those of $f$, so
Definition~\ref{def:null_homotopy} admits none of them in $1<{\rm Im}\,\tau\le1+\delta$, and the
same contour serves every integrand. There are unique $c_{a,\alpha}\in\CC$, for $1\le a\le N$ and
$\alpha\in A_{p_a}$, such that $f-\sum_{a,\alpha}c_{a,\alpha}\Psi^{(\alpha)}_{p_a}$ is holomorphic
on $\HH$, and then
\begin{equation}\label{eq:period}
\tilde r_f\ :=\ r_f-\sum_{a=1}^{N}\sum_{\alpha\in A_{p_a}}c_{a,\alpha}\,r_{\Psi^{(\alpha)}_{p_a}}
\ \in\ W .
\end{equation}
Both $r_f$ and each $r_{\Psi^{(\alpha)}_{p_a}}$ are given by Definition~\ref{def:rf}, so
$\tilde r_f$ is assembled from the values $L^*(\,\cdot\,,\ell+1)$, $0\le\ell\le n$, of the
finite-contour $L$-function alone, and the $c_{a,\alpha}$ are determined by the principal parts
of $f$.
\end{theorem}
\begin{proof}
Fix $a$ and write $p=p_a$. The residue of $f(\tau)(X-\tau Y)^n\,d\tau$ at $p$ is
$\sum_{\alpha\le m_a}r^*_{f,p}(\alpha)B_\alpha(p)$ as in~\eqref{eq:Qf}. That residue is
$\Gamma_p$-invariant: the $1$-form is $\SLZ$-invariant and $\Gamma_p$ fixes $p$, so averaging over
$\Gamma_p$ leaves it unchanged. Hence it lies in the span of $\{B_\beta(p):\beta\in A_p\}$, which is
the image of that average. The residues~\eqref{eq:psiform} of the reference family are triangular in
the same $B_\beta$, $\beta\in A_p$, with nonvanishing diagonal, hence a basis of that span; so the
$c_{a,\alpha}$ matching the two exist and are unique. At a non-elliptic $p$ this is the whole span
and the statement is the evident one; at $\tau=i$ or $e^{i\pi/3}$ the individual Laurent
coefficients $r^*_{f,p}(\alpha)$ with $\alpha\notin A_p$ need not vanish, but their contributions
cancel in the invariant combination.

Since $\Psi^{(\alpha)}_{p_a}$ has no pole off $\SLZ\!\cdot\!p_a$, the orbits do not interfere and
the choices are independent in $a$. Matching the residue at $p_a$ matches the entire principal part
there, and covariance $r^*_{g,\gamma p}(\alpha)=(cp+d)^{n}r^*_{g,p}(\alpha)$ propagates this along
the orbit, so the difference has no pole in $\HH$ and lies in $M^!_k$. Its constant Fourier
coefficient is $c_f(0)=0$, so it lies in $S^!_k$, where Lemma~\ref{lem:rfW} gives a period
polynomial in $W$; $r$ is linear.
\end{proof}

The hypothesis $c_f(0)=0$ costs nothing. For arbitrary $f\in F_k$ the form $f-c_f(0)E_k$ has the
same poles in $\HH$ and vanishing constant term, so Theorem~\ref{thm:period} applies to it, and
\begin{equation}\label{eq:rhat}
\hat r_f\ :=\ \tilde r_{f-c_f(0)E_k}\ +\ c_f(0)\,r_{E_k}
\end{equation}
is defined for every $f\in F_k$. It satisfies $\hat r_f|_{(1+S)}=0$, since both terms do, and
$\hat r_f|_{(1+U+U^2)}=c_f(0)\,r_{E_k}|_{(1+U+U^2)}$. So $\hat r_f$ lies in $W$ exactly when
$c_f(0)=0$, and otherwise in $W\oplus\CC\,r_{E_k}$; this is the usual Eisenstein correction, already
present for holomorphic $f\in M_k\smallsetminus S_k$, and not a feature of the poles.

\begin{corollary}\label{cor:explicit}
Let $k=12$ and $f=\Delta/(j-j_0)$ with $j_0\notin\{0,1728\}$, and take the reference family
$\Psi^{(1)}_p:=E_{12}/(j-j_0)$ at a pole $p$. Then $N=1$, $m_1=1$,
\begin{equation}\label{eq:crational}
c_{1,1}=\frac{\Delta(p)}{E_{12}(p)}=\frac{691}{691\,j_0-432000},
\qquad
\tilde r_f=-\,c_{1,1}\,r_\Delta .
\end{equation}
In particular $c_{1,1}$ is rational whenever $j_0$ is, with no assumption of complex
multiplication, and $\tilde r_f$ is then a rational multiple of the period polynomial of $\Delta$.
\end{corollary}
\begin{proof}
$E_{12}/(j-j_0)$ is holomorphic off $\SLZ\!\cdot\!p$, has a simple pole there with nonzero residue,
and vanishes at the cusp, so it is a reference family of depth $1$. From $E_4^3=j\Delta$ and
$E_6^2=(j-1728)\Delta$ one gets $E_{12}=\Delta\,(691j-432000)/691$, so $\Delta/E_{12}$ takes the
stated value at $p$, and
substituting gives $f-c_{1,1}\Psi^{(1)}_p=-c_{1,1}\Delta$ identically. Apply
Theorem~\ref{thm:period} and linearity.
\end{proof}

\begin{corollary}\label{cor:CM}
Let $d>4$ have class number one, let $\alpha_d$ be the associated CM point, and set
$j_d:=j(\alpha_d)\in\ZZ$. Write $D:=\tfrac{1}{2\pi i}\tfrac{d}{d\tau}$, and let $H_d$ be the Hilbert
class polynomial, whose roots are the singular moduli of discriminant $d$~\cite{ZagierTSM}. Then
\begin{equation}\label{eq:Lambdad}
\Lambda_d:=D\log H_d(j)=\frac{-E_4^2E_6}{\Delta}\,\frac{H_d'(j)}{H_d(j)}
=\frac{-E_4^2E_6}{\Delta\,(j-j_d)}
\end{equation}
is a weight-two form with a simple pole of residue $1/2\pi i$ at every CM point of discriminant $d$,
the last equality holding because $h(d)=1$ makes $H_d(j)=j-j_d$. Moreover
\begin{equation}\label{eq:fd}
f_d:=\Delta E_6\,\Lambda_d=\frac{-E_4^2E_6^2}{j-j_d}\ \in\ F_{20}
\end{equation}
has integral $q$-expansion, vanishes at the cusp, and no pole of $f_d$ lies in $\mathcal{T}$; hence
$\tilde r_{f_d}=r_{f_d}-c_{f_d}r_{E_{20}}\in W$ is determined, with no reference family and no
choice. Explicitly, with $\Delta E_8$ the normalised cusp form spanning $S_{20}$,
\begin{equation}\label{eq:rfd}
\tilde r_{f_d}=\frac{-92207808}{174611\,j_d-209520000}\;r_{\Delta E_8}.
\end{equation}
\end{corollary}
\begin{proof}
$Dj=-E_4^2E_6/\Delta$ gives the second expression in~\eqref{eq:Lambdad}, and $\Delta E_6\Lambda_d$
reduces to~\eqref{eq:fd}; the coefficients are integral because $j_d\in\ZZ$ for class number
one. Since $h(d)=1$ the only reduced form of discriminant $d$ has leading coefficient $1$, so
${\rm Im}\,\alpha_d=\tfrac{\sqrt d}{2}>1$ for $d>4$; by the criterion above, its orbit misses
$\mathcal{T}$, and $Q_{f_d}=0$. For~\eqref{eq:rfd}, $E_8=E_4^2$ and $E_6^2=(j-1728)\Delta$ give
\begin{equation}\label{eq:fdsplit}
f_d=\frac{-\Delta E_8\,(j-1728)}{j-j_d}=-\Delta E_8\;-\;(j_d-1728)\,\frac{\Delta E_8}{j-j_d}.
\end{equation}
Running the computation of Corollary~\ref{cor:explicit} with $(\Delta,E_{12})$ replaced by
$(\Delta E_8,E_{20})$, where $E_{20}=\Delta E_8\,(174611j-209520000)/174611$, the reference family
$E_{20}/(j-j_d)$ gives $c=174611/(174611j_d-209520000)$ and
$\tilde r_{\Delta E_8/(j-j_d)}=-c\,r_{\Delta E_8}$. Since $\Delta E_8\in S_{20}$, linearity of $L^*$
turns~\eqref{eq:fdsplit} into $\tilde r_{f_d}=\big[(j_d-1728)c-1\big]r_{\Delta E_8}$, and
$209520000-1728\cdot174611=-92207808$ gives~\eqref{eq:rfd}.
\end{proof}

These are the discriminants $d=7,8,11,19,43,67,163$. For $d=3$ and $d=4$ the CM point is elliptic
and ${\rm Im}\,\alpha_d\le1$, so both the elliptic bookkeeping of Definition~\ref{def:reffam} and a
correction are needed; and for $h(d)>1$ the reduced forms with leading coefficient $a>1$ give CM
points as low as ${\rm Im}=\tfrac{\sqrt3}{2}$, whose orbits do meet $\mathcal{T}$.

Definition~\ref{def:reffam} constrains the $\Psi^{(\alpha)}_p$ only through their poles, so it does
not determine them, and $\tilde r_f$ inherits that freedom. The freedom is exactly measured. If
$\{\Psi^{(\alpha)}_p\}$ and $\{\Psi'^{(\alpha)}_p\}$ are two reference families and
$c_{a,\alpha},c'_{a,\alpha}$ the coefficients they produce for the same $f$, then
$\sum c_{a,\alpha}\Psi^{(\alpha)}_{p_a}$ and $\sum c'_{a,\alpha}\Psi'^{(\alpha)}_{p_a}$ have the
same principal part as $f$ at every pole, so their difference $g$ is holomorphic on $\HH$ with
$c_g(0)=0$, that is $g\in S^!_k$, and
\begin{equation}\label{eq:ambig}
\tilde r_f-\tilde r\,'_f=r_g\ \in\ W
\end{equation}
by Lemma~\ref{lem:rfW}. The ambiguity is therefore not merely $W$-valued by accident: it is the set
$r(S^!_k)\subseteq W$ of period polynomials of the pole-free forms, which is precisely what
Definition~\ref{def:reffam} cannot see.

That set is everything. Adding any $h\in S^!_k$ to each $\Psi^{(\alpha)}_{p_a}$ yields another
reference family, since $h$ is holomorphic with $c_h(0)=0$ and so alters neither the pole locus nor
any residue in~\eqref{eq:psiform}; the matched coefficients are unchanged, and $\tilde r_f$ moves by
$r_g$ with $g=\sum c_{a,\alpha}h_\alpha$ arbitrary in $S^!_k$. And $r$ is already onto $W$ there,
for two independent reasons. So long as the determinant pairing the $\dim S_k$ periods against the
$\dim S_k$ quasi-periods is nonzero, which is the Legendre relation~\cite{BrownHain2018}, the
$2\dim S_k$ period polynomials of the cusp forms and of their weakly holomorphic partners are
independent modulo $p_0$, and so span a complement to $\CC\,p_0$ in $W$. The remaining Eisenstein
direction $p_0$ is supplied separately, by the Bol image $\partial^{k-1}M^!_{2-k}$: it lies in
$S^!_k$, since $\partial^{k-1}$ annihilates the constant Fourier mode, and its period polynomials
are coboundaries, hence multiples of $p_0$ — the only coboundary compatible with $\hat C_T=0$,
as $Y^n$ spans the $T$-invariants of $V_n$~\cite{VerifyScripts}. So Theorem~\ref{thm:period}
determines nothing about $\tilde r_f$ beyond its membership in $W$, and pinning it down requires
restricting Definition~\ref{def:reffam}.

Two restrictions suggest themselves, and they disagree. One may fix the reference forms once and for
all, taking $\Psi^{(\alpha)}_p$ to be the Poincar\'e series of~\cite[Def.\ 3.1]{BrownFonseca2025},
which are canonically defined; no freedom then remains, at the price of resting on a normalisation
that~\cite[Rem.\ 5.17]{BrownFonseca2025} notes is not rational. Alternatively one may define
$\tilde r_f$ by continuation in the pole location. For $f$ with no pole in $\mathcal{T}$,
Lemma~\ref{lem:rfWmero} gives $r_f\in W$ with no reference form at all; moving a pole into
$\mathcal{T}$ crosses a side of $\partial\mathcal{T}$, which is a $\Gamma$-translate of
$\hat\gamma^T_f$, so $r_f$ jumps by the crossing of Lemma~\ref{lem:wall}, and continuing through
that jump determines $\tilde r_f$ up to which side was crossed. The residual freedom is then
generated by the windings about the points $Sp$, one per enclosed orbit, and is a proper subspace of
$W$: winding about any other point leaves $W$ altogether and so is not an admissible
correction~\cite{VerifyScripts}.

The two can be compared exactly. Take $f$ with a single orbit of simple poles, none in
$\mathcal{T}$, so that the continuation prescription reduces to $r_f-c_fr_{E_k}$ with no reference
form. Since $\Psi^{(1)}_p$ also vanishes at the cusp, its strip constant is $-2\pi i$ times its
residue, and the residue-matching coefficient $c$ therefore satisfies $c\,c_\Psi=c_f$; subtracting,
\begin{equation}\label{eq:prescdiff}
\big(r_f-c_f\,r_{E_k}\big)-\big(r_f-c\,r_{\Psi^{(1)}_p}\big)
=c\,\big(r_{\Psi^{(1)}_p}-c_\Psi\,r_{E_k}\big) .
\end{equation}
The right side is $c$ times the corrected period polynomial of the reference form itself. The two
prescriptions agree precisely when that vanishes, and since $r(S^!_k)=W$ a family with
$r_{\Psi}-c_\Psi r_{E_k}=0$ can always be arranged by adding a suitable element of $S^!_k$;
for $\Psi^{(1)}_p=E_k/(j-j(p))$ it does not vanish~\cite{VerifyScripts}. Which normalisation is the
right one we leave open, noting only that the continuation prescription agrees by construction with
the unambiguous value on the complement of $\mathcal{T}$, where no reference form is called for.

\begin{remark}\label{rem:literature}
\emph{The polar data used above is not new. Brown and Fonseca~\cite{BrownFonseca2025} compute the residue
of $f(\tau)(X-\tau Y)^{n}\,d\tau$ at a pole $w$, in their Lemma 5.10, as
$\sum_{j}\binom{n}{j}c_{-j-1}(w)(X-wY)^{n-j}(-Y)^{j}$ with $c_{-j-1}(w)$ the Laurent coefficients
of $f$ there; under $j=\alpha-1$ this is Definition~\ref{def:Qf} term by term. The reference forms
$\Psi^{(\alpha)}_p$ of~\eqref{eq:psiform} are theirs, as are the residue evaluation used in the
proof of Theorem~\ref{thm:period}, the local character sums fixing $A_p$, and the identification of
$E_k/(j-j(p))$ in the simple-pole case. What
is added here is the $L$-function: $r_f$ and the $r_{\Psi^{(\alpha)}_p}$ are values of the
finite-contour $L^*$, and the region $\mathcal{T}$, which decides by
Lemma~\ref{lem:rfWmero} whether any correction is needed at all, presupposes a contour and has no
counterpart in their setting.}

\emph{The dependence of $\tilde r_f$, and hence of $\hat r_f$ in~\eqref{eq:rhat}, on the choice of
reference family is~\eqref{eq:ambig}. For adjacent work see~\cite{BrownHain2018}
for poles at the cusp, \cite{Matthes2022} for meromorphic forms with restricted poles, and
\cite{BKvP2019} for regularised inner products. The forms $f_{k,P}$
of~\cite{LobrichSchwagenscheidt2022}, going back to~\cite{Zagier1977}, have weight $2k$ and poles
of order $k$ at CM points, and supply test cases for~\eqref{eq:period} of arbitrary order.}
\end{remark}

For the reconstruction of period polynomials from special values of $L^*$, which
Section~\ref{sec:Lfun} carried out for $f\in S^!_k$, we needed a region free of poles. As it
happens, and as we show below, when the L-functions defined in this
section are used to build period polynomials that are naturally elements of $W$, choosing the base
point $\tau_0=\rho+1=\tfrac{1}{2}+\tfrac{\sqrt3}{2}\,i$ and taking the base contour of the
reference homotopy class to lie on the arc $|\tau|=1$ running from $\rho$ to $\rho+1$, along the
lower boundary of the fundamental domain and its $S$-image, makes that pole-free region of
codimension zero in the upper half-plane. Motivated by this, we derive here explicit expressions
for the L-function along this reference contour, defined so as to allow similarly explicit
representations in terms of the $q$-series coefficients of generic $f\in F_k$.

For $f\in F_k$ with poles in $\HH$, the two segment integrals of Definition~\ref{def:Lint} depend
on their contours; we show their sum depends only on the homotopy class of the pair.

\begin{proposition}\label{prop:indepHomotopy_arc}
For $f\in F_k$ and $\tau_0\in\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$, the L-integral
$L^*(f,s;\gamma^S_{\tau_0},\gamma^T_{\tau_0})$ depends only on the homotopy class of its
two segments in $\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$, and not on the basepoint
$\tau_0$. At fixed $\tau_0$ each of $L^*_S,L^*_T$ is separately path-independent.
\end{proposition}

\begin{proof}
At fixed $\tau_0$ the integrands $f(\tau)\tau^{s-1}$ and $f(\tau)\tk(\tau,s)$ are
holomorphic on $\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$, so Cauchy's theorem makes each
of $L^*_S,L^*_T$ constant on the homotopy class of its own segment. Under a motion of the
basepoint, $\partial_{\tau_0}L^*$ is evaluated at the moving endpoints alone; the
computation of Proposition~\ref{prop:indep}, which uses only $f(\tau_0-1)=f(\tau_0)$ and
$f(-1/\tau_0)=\tau_0^{k}f(\tau_0)$, cancels the boundary terms of $L^*_S$ against those
of $L^*_T$, the interior poles lying off the contours. Hence $\partial_{\tau_0}L^*=0$ along
any motion in $\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$, while each summand may
separately vary.
\end{proof}

\begin{definition}\label{def:class_arc}
For $f\in F_k$ and $\tau_0\in\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$, let
$[\gamma^S_{\tau_0}]$, $[\gamma^T_{\tau_0}]$ denote the homotopy classes of the two
segments in $\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$.
\end{definition}

Thus, with $\gamma^S_{\tau_0} \in [\gamma^S_{\tau_0}]$ and
$\gamma^T_{\tau_0} \in [\gamma^T_{\tau_0}]$, by Proposition~\ref{prop:indepHomotopy_arc}
the value
\begin{equation}\label{eq:Lclass_arc}
L^*\big(f,s;[\gamma^S_{\tau_0}],[\gamma^T_{\tau_0}]\big)
:= L^*\big(f,s;\gamma^S_{\tau_0} , \gamma^T_{\tau_0}\big)
= L^*_S(f,s;\gamma^S_{\tau_0}) + L^*_T(f,s;\gamma^T_{\tau_0})
\end{equation}
is independent of the chosen representative paths and independent of $\tau_0$ (subject to
natural constraints of $\tau_0$ monodromies around singular points, which move between
homotopy classes).

We would like to speak of ``the'' L-integral, and homotopy classes split this into L-integrals for
contours in different classes. We therefore fix a reference class, and we fix it on the boundary of
the fundamental domain.

Write $\gamma^{\rm arc}$ for the geodesic segment
$\{e^{i\theta}:\tfrac{\pi}{3}\le\theta\le\tfrac{2\pi}{3}\}$, oriented from $\rho$ to $\rho+1$.
Taking $\tau_0=\rho+1$ and both segments equal to $\gamma^{\rm arc}$ is admissible by
Definition~\ref{def:segments}, since $-1/(\rho+1)=\rho$ and $(\rho+1)-1=\rho$: the two segments
share their endpoints and may be taken to be the same path. The arc passes through the three
elliptic points $\rho$, $i$ and $\rho+1$, and a form with a pole on $\SLZ\!\cdot\!\rho$ therefore
has one at each endpoint, which the hypothesis $\tau_0\notin\Gamma\!\cdot\!f^{-1}(\infty)$ of
Proposition~\ref{prop:indepHomotopy_arc} forbids. The remedy is to move the base point along the arc
rather than off it: set
\begin{equation}\label{eq:geodelta}
\tau_0(\delta):=e^{i(\frac{\pi}{3}+\delta)},\qquad \delta>0,
\end{equation}
so that $S^{-1}\tau_0(\delta)=e^{i(\frac{2\pi}{3}-\delta)}$ and the segments run between these two
points, meeting neither $\rho$ nor $\rho+1$.%
\footnote{\textbf{$\;$Placeholder. Once poles on the arc are handled by the splitting prescription
below, the base point is no longer a pole of the deformed form, and~\eqref{eq:geodelta} may be
unnecessary. It is retained until that is confirmed.}}

Poles on $\gamma^{\rm arc}$ never occur singly. As $S$ acts on the unit circle by
$e^{i\theta}\mapsto e^{i(\pi-\theta)}$, preserving $\tfrac{\pi}{3}\le\theta\le\tfrac{2\pi}{3}$,
while $f^{-1}(\infty)$ is $\Gamma$-stable and $f|_kS=f$, a pole of order $P$ at $e^{i\phi}$ on the
arc is accompanied by one of the same order at $e^{i(\pi-\phi)}$, again on the arc. These $S$-pairs
have two degenerations: the pair collapses to a single point exactly when $\phi=\tfrac{\pi}{2}$,
and it consists of the two endpoints exactly when $\phi\in\{\tfrac{\pi}{3},\tfrac{2\pi}{3}\}$,
since $S(\rho+1)=-1/(\rho+1)=\rho$. All three cases are treated the same way, by moving the poles
off the contour rather than moving the contour.

\begin{definition}[Splitting prescription]\label{def:arcsplit}
Let $f\in F_k$ have poles on $\gamma^{\rm arc}$, and let $\{f_\eta\}$ be a family in $F_k$,
holomorphic in $\eta$ near $0$, with $f_0=f$ and with no pole of $f_\eta$ on
$\SLZ\!\cdot\!\gamma^{\rm arc}$ for $\eta\neq0$. Set
\begin{equation}\label{eq:arcsplit}
L^*(f,s):=\lim_{\eta\to0}\ \eta^{\,\kappa}\,L^*(f_\eta,s),
\end{equation}
where $\kappa\ge0$ is the least exponent for which the limit exists.\footnote{\textbf{$\;$Placeholder,
in two respects. First, $\kappa$ is $0$ away from the elliptic points --- there the deformed pole
leaves the arc on one side, nothing is trapped, and $\int_{-a}^{a}(u-i\delta)^{-m}\,du$ stays
bounded as $\delta\to0$ at every $m$ --- while at an elliptic point of order $n$ the $S$-pair or
its $U$-orbit pinches the contour from opposite sides and $\kappa$ is positive. Second, and more
seriously, this definition does not pin the family $\{f_\eta\}$. Different deformations enclose
different residues and so give different answers, and which residues are enclosed is precisely what
fixing a reference class is for. Characterising that dependence is left open here.}}
\end{definition}

\begin{definition}[Reference homotopy class on the arc]\label{def:georef}
Let $f\in F_k$. If $f$ has no pole on $\SLZ\!\cdot\!\gamma^{\rm arc}$, the \emph{geodesic reference
class} $[\hat\gamma_f]$ is the class of $\hat\gamma^S_f=\hat\gamma^T_f=\gamma^{\rm arc}$ with
$\tau_0=\rho+1$. Otherwise $L^*(f,s)$ is given by Definition~\ref{def:arcsplit}.
\end{definition}

The second clause is not a regularisation. For every such $\delta$ the hypothesis of
Proposition~\ref{prop:indepHomotopy_arc} holds, so $L^*$ does not depend on $\delta$ at all: the
shortening of the segments is compensated by the movement of the base point, and no limit is
taken. The displacement stays on $\gamma^{\rm arc}$, which is stable under $S$.

In the third clause the oddness of $\log r$ is what the class requires, not a convenience: $S$
acts on the unit circle by $(r,\theta)\mapsto(1/r,\pi-\theta)$, so the deformed path retraces
itself under $S$ exactly when $\log r$ is odd about $\theta=\tfrac{\pi}{2}$. Oddness is thus
$S$-equivariance, and it is what keeps the closed path traced by $S^2$ degenerate. It also fixes
the endpoints correctly: it forces $\log r(\tfrac{2\pi}{3})=-\log r(\tfrac{\pi}{3})$, so writing
$r(\tfrac{\pi}{3})=c$ the deformed segment runs from $c^{-1}e^{2\pi i/3}$ to $ce^{i\pi/3}$, and
$S^{-1}(ce^{i\pi/3})=c^{-1}e^{2\pi i/3}$; a deformation lifting the endpoints off the arc simply
carries the base point with them, and Definition~\ref{def:segments} is still satisfied.

Definition~\ref{def:georef} stops where a choice of contour suffices, and at an elliptic point no
choice does. The reason is geometric rather than technical. At an elliptic point $\tau_e$ of order
$n$ the function $j-j(\tau_e)$ vanishes to order $n$, so a pole of $f$ at $\tau_e$ is the merger of
a full stabiliser orbit of $n$ poles: nearby, $f$ has simple poles at $n$ distinct points permuted
by the stabiliser, and as they coalesce they pinch the contour from every side at once. Indenting
past one member of the orbit passes around the others, so no deformation of the contour separates
them.

The remedy is to move the pole rather than the contour. Lift the orbit off the arc by deforming
$f$, evaluate there --- where the first clause of Definition~\ref{def:georef} applies and no choice
remains --- and continue the result back to $\tau_e$. This is how every elliptic pole will be
handled.\footnote{\textbf{What this principle does and does not supply. It does not supply
existence of the limit: that must be established case by case, and in general the limit exists only
after rescaling by a power of the deformation parameter. What it supplies is that the answer cannot
land outside $W$. Every deformed form falls under the first clause of
Definition~\ref{def:georef}, so its period polynomial lies in $W$; $W$ is a closed subspace; hence
any limit that exists lies in $W$. The rescaling is by a power of the deformation parameter rather
than of a contour parameter, so it is not a choice of how to draw a path --- but it remains a
regularisation, extracting the leading coefficient of a divergence, and whether different
deformations of the same $f$ agree is not settled here.}}

With this definition we may unambiguously write
\begin{align}\label{eq:Lstararc}
L^*(f,s) := L^*\big(f, s; [\hat\gamma_f^S], [\hat\gamma_f^T]\big)~.
\end{align}
Having fixed this reference value, consider two paths $\gamma^S_{\tau_0}$ and
$\gamma^T_{\tau_0}$ which may lie in any homotopy class, and let $X_S(p),X_T(p)\in\ZZ$ count the
windings of $\gamma^S_{\tau_0}$, $\gamma^T_{\tau_0}$ about each pole $\tau_p\in f^{-1}(\infty)$
relative to the reference class.

\begin{lemma}\label{lem:wall_arc}
Let $f\in F_k$, $r_S(f, \tau_p) := \Res_{\tau=\tau_p} (f(\tau) \tau^{s-1})$
and $r_T(f, \tau_p) := \Res_{\tau=\tau_p} (f(\tau) \tk(\tau, s))$. Then,
\begin{equation}\label{eq:wall_arc}
L^*\big(f,s;[\gamma^S_{\tau_0}],[\gamma^T_{\tau_0}]\big)
= L^*(f,s)
+ 2\pi i\,e^{-\pi i s/2}\sum_{\tau_p\in f^{-1}(\infty)}\Big[
X_S(\tau_p) r_S(f, \tau_p)
+ X_T(\tau_p) r_T(f, \tau_p) \Big].
\end{equation}
\end{lemma}

\begin{proof}
Two pairs whose winding data differ by one unit at a single pole $p$, on one segment, bound a loop
encircling $p$ once; the residue theorem adds $2\pi i$ times the residue of that segment's
integrand, $f(\tau)\tau^{s-1}$ for $\gamma^S$ or $f(\tau)\tk(\tau,s)$ for $\gamma^T$, and the
prefactor $e^{-\pi i s/2}$ of Definition~\ref{def:Lint} multiplies through. Summing over crossings
gives~\eqref{eq:wall_arc}.
\end{proof}

In particular the two choices of side available in the third clause of
Definition~\ref{def:georef} differ by the residues of the poles between them.

This provides the basics for the finite L-functions of $f\in F_k$ in this class. We wish, however,
to present them in terms of their $q$-series coefficients, as was done for $f\in M^!_k$ in
Section~\ref{sec:Lfun}. The obstruction to doing so directly is not a technical one. For
$f\in M^!_k$ the series $\sum_n c_f(n)q^n$ converges on all of $\HH$, as $f$ is holomorphic there,
and term-by-term evaluation is legitimate throughout. A pole at finite height ends this. The
expansion about the cusp then represents $f$ only above the highest pole, on
${\rm Im}\,\tau>y_f:=\sup\{{\rm Im}\,\tau_p\}$, with coefficients growing as
$|c_f(n)|\gtrsim n^{m-1}e^{2\pi n\,{\rm Im}\,\tau_p}$ for a pole of order $m$~\cite{1806}. The arc
has height varying between $\tfrac{\sqrt3}{2}$ and $1$, so the operative threshold is its lowest
point $\tfrac{\sqrt3}{2}$: as soon as $f$ has a pole at or above that height the $q$-series
diverges somewhere on the contour, and there is then no series to integrate term by term, rather
than a slowly convergent one to be handled with care. To do this, we must introduce the
polylogarithmic projections of Ref.~\cite{1806}.

\subsection{Explicit formulae for $L^*(f,s)$ on the arc}\label{sec:arcL}

We lay out the starting definitions and facts.

\begin{definition}\label{def:Finfty_arc}
$F(\infty)$ is the space of meromorphic functions $g:\HH\to\CC$ with $g(\tau+1)=g(\tau)$, finitely
many poles in each compact subset of the strip $|{\rm Re}\,\tau|\le\tfrac12$, and bounded pole
order at $q=0$.
\end{definition}

\begin{definition}\label{def:proj_arc}
Let $f\in F_k$ have a pole of order $m_p$ at $\tau_p\in\HH$. On a punctured $\varepsilon$-ball
about $\tau_p$ small enough to exclude the other poles of $f$, we have
\begin{equation}\label{eq:laurent_arc}
f(\tau)=\sum_{m=1}^{m_p}\frac{r^*_{f,\tau_p}(m)}{(\tau-\tau_p)^{m}}+g(\tau),
\qquad
r^*_{f,\tau_p}(m)=\frac{1}{2\pi i}\oint_{|\tau-\tau_p|=\varepsilon}
f(\tau)\,(\tau-\tau_p)^{m-1}\,d\tau ,
\end{equation}
with $g$ holomorphic at $\tau_p$. We call $r^*_{f,\tau_p}(1),\dots,r^*_{f,\tau_p}(m_p)$ the
\emph{pole weights} of $f$ at $\tau_p$; the first is the residue, and the last is nonzero, which is
what fixes the order. The \emph{polylog projection} of $f$ at $\tau_p$ is
\begin{equation}\label{eq:polyproj_arc}
\widehat P_{\tau_p}f(\tau):=\sum_{m=1}^{m_p} r^*_{f,\tau_p}(m)\,
\frac{(-2\pi i)^m}{(m-1)!}\,\Li_{1-m}\big(e^{2\pi i(\tau-\tau_p)}\big),
\qquad \Li_s(x):=\sum_{n\ge1}\frac{x^{n}}{n^{s}}.
\end{equation}
\end{definition}

The composite projector is the one place where the contour enters. Its threshold is set so that
$(1-\widehat P)f$ admits a $q$-series everywhere it is used, which is a statement about a
half-plane: a $q$-series converges exactly on ${\rm Im}\,\tau>h$ with $h$ the height of the highest
surviving pole. The operative height is the lowest point of $\gamma^{\rm arc}$, namely
$\tfrac{\sqrt3}{2}$, since the mode sums below are controlled by $e^{-2\pi n\,{\rm Im}\,\tau}$ and
the arc descends to that height at its endpoints.

\begin{definition}\label{def:geoproj}
For $f\in F_k$ with $T$-inequivalent finite-order poles $\{\tau_p\}$ of orders $\{m_p\}$, set
\begin{equation}\label{eq:polesplit_arc}
\widehat P^{\,\rm geo}:=\sum_{{\rm Im}\,\tau_p\ge\frac{\sqrt3}{2}}\widehat P_{\tau_p},
\end{equation}
with $\widehat P_{\tau_p}$ as in Definition~\ref{def:proj_arc}, and write
$f=\widehat P^{\,\rm geo}f+(I-\widehat P^{\,\rm geo})f$.
\end{definition}

\begin{lemma}\label{lem:proj_arc}
The normalized block $\frac{(-2\pi i)^m}{(m-1)!}\Li_{1-m}\big(e^{2\pi i(\tau-\tau_p)}\big)$ has
principal part exactly $(\tau-\tau_p)^{-m}$, so $\widehat P_{\tau_p}f$ matches the principal part
of $f$ and $(I-\widehat P_{\tau_p})f$ is holomorphic at $\tau_p$.
\end{lemma}
\begin{proof}
By the small-$t$ expansion $\Li_{1-m}(e^{-t})=(m-1)!\,t^{-m}+O(t^0)$ with
$t=-2\pi i(\tau-\tau_p)$, the block has a \emph{pure} pole
$(\tau-\tau_p)^{-m}+O((\tau-\tau_p)^0)$ at $\tau_p$, with no lower-order poles. Summing against the
Laurent coefficients $r^*_{f,\tau_p}(m)$ reproduces the principal part of $f$ term by term, leaving
$(I-\widehat P_{\tau_p})f$ regular at $\tau_p$.
\end{proof}

\begin{definition}\label{def:cRf_arc}
For $f\in F_k$ and $\widehat P^{\,\rm geo}$ as in Definition~\ref{def:geoproj}, write
\begin{equation}
(I-\widehat P^{\,\rm geo})f(\tau)=\sum_n c_{Rf}(n)\,q^n,\qquad q=e^{2\pi i\tau}.
\end{equation}
\end{definition}

\begin{lemma}\label{lem:projspace_arc}
$F_k\subset F(\infty)$ and $(I-\widehat P^{\,\rm geo})f\in F(\infty)$. Every pole of
$(I-\widehat P^{\,\rm geo})f$ has ${\rm Im}\,\tau<\tfrac{\sqrt3}{2}$, with a maximal height
$y_b<\tfrac{\sqrt3}{2}$ (set $y_b=0$ if $(I-\widehat P^{\,\rm geo})f$ is holomorphic).
\end{lemma}
\begin{proof}
A weakly holomorphic form is $T$-periodic, meromorphic, and $O(e^{C{\rm Im}\tau})$ at the cusp, so
$F_k\subset F(\infty)$; the projected polylogs $\Li_{1-m}(e^{2\pi i(\tau-\tau_p)})$ share these
properties, so $(I-\widehat P^{\,\rm geo})f\in F(\infty)$ (and leaves $F_k$, being non-modular). By
Lemma~\ref{lem:proj_arc}, $\widehat P^{\,\rm geo}$ cancels the principal part of every pole with
${\rm Im}\,\tau\ge\tfrac{\sqrt3}{2}$, so the poles of $(I-\widehat P^{\,\rm geo})f$ lie strictly
below $\tfrac{\sqrt3}{2}$; finite in each compact set and accumulating only at ${\rm Im}\,\tau=0$,
they have a maximal height $y_b<\tfrac{\sqrt3}{2}$.
\end{proof}

The Fourier coefficients of a $T$-periodic meromorphic function are controlled by its highest pole:
$|c_{Rf}(n)|\le e^{2\pi n(y_b+\varepsilon)}$ for every $\varepsilon>0$ and all large
$n$~\cite{1806}. Since $y_b<\tfrac{\sqrt3}{2}$, the series $\sum_n c_{Rf}(n)q^n$ converges
geometrically down to ${\rm Im}\,\tau=\tfrac{\sqrt3}{2}$, at base
$e^{-2\pi(\frac{\sqrt3}{2}-y_b-\varepsilon)}$, and so on $\gamma^{\rm arc}$ itself. This is the
point of the projection: convergence merely for large ${\rm Im}\,\tau$ would be automatic and of no
use, as the contour descends to height $\tfrac{\sqrt3}{2}$.

One further integral of $f$ along the contour will be needed, both for the residue of $L^*$ at
$s=0$ and, later, for period polynomials.

\begin{definition}\label{def:Phi}
For $f\in F_k$, set $\Phi(f):=\int_{\hat\gamma^T_f}f(\tau)\,d\tau$, the integral of $f$ along the
$T$-segment of the contour defining $L^*(f,s)$.
\end{definition}

$\Phi$ is not an independent choice: it is what the parabolicity condition $\hat C_T=0$ measures,
and $\hat C_T$ is assembled from the $T$-transport, so $\Phi$ follows the contour. Deforming the
segments replaces $\gamma^{\rm arc}$ in the integral by whatever $\hat\gamma^T_f$ then is, with no
further stipulation.

Since $\hat\gamma^T_f$ runs from $T^{-1}\tau_0$ to $\tau_0$, and $e^{2\pi in\tau}$ takes equal
values at those two endpoints, the modes $n\neq0$ of the $q$-series of $f$ cancel between them.
Hence $\Phi(f)=c_f(0)$ whenever that series converges along the segment, which is to say whenever
no pole of $f$ lies above $\hat\gamma^T_f$; a pole below is harmless. Otherwise $\Phi(f)$ is
$c_f(0)$ less $2\pi i$ times the residues of the poles in the strip between the segment and the
cusp. In particular $\Phi(E_k)=1$, and $\Phi(f)=0$ for $f\in S_k$.

The relations satisfied by the pole weights at an elliptic point are contour-free, so they hold
here verbatim; we record them now, as both the block integral below and the period polynomials
constructed later use them.

\begin{lemma}\label{lem:ellLaurent}
Let $\tau_e$ be $i$ or $\rho$, of order $n=2$ or $n=3$, let
$\gamma=\left(\begin{smallmatrix}a&b\\c&d\end{smallmatrix}\right)$ be the generator $S$ or $ST$ of
its stabiliser in $\SLZ$, and set $\zeta:=c\tau_e+d$, so that $c=1$ and $\zeta=e^{i\pi/n}$. Let
$f\in F_k$ have a pole of order $P$ at $\tau_e$ with pole weights $r^*_{f,\tau_e}(m)$. Modularity is
equivalent to one relation for each principal-part order $\mu$,
\begin{equation}\label{eq:ellLaurent}
\sum_{m=\mu}^{P} r^*_{f,\tau_e}(m)\binom{m}{\mu}\,\zeta^{m}
=\zeta^{k}\sum_{m=\mu}^{P} r^*_{f,\tau_e}(m)\binom{k}{m-\mu}\,\zeta^{-m},
\qquad \forall\mu=1,\dots,P,
\end{equation}
and the leading relation is $\big(\zeta^{2P}-\zeta^{k}\big)r^*_{f,\tau_e}(P)=0$, so that
$2P\equiv k \pmod{2n}$.
\end{lemma}
\begin{proof}
Write $\tau=\tau_e+\varepsilon$. Since $\det\gamma=1$ and $\gamma\tau_e=\tau_e$,
\begin{equation}\label{eq:ellshift}
\gamma\tau-\tau_e=\frac{\tau-\tau_e}{(c\tau+d)(c\tau_e+d)}=\frac{\varepsilon}{\zeta(\zeta+\varepsilon)},
\qquad (c\tau+d)^{k}=\zeta^{k}\big(1+\varepsilon/\zeta\big)^{k},
\end{equation}
using $c=1$. Hence $(\gamma\tau-\tau_e)^{-m}=\zeta^{2m}\varepsilon^{-m}(1+\varepsilon/\zeta)^{m}$,
and matching the coefficient of $\varepsilon^{-\mu}$ in $f(\gamma\tau)=(c\tau+d)^{k}f(\tau)$
through $(1+\varepsilon/\zeta)^{j}=\sum_{l}\binom{j}{l}\zeta^{-l}\varepsilon^{l}$ gives
$\sum_{m\ge\mu}r^*(m)\zeta^{2m}\binom{m}{\mu}\zeta^{-(m-\mu)}
=\zeta^{k}\sum_{m\ge\mu}r^*(m)\binom{k}{m-\mu}\zeta^{-(m-\mu)}$; dividing by $\zeta^{\mu}$
yields~\eqref{eq:ellLaurent}. At $\mu=P$ only $m=P$ survives on each side.
\end{proof}

At $n=2$ the order condition forces $P$ odd when $k\equiv2\pmod4$ and $P$ even when $4\mid k$; at
$n=3$ it reads $2P\equiv k\pmod 6$.

Each pole with ${\rm Im}\,\tau_p\ge\tfrac{\sqrt3}{2}$ contributes one projected polylog to
$L^*(f,s)$, whose L-integral we now record.

\begin{definition}\label{def:geoIN}
For a pole of order $N+1$ projected at $z$, let $\lambda_{N,z}$ denote the corresponding block
of~\eqref{eq:polyproj_arc}, and set
$I^{\rm geo}_N(s,z):=L^*\big(\lambda_{N,z},\,s;\,[\hat\gamma^S_f],[\hat\gamma^T_f]\big)$,
evaluated in the geodesic reference class of Definition~\ref{def:georef}.
\end{definition}

The blocks need not be integrated on the arc. Both kernels $\tau^{s-1}$ and $\tk(\tau,s)$ are
holomorphic on all of $\HH$, so the only obstruction to deforming is the block's own poles at
$z+\ZZ$, and one may evaluate on the horizontal chord from $\rho$ to $\rho+1$ instead. Let
$\Lambda$ be the open lens between the two, bounded above by $\gamma^{\rm arc}$ and below by the
chord, so that
$\Lambda\subset\{|\tau|<1,\ {\rm Im}\,\tau>\tfrac{\sqrt3}{2},\ |{\rm Re}\,\tau|<\tfrac12\}$. At
most one member of $z+\ZZ$ lies in $\Lambda$, and the deformation costs one residue exactly when
one does.

\begin{lemma}\label{lem:geopolylog}
Let ${\rm Im}\,z>\tfrac{\sqrt3}{2}$ with $z\notin\gamma^{\rm arc}$, set $\tau_0=\rho+1$, let
$\gamma^*(s,x):=\sum_{j\ge0}\frac{(-x)^{j}}{j!\,(j+s)}$, entire in $x$, read empty products below
as $1$, and set $(-2\pi n)^{w}:=e^{w(\log 2\pi n-i\pi)}$. With
\begin{equation}\label{eq:geoG}
G^{\rm geo}_{s,k}(-n):=e^{-\pi is/2}\Big[\tau_0^{\,s}\,\gamma^*\big(s,2\pi in\tau_0\big)
-\rho^{\,s}\,\gamma^*\big(s,2\pi in\rho\big)\Big]
\;+\;\frac{\Gamma\big(s,2\pi in\tau_0\big)}{(-2\pi n)^{s}}
\;+\;i^{k}\frac{\Gamma\big(k-s,2\pi in\tau_0\big)}{(-2\pi n)^{k-s}},
\end{equation}
\begin{equation}\label{eq:geoG0}
G^{\rm geo}_{s,k}(0):=e^{-\pi is/2}\,\frac{\tau_0^{\,s}-\rho^{\,s}}{s}
-\frac{(\tau_0/i)^{s}}{s}-i^{k}\frac{(\tau_0/i)^{k-s}}{k-s},
\end{equation}
\begin{equation}\label{eq:geocross}
X_N(s,z):=\frac{e^{-\pi is/2}}{(2\pi i)^{N}}\bigg[(-1)^{N}z^{\,s-1-N}\prod_{j=0}^{N-1}(s-1-j)
+\zeta\big(1-s+N,\,z+1\big)\prod_{j=0}^{N-1}(1-s+j)
-e^{i\pi(s-1)}\zeta\big(1-k+s+N,\,z+1\big)\prod_{j=0}^{N-1}(1-k+s+j)\bigg],
\end{equation}
we have
\begin{equation}\label{eq:geoIN}
I^{\rm geo}_N(s,z)=-\delta_{N,0}\,G^{\rm geo}_{s,k}(0)
+(-1)^{N+1}\sum_{n\ge1}n^{N}e^{2\pi inz}\,G^{\rm geo}_{s,k}(-n)
+\begin{cases}X_N(s,z),&z\in\Lambda,\\ 0,&\text{otherwise},\end{cases}
\end{equation}
the sum converging at rate $e^{-2\pi({\rm Im}\,z-\frac{\sqrt3}{2})}$.
\end{lemma}

\begin{proof}
Both kernels are holomorphic on $\HH$, and the only poles of $\lambda_{N,z}$ are at $z+\ZZ$, so
Cauchy's theorem moves both segment integrals to the chord at the cost of whatever residues lie
between the two paths; by the paragraph above that is $\{z\}$ when $z\in\Lambda$ and nothing
otherwise. The chord is not a
convenience: along $\gamma^{\rm arc}$ itself ${\rm Im}\,\tau$ ranges over
$[\tfrac{\sqrt3}{2},1]$, so for $\tfrac{\sqrt3}{2}<{\rm Im}\,z<1$ the comparison below would change
sign partway along, whereas on the chord ${\rm Im}\,\tau\equiv\tfrac{\sqrt3}{2}$ and it is a single
global inequality.

On the chord $|e^{2\pi i(\tau-z)}|>1$, so the defining series of $\Li_{-N}$ diverges and inversion
$\Li_{-N}(w)=(-1)^{N+1}\Li_{-N}(1/w)-\delta_{N,0}$ rewrites $\lambda_{N,z}$ as
$(-1)^{N+1}\sum_{n\ge1}n^{N}e^{2\pi inz}q^{-n}-\delta_{N,0}$, uniformly convergent there since
${\rm Im}\,z>\tfrac{\sqrt3}{2}$. Integrating term by term, $G^{\rm geo}_{s,k}(-n)=L^*(q^{-n},s)$ in
this class: the last two terms of~\eqref{eq:geoG} are $L^*_T$ by Lemma~\ref{lem:Tclosed-Mk} at
$\tau_0$, and the first is $L^*_S=e^{-\pi is/2}\int_{\rho}^{\tau_0}e^{-2\pi in\tau}\tau^{s-1}d\tau$,
which on expanding $e^{-2\pi in\tau}$ and integrating $\tau^{j+s-1}$ term by term gives
$e^{-\pi is/2}\sum_{j\ge0}\frac{(-2\pi in)^{j}}{j!\,(j+s)}\big[\tau_0^{\,j+s}-\rho^{\,j+s}\big]$,
the first bracket of~\eqref{eq:geoG}. Likewise $G^{\rm geo}_{s,k}(0)=L^*(1,s)$. The rate follows
from $|e^{2\pi inz}|=e^{-2\pi n\,{\rm Im}\,z}$ against
$|G^{\rm geo}_{s,k}(-n)|\sim e^{\pi\sqrt3\,n}$.

For the crossing term put $u=\tau-z$ and $v=2\pi iu+t$. Then
$\sum_{N\ge0}\Li_{-N}\big(e^{2\pi iu}\big)\tfrac{t^{N}}{N!}=\tfrac{e^{v}}{1-e^{v}}
=-\sum_{m\ge0}\tfrac{B_m(-v)^{m}}{m!\,v}$, of which only the $m=0$ term $-1/v$ carries negative
powers of $u$; expanding $-1/(2\pi iu+t)=-\sum_{r\ge0}(-1)^{r}t^{r}(2\pi iu)^{-r-1}$ and reading
off $t^{N}/N!$ gives
\begin{equation}\label{eq:blockpp}
\lambda_{N,z}(\tau)=\frac{(-1)^{N+1}N!}{\big(2\pi i(\tau-z)\big)^{N+1}}+O(1),
\end{equation}
a lone singular term, so $\Res_{\tau=z}\big(\lambda_{N,z}\kappa\big)
=(-1)^{N+1}\kappa^{(N)}(z)/(2\pi i)^{N+1}$ for any $\kappa$ holomorphic at $z$. Both paths run
$\rho\to\rho+1$ with the arc above, so together they bound $\Lambda$ clockwise and the deformation
contributes $-2\pi i$ times that residue for each kernel. With
$\partial_z^{N}z^{s-1}=z^{s-1-N}\prod_{j=0}^{N-1}(s-1-j)$ and
$\partial_z^{N}\zeta(w,z+1)=(-1)^{N}\zeta(w+N,z+1)\prod_{j=0}^{N-1}(w+j)$ applied to both terms of
$\tk$, this is $X_N(s,z)$ of~\eqref{eq:geocross}.
\end{proof}

Corollary~\ref{cor:Sclosed} is deliberately not used here: it folds the $S$-segment about $i$ using
modularity, and $q^{-n}$ is only $T$-periodic, so its $S$-integral must be taken directly. This is
the one place where the two classes genuinely differ. At $\tau_0=i$ the $S$-segment is degenerate
and $L^*_S$ drops out of Lemma~\ref{lem:polylogcf} entirely; here it contributes the first term
of~\eqref{eq:geoG}, and the threshold in the convergence rate stands at $\tfrac{\sqrt3}{2}$, in
step with Definition~\ref{def:geoproj}.

The $S$-term is written with $\gamma^*$ because that form is branch-free: $\gamma^*$ is
single-valued, so the only multivalued factors are the explicit $\tau_0^{\,s}$ and $\rho^{\,s}$,
and those are unambiguous since $\arg\tau\in[\tfrac{\pi}{3},\tfrac{2\pi}{3}]$ along the chord. It
may equally be written with upper incomplete gammas, provided the sheet relation is carried. Under
$u=2\pi in\tau$ the chord's image runs from $\arg u=\tfrac{7\pi}{6}$ at $\rho$ to
$\tfrac{5\pi}{6}$ at $\tau_0$, so the $\rho$ endpoint sits one turn off principal; since
$\gamma(s,x)=x^{s}\gamma^*(s,x)$ with $\gamma^*$ single-valued, a shift of $\arg x$ by $2\pi$ gives
\begin{equation}\label{eq:sheet}
\Gamma^{\rm cont}(s,x)=e^{2\pi is}\,\Gamma(s,x)+\big(1-e^{2\pi is}\big)\Gamma(s),
\end{equation}
and hence
\begin{equation}\label{eq:geoSupper}
S_n(s)=(2\pi in)^{-s}\Big[e^{2\pi is}\,\Gamma\big(s,2\pi in\rho\big)
+\big(1-e^{2\pi is}\big)\Gamma(s)-\Gamma\big(s,2\pi in\tau_0\big)\Big].
\end{equation}
Both the sheet factor and the additive constant are trivial at integer $s$, which is all that
Definition~\ref{def:rf} requires. The $T$-term needs neither, since by
Lemma~\ref{lem:Tclosed-Mk} only $\tau_0$ appears there, at $\arg(2\pi in\tau_0)=\tfrac{5\pi}{6}$,
the $\rho$ endpoint having been absorbed by the periodicity of $\tk$. It does need the branch of
$(-2\pi n)^{w}$ fixed above; the opposite choice differs by $e^{2\pi is}$, and so is invisible at
integer $s$ and wrong elsewhere. In practice the $\gamma^*$
series loses about $2.7n$ digits to cancellation, so~\eqref{eq:geoSupper} is preferable if large
$n$ is needed.

Two hypotheses of Lemma~\ref{lem:geopolylog} limit its reach, and neither is
cosmetic.\footnote{\textbf{$\;$It assumes ${\rm Im}\,z>\tfrac{\sqrt3}{2}$ strictly, so it says
nothing on the threshold line itself; and it assumes $|z|\ge1$, so it says nothing about poles
lying below the arc but above $\tfrac{\sqrt3}{2}$. Completing the analogue of
Section~\ref{sec:mero} in this class further requires the counterpart of
Lemma~\ref{lem:polylogcf} on the chord, and those of Lemma~\ref{lem:polylogedge} and
Definition~\ref{def:FP} at height $\tfrac{\sqrt3}{2}$; the assembly corresponding to
Theorem~\ref{thm:mero} then follows from these. None of that is carried out here, so this
subsection supplies the ingredients rather than the closed formula. The chord case should be the
easier one, since a horizontal contour makes the inversion criterion a single global height
comparison rather than a varying one.}}

We pause to connect the L-functions of this section to the definition in Ref.~\cite{1806}.

\begin{remark}\label{rem:arcresidue}
\emph{The $1/s$ residue is carried entirely by the $T$-segment. $L^*_S$ is a finite integral of
$f\,\tau^{s-1}$ and is entire in $s$, while $\tk$ has a simple pole at $s=0$ whose coefficient is
the integral of $f$ along that segment, the constant Fourier mode of $f$ in the strip containing
it. In the geodesic class that integral is $\Phi(f)$ of Definition~\ref{def:Phi}, so
$\mathop{\rm Res}_{s=0}L^*(f,s)=-\Phi(f)$, whereas Ref.~\cite{1806} anchors its contour at the cusp,
above every pole, and carries $-c_f(0)$. The two differ by the residues lying in the strip between
the contours, which is Lemma~\ref{lem:wall_arc}. The quantity that must be subtracted to reach $W$
is therefore the same object as the departure from the residue convention of~\cite{1806}, and not a
second normalisation imposed on top of it.}
\end{remark}

\begin{remark}\label{rem:arcextremal}
\emph{The mechanism is already visible in~\eqref{eq:geoG0}, before any assembly. Its $S$-part
$e^{-\pi is/2}(\tau_0^{\,s}-\rho^{\,s})/s$ is finite at $s=0$, since $\tau_0^{\,s}-\rho^{\,s}$
vanishes like $s\log(\tau_0/\rho)$, while $-(\tau_0/i)^{s}/s$ carries the pole. The two classes sit
at opposite ends of one deformation: the contour of~\cite{1806} lies above every pole, making the
residue bookkeeping trivial and the pole-free region empty, while $\gamma^{\rm arc}$ lies below
every pole orbit not on it, making the pole-free region of codimension zero and the residue
bookkeeping maximal.}
\end{remark}

\begin{definition}\label{def:rfhat}
For $f\in F_k$, let $r_f$ be given by Definition~\ref{def:rf} with $L^*(f,s)$ evaluated in the
geodesic reference class of Definition~\ref{def:georef}, and set
\begin{equation}\label{eq:rfhat}
\hat r_f\;:=\;r_f-\Phi(f)\,r_{E_k}.
\end{equation}
\end{definition}

\begin{lemma}\label{lem:geoloops}
$\SLZ$ is generated by $S$ and $T$ subject to $S^2=(ST)^3=-I$. In the geodesic reference class the
closed path traced by each of these two words retraces itself, so $\hat C_{\pm I}=0$ for every
$f\in F_k$ with no pole on $\SLZ\!\cdot\!\gamma^{\rm arc}$.
\end{lemma}

\begin{proof}
$S$ acts on the unit circle by $e^{i\theta}\mapsto e^{i(\pi-\theta)}$, preserving
$\tfrac{\pi}{3}\le\theta\le\tfrac{2\pi}{3}$, reversing orientation and fixing $i$; so
$S\hat\gamma^S_f$ is $\hat\gamma^S_f$ reversed and the path traced by $S^2$ retraces itself, for any
base point on the arc. For $(ST)^3$, the fixed points of $U=TS$ are the roots of $\tau^2-\tau+1$, so
$U\tau_0=\tau_0$ at $\tau_0=\rho+1$ and the $U$-path is closed. With
$\hat\gamma^S_f=\hat\gamma^T_f$ it is $S^{-1}\hat\gamma^T_f$ followed by $\hat\gamma^S_f$, the arc
traversed and then retraced; its $U$- and $U^2$-transports retrace $U\gamma^{\rm arc}$ and
$U^2\gamma^{\rm arc}$ through the same fixed point. Each bounds nothing, so the winding number about
every pole of $f$ vanishes and Cauchy's theorem gives $\hat C_{-I}=C_{-I}=0$.
\end{proof}

Definition~\ref{def:rfhat} covers those $f$ whose poles avoid the contour, and since
$\gamma^{\rm arc}$ is a curve that is the generic situation: the excluded locus is of codimension
one in $\HH$, so $\hat r_f$ is defined outright on a set of codimension zero. The remaining cases
are reached by continuity from it. Three cases exhaust the possibilities:
\begin{enumerate}
\item $f$ has no pole on $\SLZ\!\cdot\!\gamma^{\rm arc}$;
\item $f$ has a pole on $\gamma^{\rm arc}$, away from $i$ and $\rho$;
\item $f$ has a pole at $i$ or at $\rho$.
\end{enumerate}

\begin{lemma}\label{lem:arcoff}
Let $f\in F_k$ have no pole on $\SLZ\!\cdot\!\gamma^{\rm arc}$. Then $\hat r_f\in W$, with no
condition on the location or the order of the poles of $f$.
\end{lemma}

\begin{proof}
Throughout, $\hat C$ is the cocycle of Lemma~\ref{lem:cocycle}, so that $\hat C_S=r_f$ and
$\hat C_{\gamma\delta}=\hat C_\gamma|_\delta+\hat C_\delta$.

For the first relation, taking $\gamma=\delta=S$ gives
$\hat C_{S^2}=\hat C_S|_S+\hat C_S=r_f|_{(1+S)}$, and $S^2=-I$, so
Lemma~\ref{lem:geoloops} gives
\begin{equation}\label{eq:arcoffS}
r_f\big|_{(1+S)}=\hat C_{-I}=0 .
\end{equation}

For the second, take $\gamma=T$, $\delta=S$, so that $U=TS$ and
$\hat C_U=\hat C_T|_S+\hat C_S$. Applying the cocycle relation twice more,
\begin{equation*}
\hat C_{U^3}=\hat C_U|_{U^2}+\hat C_{U^2}
=\hat C_U|_{U^2}+\hat C_U|_{U}+\hat C_U=\hat C_U\big|_{(1+U+U^2)} ,
\end{equation*}
and $U^3=-I$, so Lemma~\ref{lem:geoloops} makes the left side vanish. Substituting
$\hat C_U=\hat C_T|_S+\hat C_S$ and using $\hat C_S=r_f$,
\begin{equation}\label{eq:arcoffU}
0=\hat C_T\big|_{S(1+U+U^2)}+r_f\big|_{(1+U+U^2)},
\qquad\text{so}\qquad
r_f\big|_{(1+U+U^2)}=-\,\hat C_T\big|_{S(1+U+U^2)} .
\end{equation}
This is where the hypothesis $f\in S^!_k$ was used in Lemma~\ref{lem:rfW}, and it is the only
place: there $\hat C_T$ vanished, here it need not.

The computation of $\hat C_T$ in the proof of Lemma~\ref{lem:cocycle} does not use $c_f(0)=0$ until
its final step. It shows that $A(\tau):=\chi(\tau;X+Y,Y)-\chi(\tau-1;X,Y)$ is a $1$-periodic
polynomial, hence a constant $A_0$ depending on $k$ alone, and that
\begin{equation}\label{eq:arcoffCT}
\hat C_T=-(2\pi i)^{n+1}A_0\int_{\hat\gamma^T_f}f(\tau)\,d\tau=-(2\pi i)^{n+1}A_0\,\Phi(f)
\end{equation}
by Definition~\ref{def:Phi}. Since $A_0$ does not depend on $f$, neither does
$\mathcal D:=(2\pi i)^{n+1}A_0\big|_{S(1+U+U^2)}$, and \eqref{eq:arcoffU} becomes
\begin{equation}\label{eq:arcoffprop}
r_f\big|_{(1+U+U^2)}=\Phi(f)\,\mathcal{D}.
\end{equation}
The defect is therefore proportional to $\Phi(f)$, with a constant of proportionality common to
every $f\in F_k$ whose poles avoid $\SLZ\!\cdot\!\gamma^{\rm arc}$.

It remains to identify $\mathcal D$. Apply \eqref{eq:arcoffprop} to $f=E_k$, which is holomorphic
and so certainly admissible, and for which $\Phi(E_k)=1$: this gives
$\mathcal D=r_{E_k}|_{(1+U+U^2)}$. Hence
\begin{equation*}
\hat r_f\big|_{(1+U+U^2)}
=r_f\big|_{(1+U+U^2)}-\Phi(f)\,r_{E_k}\big|_{(1+U+U^2)}
=\Phi(f)\,\mathcal D-\Phi(f)\,\mathcal D=0 ,
\end{equation*}
while \eqref{eq:arcoffS} applied to both $f$ and $E_k$ gives
$\hat r_f|_{(1+S)}=0-\Phi(f)\cdot 0=0$. Both defining relations of $W$ hold. Nothing in the
argument constrains where the poles of $f$ lie, or how deep they are, beyond their absence
from $\SLZ\!\cdot\!\gamma^{\rm arc}$, which was used only through Lemma~\ref{lem:geoloops}.
\end{proof}

\begin{lemma}\label{lem:arcon}
Let $f\in F_k$ have poles on $\gamma^{\rm arc}$, none at $i$ and none on $\SLZ\!\cdot\!\rho$, and
let $\hat\gamma^S$ be $S$-symmetric. Write $D_E:=r_{E_k}\big|_{(1+U+U^2)}$. Then
$\hat r_f\big|_{(1+S)}=0$ for every admissible $\hat\gamma^T$. If $\hat\gamma^T$ passes on the same
side of every pole as $\hat\gamma^S$ then $\hat r_f\big|_{(1+U+U^2)}=0$; otherwise, with
$p_1,\dots,p_m$ the poles at which the two segments part,
\begin{equation}\label{eq:arcondefect}
\hat r_f\big|_{(1+U+U^2)}
=-(2\pi i)^{n+2}\sum_{j=1}^{m}\nu_j\,a_{p_j}\,(X-p_jY)^{n}
\;-\;2\pi i\Big(\sum_{j=1}^{m}X_T(p_j)\,a_{p_j}\Big)D_E ,
\end{equation}
where $a_p=\Res_{\tau=p}f$ and $\nu_j\in\ZZ\setminus\{0\}$, and this is non-zero unless $D_E$ lies
in the span of $\{(X-p_jY)^{n}\}_{j\le m}$.
\end{lemma}

\begin{proof}
By Lemma~\ref{lem:cocycle}, $\hat C_S=r_f$ with $\hat C_S=C_S-(E|_S-E)$ and
$C_S=(2\pi i)^{n+1}\int_{\hat\gamma^S}\omega_f$; the $T$-segment enters $r_f$ only through the
coboundary $E-E|_S$. The cocycle relation gives $r_f|_{(1+S)}=\hat C_S|_{(1+S)}=\hat C_{S^2}$, and
as $n$ is even the coboundary drops, leaving the integral of $\omega_f$ over the closed path formed
by $\hat\gamma^S$ followed by its $S$-image. In polar coordinates $S$ acts by
$(r,\theta)\mapsto(1/r,\pi-\theta)$, so an $S$-symmetric $\hat\gamma^S$ is carried onto itself with
reversed orientation; that path retraces and the integral vanishes. No property of $\hat\gamma^T$ is
used, and the subtracted term contributes nothing since $r_{E_k}|_{(1+S)}=0$.

For the second relation, $\Phi$ is by construction the quantity $\hat C_T$ measures, so
Definition~\ref{def:rfhat} gives $\hat C_T=0$; then $\hat C_U=\hat C_T|_S+\hat C_S=\hat C_S$ and
$\hat r_f|_{(1+U+U^2)}=\hat C_{U^3}-\Phi(f)D_E$, with $\hat C_{U^3}=\hat C_{-I}$ the integral of
$\omega_f$ over the closed path traced by $(TS)^3$. As $U=TS$, that path is assembled from both
segments. If $\hat\gamma^S=\hat\gamma^T$ it retraces, as in Lemma~\ref{lem:geoloops}, and the two
terms cancel exactly as in Lemma~\ref{lem:arcoff}.

If the segments pass on opposite sides at $p_1,\dots,p_m$, the two paths bound regions containing
exactly those poles and no others, and the residue theorem gives the first term
of~\eqref{eq:arcondefect}; the second is the shift induced on $\Phi$, which is the $T$-segment
integral and therefore moves with $\hat\gamma^T$. For the last claim, the coefficient vector of
$(X-pY)^n$ is $\big(\binom{n}{l}(-p)^{l}\big)_{l}$, so the matrix of $\{(X-p_jY)^n\}_{j\le m}$ is
$\mathrm{diag}\binom{n}{l}$ times a Vandermonde in $(-p_1,\dots,-p_m)$, of rank $m$ whenever the
$p_j$ are distinct and $m\le n+1$. The residue terms therefore cannot cancel among themselves, and
\eqref{eq:arcondefect} can vanish only if $D_E$ lies in their span.
\end{proof}

The deformation of Definition~\ref{def:arcsplit} is not a second prescription but the same one.

\begin{lemma}\label{lem:arcdeform}
Let $f$ be as in Lemma~\ref{lem:arcon} and let $f_\eta$ be as in
Definition~\ref{def:arcsplit}, so that for $\eta\neq0$ the poles of $f_\eta$ leave
$\gamma^{\rm arc}$. Then $\lim_{\eta\to0}\hat r_{f_\eta}$ exists and equals the value of
Lemma~\ref{lem:arcon} on the pair with $\hat\gamma^S=\hat\gamma^T$, the side being fixed by the
direction of the deformation.
\end{lemma}

\begin{proof}
Only the relative position of contour and pole enters, and moving the pole off a fixed contour
produces the same configuration as moving the contour past a fixed pole. Both segments lie on
$\gamma^{\rm arc}$ and the poles move relative to them identically, so the configuration reached is
one with $\hat\gamma^S=\hat\gamma^T$; the sign of $\eta$ decides which. Each segment integral is
holomorphic in $\eta$ off the locus where a pole meets the contour, so the one-sided limits exist.
\end{proof}

The two relations are therefore sensitive to the two segments in quite different ways. The
$S$-relation constrains $\hat\gamma^S$ alone: $S$-symmetry makes the $S^2$-loop degenerate whatever
$\hat\gamma^T$ does. The $(1+U+U^2)$ relation mixes them, since $U=TS$, and vanishes only when the
two segments agree. Requiring $\hat\gamma^S=\hat\gamma^T$ in Definition~\ref{def:georef} is thus not
a stipulation of convenience: membership in $W$ selects it.

\begin{lemma}\label{lem:arcell}
Let $\tau_e$ be an elliptic point of order $n$, let $g\in M^!_k$ have
$\mathrm{ord}_{\tau_e}g=a$ with $0\le a<n$, let $m\ge1$, and set
\begin{equation}\label{eq:elldef}
f_\eta:=\frac{g}{\big(j-j(\tau_e)-\eta\big)^{m}},\qquad
\eta\in\CC^{\times},\qquad j(\tau_e)+\eta\notin[0,1728],
\end{equation}
so that $f_0$ has a pole of order $P=mn-a$ at $\tau_e$. Then no $f_\eta$ with $\eta\neq0$ has a
pole on $\SLZ\!\cdot\!\gamma^{\rm arc}$, the limit
\begin{equation}\label{eq:arcell}
\hat r_{f_0}\;:=\;\lim_{\eta\to0}\ \eta^{(P-1)/n}\,\hat r_{f_\eta}
\end{equation}
exists and lies in $W$, and is independent of the direction of approach up to an $n$-th root of
unity.\footnote{\textbf{$\;$This lemma and Lemma~\ref{lem:arcon} overlap: at $P=1$ the exponent is
$0$, no rescaling occurs, and~\eqref{eq:arcell} degenerates to the indentation case. The two may
want merging.}}
\end{lemma}

\begin{proof}
Write $j(\tau)-j(\tau_e)=c\,(\tau-\tau_e)^{n}\big(1+O(\tau-\tau_e)\big)$ with $c\neq0$, the zero
being of order exactly $n$, and $g=g_a(\tau-\tau_e)^{a}\big(1+O(\tau-\tau_e)\big)$ with $g_a\neq0$.
Let $\kappa$ denote either kernel of Definition~\ref{def:Lint}; both are holomorphic near $\tau_e$.

Fix $R>0$ so small that $D_R=\{|\tau-\tau_e|<R\}$ meets $f_0^{-1}(\infty)$ only at $\tau_e$,
contains no other point of $\SLZ\!\cdot\!\tau_e$, and carries both expansions; split the contour as
$\gamma_{\rm in}\cup\gamma_{\rm out}$ accordingly. For small $\eta$ the poles of $f_\eta$ near
$\tau_e$ stay in $D_R$, so $f_\eta\to f_0$ uniformly on $\gamma_{\rm out}$ and
$\int_{\gamma_{\rm out}}f_\eta\,\kappa\,d\tau=O(1)$.

Choose $\delta$ with $c\,\delta^{n}=\eta$ and substitute $\tau=\tau_e+\delta u$. Then
\begin{equation}\label{eq:ellscale}
j-j(\tau_e)-\eta=c\,\delta^{n}\big(u^{n}-1+O(\delta u^{n+1})\big),\qquad
g=g_a\,\delta^{a}u^{a}\big(1+O(\delta u)\big),
\end{equation}
while $d\tau=\delta\,du$ and $\kappa(\tau)=\kappa(\tau_e)+O(\delta u)$, so that
\begin{equation}\label{eq:ellmodel}
f_\eta\,\kappa\,d\tau
=\frac{g_a\,\kappa(\tau_e)}{c^{m}}\;\delta^{-(P-1)}\;
\frac{u^{a}\,du}{(u^{n}-1)^{m}}\,\big(1+O(\delta u)\big),
\end{equation}
the exponent $a+1-mn=-(P-1)$ being fixed by $P$ and $n$ alone.

Under this substitution the $n$ poles of $f_\eta$ nearest $\tau_e$ sit at $u^{n}=1$, and
$\gamma_{\rm in}$ becomes a path reaching out to $|u|\sim R/|\delta|$, tending as $\delta\to0$ to
the straight path $\mathcal{C}$ through $u=0$ along the direction of $\gamma^{\rm arc}$ at
$\tau_e$: a full line when $\tau_e=i$ lies in the interior of the arc, and a ray when
$\tau_e\in\SLZ\!\cdot\!\rho$ is an endpoint. Such a path meets a pole exactly when its direction is
an $n$-th root of unity, that is when $\arg\eta\equiv n\arg(\text{direction})+\arg c\pmod{2\pi}$ ---
a single value of $\arg\eta$, and necessarily the one for which the displaced poles still lie on
$\gamma^{\rm arc}$. The hypothesis $j(\tau_e)+\eta\notin[0,1728]=j(\gamma^{\rm arc})$ excludes
precisely that value, so $\mathcal{C}$ is bounded away from $u^{n}=1$.

The integrand of~\eqref{eq:ellmodel} decays as $u^{a-mn}=u^{-P}$, so its tail converges for $P>1$,
and replacing $|u|<R/|\delta|$ by all of $\mathcal{C}$ costs $O\big((R/|\delta|)^{1-P}\big)$; the
correction $O(\delta u)$ contributes $O(\delta)$ for $P>2$ and $O\big(\delta\log(R/|\delta|)\big)$
at $P=2$. As $\delta\propto\eta^{1/n}$, multiplying by $\eta^{(P-1)/n}\propto\delta^{P-1}$ sends
the $\gamma_{\rm out}$ term to zero and leaves
\begin{equation}\label{eq:elllimit}
\lim_{\eta\to0}\eta^{(P-1)/n}\!\!\int_{\gamma}\!f_\eta\,\kappa\,d\tau
\;\propto\;\frac{g_a\,\kappa(\tau_e)}{c^{m}}\int_{\mathcal{C}}\frac{u^{a}\,du}{(u^{n}-1)^{m}} ,
\end{equation}
finite because $\mathcal{C}$ avoids the singularities of the integrand.

Every special value of $L^*$ scales in this way, as does $\Phi(f_\eta)$, which is the same
computation with $\kappa\equiv1$, while $r_{E_k}$ does not depend on $\eta$; hence
$\eta^{(P-1)/n}\hat r_{f_\eta}$ converges. Each $f_\eta$ with $\eta\neq0$ has its poles off
$\SLZ\!\cdot\!\gamma^{\rm arc}$, so $\hat r_{f_\eta}\in W$ by Lemma~\ref{lem:arcoff}, and $W$ is
closed; the limit therefore lies in $W$. Finally $c\,\delta^{n}=\eta$ fixes $\delta$ only up to an
$n$-th root of unity, and another choice multiplies $\delta^{P-1}$ by such a root.
\end{proof}

As $m$ ranges over $m\ge1$ and $a$ over $0\le a<n$, the order $P=mn-a$ of~\eqref{eq:elldef} runs
over every positive integer, so Lemma~\ref{lem:arcell} is not restricted to special pole orders.

\begin{theorem}\label{thm:arcperiod}
For $f\in F_k$, and $W\subset V_n$ from Definition~\ref{def:w}, the period polynomial
$\hat r_f(X,Y)\in W$.
\end{theorem}

\begin{proof}
Let $E\subset f^{-1}(\infty)$ be the poles of $f$ lying on $\SLZ\!\cdot\!\rho$ or on
$\SLZ\!\cdot\!i$, and $N$ those lying on $\gamma^{\rm arc}$ but not at an elliptic point; the
remaining poles meet $\gamma^{\rm arc}$ nowhere. The three mechanisms are local to these three
families and may be applied together.

If $E$ and $N$ are both empty, Lemma~\ref{lem:arcoff} gives $\hat r_f\in W$ directly, with no
condition on where the other poles lie or how deep they are.

If $E$ is empty and $N$ is not, Definition~\ref{def:georef} takes $\hat\gamma^S_f=\hat\gamma^T_f$,
and both relations hold by Lemma~\ref{lem:arcon}; the same value is obtained from
Definition~\ref{def:arcsplit} by Lemma~\ref{lem:arcdeform}, so the two prescriptions do not
compete.

If $E$ is non-empty, deform as in Definition~\ref{def:arcsplit}, moving the orbits in $E$ off
$\gamma^{\rm arc}$ while leaving the rest of $f^{-1}(\infty)$ where it is. Every member of the
resulting family falls under one of the two previous cases, so each has $\hat r_{f_\eta}\in W$;
Lemma~\ref{lem:arcell} supplies the exponent for which the limit exists, applied at each elliptic
point of $E$ in turn, and $W$ is closed. Hence $\hat r_f\in W$.
\end{proof}

The construction uses only the $k-1$ special values $L^*(\,\cdot\,,\ell+1)$, $0\le\ell\le n$, of
the finite-contour $L$-function of Section~\ref{sec:arcL}, together with the single subtraction
$\Phi(f)\,r_{E_k}$. No reference family is chosen, and no projection onto $W$ is performed.

Three choices enter, and none of them moves $\hat r_f$ out of $W$. In case~2 the side of the
indentation is free, and by Lemma~\ref{lem:arcon} the two differ within $W$. In case~3 the
deformation~\eqref{eq:elldef} and the branch of $\eta^{(P-1)/n}$ are free, the latter contributing
an $n$-th root of unity and hence an overall scalar. In Definition~\ref{def:rfhat} the reference
form $E_k$ may be replaced by any $g$ with $\Phi(g)=1$, and two such choices shift $\hat r_f$ by
$\Phi(f)\,r_{g-g'}$, where $g-g'$ runs over the forms annihilated by $\Phi$; how large that shift
space is depends on the class from which $g$ is drawn, and we do not settle it here.

\paragraph{The $c_T$ term in the even weak-form entry.}
The weak form's period polynomial $r_{\widehat\Delta_k}$ is, like the cusp form's, annihilated by
$1+S$ and $1+U+U^2$, so it too lies in the period space $\langle W_+,\,p_0,\,W_-\rangle$; but it
carries a component along the Eisenstein polynomial $p_0=X^{k-2}-Y^{k-2}$ that the cusp form lacks.
Since $p_0$ occupies only the two extreme monomials $X^{k-2},Y^{k-2}$---the $s=1$ and $s=k-1$
coefficients---and $W_+$ is not orthogonal to $p_0$ under the Haberland pairing, the projection
$\eta^+$ of~\eqref{eq:project} absorbs part of this Eisenstein content but not all. The unabsorbed
remainder surfaces as an additive constant $c_T$ in the extreme even coefficient of
$r_{\widehat\Delta_k}$ in the matrices below. It occurs only there: the odd sector has no Eisenstein
direction (so $\eta^-$ is exact), the cusp form has no $p_0$-component (so no $c_T$), and, lying
outside the projection, $c_T$ enters neither $\eta^\pm$ nor $\alpha_k$.

\subsection{The geodesic reference contour at $\tau_0=\rho+1$}\label{sec:geo}

\begin{definition}\label{def:georef}
Let $\gamma^{\rm arc}$ be the geodesic segment $\{e^{i\theta}:\tfrac{\pi}{3}\le\theta\le\tfrac{2\pi}{3}\}$,
oriented from $\rho$ to $\rho+1$, and let $\tau_0=\rho+1$. The \emph{geodesic reference class} is
the class of $\hat\gamma^S_f=\hat\gamma^T_f=\gamma^{\rm arc}$.
\end{definition}

This is admissible by Definition~\ref{def:segments}: $-1/(\rho+1)=\rho$ and $(\rho+1)-1=\rho$, so
both segments have the same endpoints and may be taken to be the same path. $\gamma^{\rm arc}$ is
the bottom boundary of $\mathcal{F}$, and passes through the three elliptic points $\rho$, $i$,
$\rho+1$.

Since $\rho$ and $\tau_0$ lie on $\gamma^{\rm arc}$, a form with a pole on $\SLZ\!\cdot\!\rho$ has one
at each endpoint, and $\tau_0=\rho+1$ is then excluded by Proposition~\ref{prop:indepHomotopy},
whose hypothesis is $\tau_0\notin\Gamma\!\cdot\!f^{-1}(\infty)$. The remedy is to move the base
point along the arc rather than off it: set
\begin{equation}\label{eq:geodelta}
\tau_0(\delta):=e^{i(\frac{\pi}{3}+\delta)},\qquad \delta>0,
\end{equation}
so that $S^{-1}\tau_0(\delta)=e^{i(\frac{2\pi}{3}-\delta)}$ and $\hat\gamma^S_f$ is the sub-arc
between them, meeting neither $\rho$ nor $\rho+1$. This is not a regularisation. For every such
$\delta$ the hypothesis of Proposition~\ref{prop:indepHomotopy} holds, so $L^*$ does not depend on
$\delta$ at all: the shortening of the segments is exactly compensated by the movement of the base
point, and no limit is taken. The displacement stays on $\gamma^{\rm arc}$, which is $S$-stable, so
the $S^2$-loop of Lemma~\ref{lem:geoloops} retraces at every $\delta$ and $r_f|_{(1+S)}=0$. The
$(1+U+U^2)$ loop does not degenerate, since $U$ fixes only $\tau_0(0)$, and the defect it leaves is
likewise independent of $\delta$. Contrast Definition~\ref{def:null_homotopy}, where the base point
is displaced off the imaginary axis onto $i(1+\delta)$, the limit is genuine, and
Lemma~\ref{lem:wallvalue} is needed to show it survives.

Independence of $\tau_0$ does not by itself fix the class, and one further choice is needed here.
The endpoints $\tau_0(\delta)-1$ and $\tau_0(\delta)$ of $\hat\gamma^T_f$ lie at distance $O(\delta)$
from the poles at $\rho$ and $\rho+1$, and paths running on opposite sides of them are not homotopic
in $\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$; by Lemma~\ref{lem:wall} the two give values of $r_f$
differing by residues. We take $\hat\gamma^T_f$ on the $|\tau|>1$ side of both, that is, in the class
of the straight segment from $\tau_0(\delta)-1$ to $\tau_0(\delta)$.

A pole on the arc is then covered by the same formula as a pole off it.

\begin{lemma}\label{lem:georho}
Let $f\in F_k$ have a pole of order $P$ at $\rho$ and no other pole on $\SLZ\!\cdot\!\gamma^{\rm arc}$,
and let $r_f$ be evaluated in the class of~\eqref{eq:geodelta}. Then
\begin{equation}\label{eq:georho}
\tilde r_f\big|_{(1+S)}=0,\qquad
\tilde r_f\big|_{(1+U+U^2)}=-Q_f ,
\end{equation}
with $\tilde r_f=r_f-\Phi(f)r_{E_k}$ as in~\eqref{eq:geoperiod} and $Q_f$ the polynomial of
Definition~\ref{def:Qf} formed from the single point $p=\rho+1$.
\end{lemma}

The enclosed set is one point rather than a triple because $U$ fixes $\rho+1$, so the $U$-orbit that
Lemma~\ref{lem:rfWFk} sums over degenerates. Writing $\zeta=\rho+1$ as in
Lemma~\ref{lem:ellLaurent} and using $\gamma p=p\Rightarrow(X-pY)^n|\gamma=(a-pc)^n(X-pY)^n$, the
eigenvalue of $U$ on $(X-\zeta Y)^n$ is $(1-\zeta)^n=\zeta^{-n}$, so
$(X-\zeta Y)^n|_{(1+U+U^2)}=\big(1+\zeta^{-n}+\zeta^{-2n}\big)(X-\zeta Y)^n$, which is $3(X-\zeta Y)^n$
when $6\mid n$ and $0$ otherwise. The first case is $k\equiv2\pmod 6$, exactly the condition under
which Lemma~\ref{lem:ellLaurent} permits a simple pole at $\rho$; when $6\nmid n$ the $\alpha=1$ term
of $Q_f$ is invisible to the relation and the defect is carried by the deeper pole weights.

The defect is removable, and for an intrinsic reason: $Q_f$ is a residue at a point that the
relation's own generator fixes.

\begin{lemma}\label{lem:kersum}
Let $n=k-2\ge2$. Then $\ker(1+S)+\ker(1+U+U^2)=V_n$. Consequently every $q\in V_n$ with $q|_U=q$
equals $w|_{(1+U+U^2)}$ for some $w\in\ker(1+S)$, unique modulo $W$.
\end{lemma}

\begin{proof}
As $n$ is even $\pm I$ acts trivially, so $S^2=U^3=1$ on $V_n$ and $1+S$, $1+U+U^2$ are the averages
over $\langle S\rangle$, $\langle U\rangle$ up to a factor; their images are the fixed subspaces
$V_n^S$, $V_n^U$. The pairing~\eqref{inner} is $\SLZ$-invariant, so
$\langle P|_\gamma,Q\rangle=\langle P,Q|_{\gamma^{-1}}\rangle$ and both operators are self-adjoint.
Hence
\begin{equation}\label{eq:kerperp}
\big(\ker(1+S)+\ker(1+U+U^2)\big)^{\perp}=\im(1+S)\cap\im(1+U+U^2)=V_n^S\cap V_n^U .
\end{equation}
Since $U=TS$, an element of that intersection is $\SLZ$-invariant; being $T$-invariant it is a
multiple of $Y^n$, which $S$ carries to a multiple of $X^n$, so it vanishes. For the second claim
split $\tfrac13q=w+v$ with $w\in\ker(1+S)$, $v\in\ker(1+U+U^2)$; then
$w|_{(1+U+U^2)}=\tfrac13q|_{(1+U+U^2)}=q$. Two such $w$ differ by an element of
$\ker(1+S)\cap\ker(1+U+U^2)=W$.
\end{proof}

\begin{lemma}\label{lem:Qinv}
Let $f\in F_k$ have a pole at $p\in\HH$ and let $\gamma\in\SLZ$ fix $p$. Then the contribution of $p$
alone to $Q_f$ satisfies $Q|_\gamma=Q$.
\end{lemma}

\begin{proof}
The $V_n$-valued form $\omega_f:=f(\tau)(X-\tau Y)^n\,d\tau$ satisfies
$\gamma^*\omega_f=\omega_f|_{\gamma^{-1}}$: combine $f|_k\gamma=f$,
$(c\tau+d)^n\big(X-(\gamma\tau)Y\big)^n=(X-\tau Y)^n|_{\gamma^{-1}}$ and
$d(\gamma\tau)=(c\tau+d)^{-2}d\tau$, whose weights cancel because $n=k-2$. Residues are unchanged by
the substitution, so $\Res_{\tau=\gamma p}\omega_f=\big(\Res_{\tau=p}\omega_f\big)\big|_{\gamma^{-1}}$,
and $Q=(2\pi i)^{n+2}\Res_{\tau=p}\omega_f$ by the proof of Lemma~\ref{lem:rfWFk}. With $\gamma p=p$
this is $Q|_{\gamma^{-1}}=Q$, and $\gamma$ has finite order.
\end{proof}

At $p=\rho+1$, fixed by $U$, Lemma~\ref{lem:Qinv} places $Q_f$ in $V_n^U$ --- this is
Lemma~\ref{lem:ellLaurent} carried from the Laurent weights to $V_n$ --- and Lemma~\ref{lem:kersum}
then returns a $w\in\ker(1+S)$ with $w|_{(1+U+U^2)}=Q_f$, so that $\tilde r_f+w\in W$
by~\eqref{eq:georho}. The hypothesis is not vacuous:
$\im\big[(1+U+U^2)|_{\ker(1+S)}\big]=V_n^U$ has dimension $2\lfloor n/6\rfloor+1$, against
$\dim V_n=n+1$. This determines $\hat r_f$ only modulo $W$, the ambiguity
Theorem~\ref{thm:period} already carries.

Poles elsewhere on $\gamma^{\rm arc}$ are handled by deforming rather than by sliding. Writing the
contour in polar form $\tau(\theta)=r(\theta)e^{i\theta}$, the map $S$ sends $(r,\theta)$ to
$(1/r,\pi-\theta)$, so the path retraces under $S$, as Lemma~\ref{lem:geoloops} requires, precisely
when $\log r$ is odd about $\theta=\tfrac{\pi}{2}$: an outward bulge on one half of the arc against
an equal inward one on the mirror half. A pole at $e^{i\phi}$ with $\phi\ne\tfrac{\pi}{2}$ and its
$S$-image at $e^{i(\pi-\phi)}$ are distinct, so passing outside the first and inside the second is
such a deformation, and the $S$-relation is untouched.

At $\phi=\tfrac{\pi}{2}$ this fails, and not because $i$ is elliptic but because it is the fixed
point of the symmetry the contour relies on. Oddness forces $\log r(\tfrac{\pi}{2})=0$, so every
$S$-symmetric path runs through $i$ and no deformation avoids it. One may still pass to one side:
by Lemma~\ref{lem:wall} the two sides differ by a full loop about the pole, hence by
$(2\pi i)^{n+2}r^*_{f,i}(1)(X-iY)^{n}$ in $r_f$, and either choice has a defect that can be
subtracted. But neither is $S$-symmetric, since $S$ exchanges the sides, and $\gamma$ together with
$S\gamma$ then bounds a region containing $i$. The principal value is the average of the two, so it
differs from each by half that residue, and it is the unique $S$-fixed choice; this is why
$r_f|_{(1+S)}=0$ survives at $i$, and why no correction is left over there.

We turn to the analogue, for this class, of the explicit formulae above. Two of their ingredients
are already available: the polylog projection needs only a change of constant, and the Laurent
relations at the elliptic points are contour-free.

The projector of Definition~\ref{def:pole_split} is thresholded so that $(1-\widehat P)f$ admits a
$q$-series where it is used. That is a statement about a half-plane, since a $q$-series converges
exactly on ${\rm Im}\,\tau>h$ with $h$ the height of the highest surviving pole, and the operative
height is that of the base point: with $X_n=-2\pi in\tau_0$ as in Theorem~\ref{thm:weakL},
$|\Gamma(s,X_n)|\sim|X_n|^{{\rm Re}\,s-1}e^{-2\pi n\,{\rm Im}\,\tau_0}$, so the mode sum converges
precisely when $c_{Rf}(n)=o\big(e^{2\pi n\,{\rm Im}\,\tau_0}\big)$. For $\tau_0=\rho+1$ this reads:

\begin{definition}\label{def:geoproj}
$\displaystyle\widehat P^{\,\rm geo}:=\sum_{{\rm Im}\,\tau_p\ge\frac{\sqrt3}{2}}\widehat P_{\tau_p}$.
\end{definition}

The threshold is not the arc but the horizontal line through its endpoints, and it is forced.
Retaining Definition~\ref{def:pole_split}'s threshold of $1$ would leave surviving poles of height
up to $1$ against a suppression of only $e^{-\pi\sqrt3\,n}$, and the mode sum would diverge like
$e^{2\pi(1-\frac{\sqrt3}{2})n}$. Thresholding at the arc itself is not available either, as the
region below the arc and above $\tfrac{\sqrt3}{2}$ would survive and carry heights arbitrarily close
to $1$, so the remainder's $q$-series would converge nowhere on the contour but at $i$. With
Definition~\ref{def:geoproj} the surviving heights are below $\tfrac{\sqrt3}{2}$ and the mode sum
converges on the whole contour, the base point being its lowest point.

The relations Lemma~\ref{lem:Laurent} imposes at $i$ hold at $\rho$ as well, in the same form:

\begin{lemma}\label{lem:ellLaurent}
Let $\tau_e$ be $i$ or $\rho$, of order $n=2$ or $n=3$, let $\gamma=\left(\begin{smallmatrix}a&b\\c&d\end{smallmatrix}\right)$
be the generator $S$ or $ST$ of its stabiliser in $\SLZ$, and set $\zeta:=c\tau_e+d$, so that
$c=1$ and $\zeta=e^{i\pi/n}$. Let $f\in F_k$ have a pole of order $P$ at $\tau_e$ with pole weights
$r^*_{f,\tau_e}(m)$. Modularity is equivalent to one relation for each principal-part order $\mu$,
\begin{equation}\label{eq:ellLaurent}
\sum_{m=\mu}^{P} r^*_{f,\tau_e}(m)\binom{m}{\mu}\,\zeta^{m}
=\zeta^{k}\sum_{m=\mu}^{P} r^*_{f,\tau_e}(m)\binom{k}{m-\mu}\,\zeta^{-m},
\qquad \forall\mu=1,\dots,P,
\end{equation}
and the leading relation is $\big(\zeta^{2P}-\zeta^{k}\big)r^*_{f,\tau_e}(P)=0$, so that
$2P\equiv k \pmod{2n}$.
\end{lemma}
\begin{proof}
Write $\tau=\tau_e+\varepsilon$. Since $\det\gamma=1$ and $\gamma\tau_e=\tau_e$,
\begin{equation}\label{eq:ellshift}
\gamma\tau-\tau_e=\frac{\tau-\tau_e}{(c\tau+d)(c\tau_e+d)}=\frac{\varepsilon}{\zeta(\zeta+\varepsilon)},
\qquad (c\tau+d)^{k}=\zeta^{k}\big(1+\varepsilon/\zeta\big)^{k},
\end{equation}
using $c=1$. Hence $(\gamma\tau-\tau_e)^{-m}=\zeta^{2m}\varepsilon^{-m}(1+\varepsilon/\zeta)^{m}$, and
matching the coefficient of $\varepsilon^{-\mu}$ in
$f(\gamma\tau)=(c\tau+d)^{k}f(\tau)$ through
$(1+\varepsilon/\zeta)^{j}=\sum_{l}\binom{j}{l}\zeta^{-l}\varepsilon^{l}$ gives
$\sum_{m\ge\mu}r^*(m)\zeta^{2m}\binom{m}{\mu}\zeta^{-(m-\mu)}
=\zeta^{k}\sum_{m\ge\mu}r^*(m)\binom{k}{m-\mu}\zeta^{-(m-\mu)}$;
dividing by $\zeta^{\mu}$ yields~\eqref{eq:ellLaurent}. At $\mu=P$ only $m=P$ survives on each side.
\end{proof}

At $n=2$, $\zeta=i$ and~\eqref{eq:ellLaurent} is~\eqref{eq:Laurent}, the order condition reading
$P\equiv k/2\pmod 2$. At $n=3$, $\zeta=\rho+1=e^{i\pi/3}$ and the order condition reads
$P\equiv-k\pmod3$, so a simple pole at $\rho$ requires $k\equiv2\pmod6$ and a double pole
$k\equiv4\pmod6$. In the double-pole case~\eqref{eq:ellLaurent} at $\mu=1$ gives
$r^*_{f,\tau_e}(1)=\tfrac{i(k-2)}{2\,{\rm Im}\,\tau_e}\,r^*_{f,\tau_e}(2)$ at both points, which is
$\tfrac{i(k-2)}{2}$ at $i$ and $\tfrac{i(k-2)}{\sqrt3}$ at $\rho$. The constraint therefore depends
on the elliptic point only through the order of its stabiliser.

It remains to evaluate the projected blocks. Write $\lambda_{N,z}$ for the function
$\Li_{-N}(e^{2\pi i(\tau-z)})$ of Definition~\ref{def:IN}.

\begin{definition}\label{def:geoIN}
$I^{\rm geo}_N(s,z):=L^*\big(\lambda_{N,z},\,s;\,[\hat\gamma^S_f],[\hat\gamma^T_f]\big)$, evaluated in
the geodesic reference class of Definition~\ref{def:georef}.
\end{definition}

Both segments run from $\rho$ to $\tau_0=\rho+1$, and $\lambda_{N,z}$ is holomorphic off $z+\ZZ$.
Since $\tk$ is singular only at $\tau\in\{-1,-2,\dots\}$ and $\tau^{s-1}$ only at $\tau=0$, neither
kernel obstructs deformation, so each integral may be taken along any path with those endpoints
avoiding $z+\ZZ$. For $|z|\ge1$ no point of $z+\ZZ$ lies between $\gamma^{\rm arc}$ and the
horizontal chord $\rho\to\tau_0$ at height $\tfrac{\sqrt3}{2}$, and we evaluate there.

\begin{lemma}\label{lem:geopolylog}
Let ${\rm Im}\,z>\tfrac{\sqrt3}{2}$ and $|z|\ge1$, set $\tau_0=\rho+1$, and let
$\gamma^*(s,x):=\sum_{j\ge0}\frac{(-x)^{j}}{j!\,(j+s)}$, entire in $x$. With
\begin{equation}\label{eq:geoG}
G^{\rm geo}_{s,k}(-n):=e^{-\pi is/2}\Big[\tau_0^{\,s}\,\gamma^*\big(s,2\pi in\tau_0\big)
-\rho^{\,s}\,\gamma^*\big(s,2\pi in\rho\big)\Big]
\;+\;\frac{\Gamma\big(s,2\pi in\tau_0\big)}{(-2\pi n)^{s}}
\;+\;i^{k}\frac{\Gamma\big(k-s,2\pi in\tau_0\big)}{(-2\pi n)^{k-s}},
\end{equation}
\begin{equation}\label{eq:geoG0}
G^{\rm geo}_{s,k}(0):=e^{-\pi is/2}\,\frac{\tau_0^{\,s}-\rho^{\,s}}{s}
-\frac{(\tau_0/i)^{s}}{s}-i^{k}\frac{(\tau_0/i)^{k-s}}{k-s}.
\end{equation}
Then
\begin{equation}\label{eq:geoIN}
I^{\rm geo}_N(s,z)=-\delta_{N,0}\,G^{\rm geo}_{s,k}(0)
+(-1)^{N+1}\sum_{n\ge1}n^{N}e^{2\pi inz}\,G^{\rm geo}_{s,k}(-n),
\end{equation}
the sum converging at rate $e^{-2\pi({\rm Im}\,z-\frac{\sqrt3}{2})}$.
\end{lemma}
\begin{proof}
On the chord $|e^{2\pi i(\tau-z)}|>1$, so the defining series of $\Li_{-N}$ diverges and inversion
$\Li_{-N}(w)=(-1)^{N+1}\Li_{-N}(1/w)-\delta_{N,0}$ rewrites $\lambda_{N,z}$ as
$(-1)^{N+1}\sum_{n\ge1}n^{N}e^{2\pi inz}q^{-n}-\delta_{N,0}$, uniformly convergent there since
${\rm Im}\,z>\tfrac{\sqrt3}{2}$. Integrating term by term, $G^{\rm geo}_{s,k}(-n)=L^*(q^{-n},s)$ in
this class: the last two terms of~\eqref{eq:geoG} are $L^*_T$ by Lemma~\ref{lem:Tclosed-Mk} at
$\tau_0$, and the first is $L^*_S=e^{-\pi is/2}\int_{\rho}^{\tau_0}e^{-2\pi in\tau}\tau^{s-1}d\tau$,
which on expanding $e^{-2\pi in\tau}$ and integrating $\tau^{j+s-1}$ term by term gives
$e^{-\pi is/2}\sum_{j\ge0}\frac{(-2\pi in)^{j}}{j!\,(j+s)}\big[\tau_0^{\,j+s}-\rho^{\,j+s}\big]$,
the first bracket of~\eqref{eq:geoG}. Likewise $G^{\rm geo}_{s,k}(0)=L^*(1,s)$. The rate follows from
$|e^{2\pi inz}|=e^{-2\pi n\,{\rm Im}\,z}$ against $|G^{\rm geo}_{s,k}(-n)|\sim e^{\pi\sqrt3\,n}$.
\end{proof}

Corollary~\ref{cor:Sclosed} is deliberately not used here: it folds the $S$-segment about $i$ using
modularity, and $q^{-n}$ is only $T$-periodic, so its $S$-integral must be taken directly. This is
the one place where the two classes genuinely differ. At $\tau_0=i$ the $S$-segment is degenerate
and $L^*_S$ drops out of Lemma~\ref{lem:polylogcf} entirely; here it contributes the first term
of~\eqref{eq:geoG}, and the threshold in the convergence rate moves from $1$ to $\tfrac{\sqrt3}{2}$
in step with Definition~\ref{def:geoproj}.

The $S$-term is written with $\gamma^*$ because that form is branch-free: $\gamma^*$ is
single-valued, so the only multivalued factors are the explicit $\tau_0^{\,s}$ and $\rho^{\,s}$,
and those are unambiguous since $\arg\tau\in[\tfrac{\pi}{3},\tfrac{2\pi}{3}]$ along the chord. It
may equally be written with upper incomplete gammas, provided the sheet relation is carried. Under
$u=2\pi in\tau$ the chord's image runs from $\arg u=\tfrac{7\pi}{6}$ at $\rho$ to $\tfrac{5\pi}{6}$
at $\tau_0$, so the $\rho$ endpoint sits one turn off principal; since $\gamma(s,x)=x^{s}\gamma^*(s,x)$
with $\gamma^*$ single-valued, a shift of $\arg x$ by $2\pi$ gives
\begin{equation}\label{eq:sheet}
\Gamma^{\rm cont}(s,x)=e^{2\pi is}\,\Gamma(s,x)+\big(1-e^{2\pi is}\big)\Gamma(s),
\end{equation}
and hence
\begin{equation}\label{eq:geoSupper}
S_n(s)=(2\pi in)^{-s}\Big[e^{2\pi is}\,\Gamma\big(s,2\pi in\rho\big)
+\big(1-e^{2\pi is}\big)\Gamma(s)-\Gamma\big(s,2\pi in\tau_0\big)\Big].
\end{equation}
Both the sheet factor and the additive constant are trivial at integer $s$, which is all that
Definition~\ref{def:rf} requires. The $T$-term needs neither, since by
Lemma~\ref{lem:Tclosed-Mk} only $\tau_0$ appears there, at $\arg(2\pi in\tau_0)=\tfrac{5\pi}{6}$,
the $\rho$ endpoint having been absorbed by the periodicity of $\tk$.

\subsection{Period polynomials in the geodesic class}\label{sec:geoperiod}

The two obstructions to $r_f\in W$ move in opposite directions as the reference contour is
lowered. The cohomological one shrinks: the region whose boundary is the $(1+U+U^2)$ loop, and
whose enclosed poles obstruct Lemma~\ref{lem:rfWmero}, gets smaller. The analytic one grows: more
of each pole orbit sits above the contour, so the $q$-series of Section~\ref{sec:mero} converges on
less of it and the strip constant of~\eqref{eq:Fcirc} picks up more residues. The horocycle
${\rm Im}\,\tau=1$ is the compromise, and a natural one, being the unique horocycle tangent to
$\partial\mathcal{F}$. This subsection develops the opposite extreme, where the cohomological
obstruction vanishes outright.

\begin{lemma}\label{lem:geoloops}
In the geodesic reference class both relator loops are degenerate: the $S^2$-loop retraces for any
base point on $\gamma^{\rm arc}$, and the $(1+U+U^2)$ loop retraces at $\tau_0=\rho+1$. Hence
$\hat C_{\pm I}=0$ for every $f\in F_k$ with no pole on $\SLZ\!\cdot\!\gamma^{\rm arc}$.
\end{lemma}
\begin{proof}
$S$ acts on the unit circle by $e^{i\theta}\mapsto e^{i(\pi-\theta)}$, which preserves
$\tfrac{\pi}{3}\le\theta\le\tfrac{2\pi}{3}$ and reverses orientation, fixing $i$. So $S\gamma^S$ is
$\gamma^S$ reversed and the $S^2$-loop retraces, for any base point on the arc. For the second,
$U=TS$ has fixed points the roots of $\tau^2-\tau+1$, i.e.\ $\tau=\rho+1$, so $U^{-1}\tau_0=\tau_0$
and the $U$-path is closed. With $\gamma^S=\gamma^T$ it is $S^{-1}\gamma^T$ followed by $\gamma^S$,
namely the arc traversed and retraced; its $U$- and $U^2$-transports are retracings of
$U\gamma^{\rm arc}$ and $U^2\gamma^{\rm arc}$ through the same fixed point. Each bounds nothing, so
the winding number about every pole is zero and Cauchy gives $\hat C_{-I}=C_{-I}=0$.
\end{proof}

The contrast with Definition~\ref{def:Tri} is the point: there $\partial\mathcal{T}$ bounds a region
of hyperbolic area $\pi-3$, and poles inside it obstruct. Here the corresponding region is empty for
every $f$, so no analogue of $\mathcal{T}$, of $Q_f$, or of the reference families of
Definition~\ref{def:reffam} is required.



\begin{definition}\label{def:Phi}
For $f\in F_k$ with no pole on $\gamma^{\rm arc}$, set
$\Phi(f):=\int_{\gamma^{\rm arc}}f(\tau)\,d\tau$.
\end{definition}

$\Phi$ is the strip constant of~\eqref{eq:Fcirc} for this contour: it equals $c_f(0)$ when $f$ has no
pole above $\gamma^{\rm arc}$, and $c_f(0)$ less $2\pi i$ times the residues in the strip otherwise.
In particular $\Phi(E_k)=1$, and $\Phi(f)=0$ for $f\in S_k$.

\begin{theorem}\label{thm:geoperiod}
Let $f\in F_k$ have no pole on $\SLZ\!\cdot\!\gamma^{\rm arc}$, and let $r_f$ be given by
Definition~\ref{def:rf} with $L^*(f,s)$ evaluated in the geodesic reference class. Then
\begin{equation}\label{eq:geoperiod}
\tilde r_f\;:=\;r_f-\Phi(f)\,r_{E_k}\;\in\;W ,
\end{equation}
with no condition on the location or order of the poles of $f$.
\end{theorem}
\begin{proof}
Lemma~\ref{lem:rfW} invokes $f\in S^!_k$ only through $\hat C_{\pm I}=0$ and $\hat C_T=0$. The first
holds for every such $f$ by Lemma~\ref{lem:geoloops}. For the second, the computation in the proof
of Lemma~\ref{lem:cocycle} gives $\hat C_T=-(2\pi i)^{n+1}A_0\Phi(f)$, so the only surviving
obstruction is linear in $\Phi(f)$ and the induced defect $r_f|_{(1+U+U^2)}$ is $\Phi(f)$ times a
polynomial independent of $f$. Taking $f=E_k$, for which $\Phi=1$, identifies that polynomial as
$r_{E_k}|_{(1+U+U^2)}$, and $r_{E_k}|_{(1+S)}=0$, giving~\eqref{eq:geoperiod}.
\end{proof}
