\title{A finite Hurwitz-kernel cocycle for periods and $L$-functions
of meromorphic modular forms on $\SLZ$}
\author{David A.\ McGady}
\date{}

\begin{document}
\maketitle

\begin{abstract}
We give one finite, regulator-free contour functional --- a two-segment
cocycle integral with an explicit Hurwitz-zeta $T$-kernel --- that
realizes both the period polynomials and the $L$-functions of modular
forms on $\SLZ$ whose poles may lie anywhere: inside the fundamental
domain and at the cusp. From a finite base point $\tau_0\in\HH$ one builds
a cocycle whose cohomology class is independent of $\tau_0$. Removing its
$T$-component by a coboundary --- possible if and only if the constant
Fourier coefficient $c_f(0)$ vanishes --- yields the parabolic
representative, whose $S$-value is the period polynomial. The same kernel,
with the period monomial $\tau^{\ell}$ replaced by $\tau^{s-1}$, gives the
completed $L$-function on all of $M^!_k$, constant term allowed, with the
classical poles of $\Lambda$; for forms with interior poles it is
determined by the contour up to the residues crossed. The construction
introduces no new objects: the periods are Brown's quasi-periods in closed
form~\cite{Brown2017}, the $L$-values are those of
Bringmann--Fricke--Kent~\cite{BFK2011}. The contribution is the explicit
all-$s$ kernel and the single finite realization. All formulae are checked
numerically to $50$ digits~\cite{VerifyScripts}.
\end{abstract}

\section{A finite L-integral}\label{sec:Lint}

\begin{definition}[$S$- and $T$-segments]\label{def:segments}
Let $\tau_0\in\HH$ and $f\in F_k$, and let
$S=\left(\begin{smallmatrix}0&-1\\1&0\end{smallmatrix}\right)$,
$T=\left(\begin{smallmatrix}1&1\\0&1\end{smallmatrix}\right)$ generate $\Gamma=\SLZ$.
The \emph{$S$- and $T$-segments at $\tau_0$} are smooth paths in
$\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$,
\begin{equation}
\gamma^S_{\tau_0}:\ -1/\tau_0\ \longrightarrow\ \tau_0,\qquad
\gamma^T_{\tau_0}:\ \tau_0-1\ \longrightarrow\ \tau_0,
\end{equation}
running from $S^{-1}\tau_0=-1/\tau_0$ and $T^{-1}\tau_0=\tau_0-1$ to $\tau_0$.
\end{definition}

The pole-avoidance is inert for $f\in M^!_k$: with no interior poles,
$\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)=\HH$ and the segments are unobstructed.
Interior poles --- and the homotopy classes of contours they create --- become
relevant only for meromorphic $f$, in the final section.

\begin{definition}[Hurwitz $T$-kernel]\label{def:kernel}
The \emph{Hurwitz $T$-kernel} is, with $\zeta(s,x)=\sum_{m\ge0}(m+x)^{-s}$ the Hurwitz
zeta function (meromorphic in $s$, holomorphic in $x\in\HH$),
\begin{equation}
\tk(\tau,s):=\zeta(1-s,\tau+1)-e^{i\pi(s-1)}\,\zeta\big(s-(k-1),\tau+1\big).
\end{equation}
\end{definition}

\begin{definition}[$L$-integral]\label{def:Lint}
Let $f\in F_k$, $\tau_0\in\HH$, and $s\in\CC$, and let $\gamma^S_{\tau_0}$,
$\gamma^T_{\tau_0}$ be the $S$- and $T$-segments at $\tau_0$, as in
Definition~\ref{def:segments}. The \emph{$L$-integral} of $f$ is
\begin{equation}
L^*(f,s;\gamma^S_{\tau_0},\gamma^T_{\tau_0})
:=i^{-s}\Big[\int_{\gamma^S_{\tau_0}}\! f(\tau)\,\tau^{s-1}\,d\tau
+\int_{\gamma^T_{\tau_0}}\! f(\tau)\,\tk(\tau,s)\,d\tau\Big].
\end{equation}
\end{definition}

Its $\tau_0$- and path-dependence is dealt with below, in the main body.

\begin{definition}[$L$-function]\label{def:Lfun}
The \emph{$L$-function} of $f\in F_k$ is
\begin{equation}
L(f,s;\gamma^S_{\tau_0},\gamma^T_{\tau_0})
:=\frac{(2\pi)^s}{\Gamma(s)}\,L^*(f,s;\gamma^S_{\tau_0},\gamma^T_{\tau_0}).
\end{equation}
\end{definition}

Likewise; once that dependence is settled below, we drop the arguments and write
simply $L^*(f,s)$ and $L(f,s)$.

\begin{lemma}[Finiteness]\label{lem:finite}
For $f\in F_k$, both integrals in Definition~\ref{def:Lint} are finite for every
$s\in\CC$, every $\tau_0\in\HH$, and every choice of paths
$\gamma^S_{\tau_0},\gamma^T_{\tau_0}$.
\end{lemma}

\begin{proof}
Both segments are compact and lie in $\HH\setminus\Gamma\!\cdot\!f^{-1}(\infty)$, so $f$ is
continuous on them; $\tau^{s-1}$ is continuous on $\HH$, and $\tk(\tau,s)$ is continuous there
since $\zeta(\nu,\tau+1)$ is holomorphic in $\tau\in\HH$ (Definition~\ref{def:kernel}). A
continuous integrand on a compact path is integrable, for every $s\in\CC$.
\end{proof}

\begin{proposition}[Basepoint and path independence]\label{prop:indep}
For $f\in M^!_k$ the value $L^*(f,s;\gamma^S_{\tau_0},\gamma^T_{\tau_0})$ is independent
of $\tau_0\in\HH$ and of the paths: with no interior poles, $\HH$ is simply connected,
so all admissible contours are homotopic. We write $L^*(f,s)$ for the common value and
$L(f,s):=\frac{(2\pi)^s}{\Gamma(s)}L^*(f,s)$.
\end{proposition}

\begin{proof}
With no interior poles the integrands $f(\tau)\tau^{s-1}$ and $f(\tau)\tk(\tau,s)$ are
holomorphic on $\HH$, which is simply connected; by Cauchy's theorem each integral is unchanged
under any deformation fixing its endpoints, hence depends on those endpoints alone. This is
path-independence.

For independence of $\tau_0$ we differentiate. From $\zeta(\nu,x)=\sum_{m\ge0}(m+x)^{-\nu}$ one
has $\zeta(\nu,x+1)-\zeta(\nu,x)=-x^{-\nu}$; applied to both terms of the kernel at $x=\tau+1$,
\begin{equation}
\delta_T\tk(\tau,s):=\tk(\tau+1,s)-\tk(\tau,s)=-(\tau+1)^{s-1}+e^{i\pi(s-1)}(\tau+1)^{k-1-s}.
\end{equation}
The $T$-segment runs $\tau_0-1\to\tau_0$, so \emph{both} endpoints move with $\tau_0$; by
Leibniz's rule and the $T$-periodicity $f(\tau_0-1)=f(\tau_0)$,
\begin{equation}
\partial_{\tau_0}\!\!\int_{\gamma^T}\!\! f\,\tk
=f(\tau_0)\tk(\tau_0,s)-f(\tau_0-1)\tk(\tau_0-1,s)
=f(\tau_0)\,\delta_T\tk(\tau_0-1,s)
=f(\tau_0)\big[-\tau_0^{s-1}+e^{i\pi(s-1)}\tau_0^{k-1-s}\big].
\end{equation}
The $S$-segment runs $-1/\tau_0\to\tau_0$; its lower endpoint moves with
$\partial_{\tau_0}(-1/\tau_0)=\tau_0^{-2}$, so by Leibniz, the modularity
$f(-1/\tau_0)=\tau_0^k f(\tau_0)$, and the phase $(-1/\tau_0)^{s-1}=e^{i\pi(s-1)}\tau_0^{1-s}$,
\begin{equation}
\partial_{\tau_0}\!\!\int_{\gamma^S}\!\! f\,\tau^{s-1}
=f(\tau_0)\tau_0^{s-1}-f(-1/\tau_0)\,(-1/\tau_0)^{s-1}\,\tau_0^{-2}
=f(\tau_0)\big[\tau_0^{s-1}-e^{i\pi(s-1)}\tau_0^{k-1-s}\big].
\end{equation}
The two brackets are negatives of each other, so
$\partial_{\tau_0}L^*=i^{-s}\big(\partial_{\tau_0}\!\int_{\gamma^S}+\partial_{\tau_0}\!\int_{\gamma^T}\big)=0$.
\end{proof}

\begin{remark}[Classical Mellin recovery]\label{rem:mellin}
For a holomorphic cusp form $f\in S_k$, the cusp limit $\tau_0\to i\infty$ collapses
the $S$-segment onto $[0,i\infty]$ while the $T$-segment vanishes exponentially, so
\begin{equation}
\lim_{\tau_0\to i\infty}L^*(f,s)=\int_0^{\infty}\! t^{s-1}f(it)\,dt,
\end{equation}
the classical completed $L$-integral.
\end{remark}

\section{The $L$-function}\label{sec:Lfun}% TODO placeholder title --- no person-names yet

\begin{proposition}[Cusp unfolding]\label{prop:unfold}
For $n>0$ and $\nu\in\CC$,
\begin{equation}
\int_i^{i+1}\! e^{2\pi inu}\,\zeta(\nu,u)\,du=i^{1-\nu}(2\pi n)^{\nu-1}\,\Gamma(1-\nu,2\pi n).
\end{equation}
\end{proposition}

\begin{proof}
Write $\zeta(\nu,u)=\sum_{m\ge0}(m+u)^{-\nu}$ and shift $u\mapsto u+m$ in the $m$-th term; as
$e^{2\pi in(u+m)}=e^{2\pi inu}$ for integer $m$, the unit segments stack into one ray,
\begin{equation}
\int_i^{i+1}\! e^{2\pi inu}\zeta(\nu,u)\,du
=\sum_{m\ge0}\int_{i+m}^{i+1+m}\! e^{2\pi inv}v^{-\nu}\,dv
=\int_i^{i+\infty}\! e^{2\pi inv}v^{-\nu}\,dv.
\end{equation}
For $n>0$ the integrand is holomorphic in the first quadrant and decays as $\im v\to\infty$
(since $|e^{2\pi inv}|=e^{-2\pi n\im v}$), so the horizontal ray rotates to the vertical
$[i,i\infty)$. With $v=it$ ($t\ge1$; then $dv=i\,dt$, $e^{2\pi inv}=e^{-2\pi nt}$,
$v^{-\nu}=i^{-\nu}t^{-\nu}$) and $w=2\pi nt$,
\begin{equation}
\int_i^{i+\infty}\! e^{2\pi inv}v^{-\nu}\,dv
=i^{1-\nu}\!\int_1^\infty\! e^{-2\pi nt}t^{-\nu}\,dt
=i^{1-\nu}(2\pi n)^{\nu-1}\Gamma(1-\nu,2\pi n).
\end{equation}
\end{proof}

\begin{theorem}[$L$-function for weakly holomorphic forms]\label{thm:weakL}
For $f\in M^!_k$, at $\tau_0=i$,
\begin{equation}
L^*(f,s)=-c_f(0)\Big(\tfrac1s+\tfrac{i^k}{k-s}\Big)
+\sum_{n\ne0}c_f(n)\Big[\tfrac{\Gamma(s,2\pi n)}{(2\pi n)^s}+i^k\tfrac{\Gamma(k-s,2\pi n)}{(2\pi n)^{k-s}}\Big]
=i^kL^*(f,k-s).
\end{equation}
\end{theorem}

\begin{proof}
At $\tau_0=i$ the $S$-segment degenerates ($-1/i=i$), so
$L^*(f,s)=i^{-s}\int_{i-1}^{i} f\,\tk\,d\tau$. Expand $f=\sum_n c_f(n)e^{2\pi in\tau}$ and
substitute $u=\tau+1$ ($e^{2\pi in(u-1)}=e^{2\pi inu}$): each $n>0$ mode is a cusp unfolding
(Proposition~\ref{prop:unfold}), and the closed form persists to the finitely many $n<0$ modes
(the cusp principal part) by analytic continuation. The first kernel term ($\nu=1-s$, so
$1-\nu=s$) gives $i^{-s}\,i^{s}(2\pi n)^{-s}\Gamma(s,2\pi n)=\Gamma(s,2\pi n)/(2\pi n)^{s}$; the
second ($\nu=s-k+1$, so $1-\nu=k-s$) gives
\begin{equation}
-\,i^{-s}e^{i\pi(s-1)}\,i^{\,k-s}\,(2\pi n)^{s-k}\,\Gamma(k-s,2\pi n)
=i^{k}\,\frac{\Gamma(k-s,2\pi n)}{(2\pi n)^{k-s}},
\end{equation}
the $s$ cancelling in the phase $-i^{-s}e^{i\pi(s-1)}i^{k-s}=i^{k}$. So each $n\ne0$ mode
contributes $\Gamma(s,2\pi n)/(2\pi n)^{s}+i^{k}\Gamma(k-s,2\pi n)/(2\pi n)^{k-s}$, while the
$n=0$ mode contributes $c_f(0)\,i^{-s}\!\int_{i-1}^{i}\tk\,d\tau=-c_f(0)\big(\tfrac1s+\tfrac{i^k}{k-s}\big)$.
Swapping the two $\Gamma$-terms and using $i^{2k}=(-1)^k=1$ (the weight $k$ is even) shows that
bracket equals $i^k$ times its $s\mapsto k-s$ image; the polar term is $i^k$-symmetric likewise,
giving $L^*(f,s)=i^kL^*(f,k-s)$.
\end{proof}

\begin{definition}[Period cocycle]\label{def:cocycle}
Let $V_n:=\CC[X,Y]_n$ with the action $(P|_\gamma)(X,Y):=P(aX+bY,cX+dY)$, and
$\omega_f:=f(\tau)\,(X-\tau Y)^n\,d\tau$. For $f\in F_k$ and $\tau_0\in\HH$, the
\emph{period cocycle} is
\begin{equation}
C^f_\gamma(\tau_0):=(2\pi i)^{n+1}\!\int_{\gamma^{-1}\tau_0}^{\tau_0}\!\omega_f\ \in V_n,
\qquad \gamma\in\Gamma,
\end{equation}
satisfying $C^f_{gh}=C^f_g|_h+C^f_h$.
\end{definition}

\begin{lemma}[Parabolic representatives exist iff $c_f(0)=0$]\label{lem:gate}
With $\delta_T P:=P|_T-P$ on $V_n$, one has $[C^f_T(\tau_0)]_{X^n}=(2\pi i)^{n+1}c_f(0)$,
so $C^f_T\in\im\delta_T$ if and only if $c_f(0)=0$ --- the principal part of $f$ never obstructs.
\end{lemma}

\begin{proof}
The $X^n$-coefficient of $\omega_f$ is $f(\tau)\,d\tau$, so
$[C^f_T(\tau_0)]_{X^n}=(2\pi i)^{n+1}\!\int_{\tau_0-1}^{\tau_0}\! f\,d\tau=(2\pi i)^{n+1}c_f(0)$,
the $m\ne0$ Fourier modes integrating to zero over a unit period. On the other side,
$(P|_T)(X,Y)=P(X+Y,Y)$ has the same $X^n$-coefficient as $P$, so $[\delta_T P]_{X^n}=0$ for every
$P\in V_n$; since $\ker\delta_T=\CC\,Y^n$ is one-dimensional, $\im\delta_T$ is the entire
hyperplane $\{Q:[Q]_{X^n}=0\}$. Hence $C^f_T\in\im\delta_T$ iff $[C^f_T]_{X^n}=0$ iff $c_f(0)=0$.
\end{proof}

\begin{definition}[Period polynomial]\label{def:rf}
For $f\in S^!_k$ (so $c_f(0)=0$), the \emph{period polynomial} is the $S$-value of the
parabolic representative,
\begin{equation}
r_f:=C^f_S(\tau_0)-\big(P|_S-P\big)\ \in V_n,
\end{equation}
with $P\in V_n$ chosen so that $\delta_T P=C^f_T$ (Lemma~\ref{lem:gate}).
\end{definition}

\begin{proposition}[Period coefficients are special $L$-values]\label{prop:periods}
For $f\in S^!_k$ and $\ell\in\{1,\dots,k-3\}$,
\begin{equation}
[r_f]_{X^{n-\ell}Y^\ell}=(2\pi i)^{n+1}(-1)^\ell\binom{n}{\ell}\,i^{\ell+1}\,L^*(f,\ell+1).
\end{equation}
\end{proposition}

\begin{proof}
The cocycle is the coboundary of the Eichler primitive
$\widetilde F(\tau)=\int^{\tau}\! f(z)(X-zY)^n\,dz$, namely $C^f_\gamma=\widetilde F-\widetilde F|_\gamma$,
and $r_f=C^f_S-(P|_S-P)$ is the $S$-value of its cusp-regular ($T$-flat) branch, $P$ solving
$\delta_T P=C^f_T$ (Lemma~\ref{lem:gate}). Read the $X^{n-\ell}Y^\ell$-coefficient. From
$(X-\tau Y)^n=\sum_j\binom{n}{j}(-\tau)^jX^{n-j}Y^j$,
\begin{equation}
[C^f_S]_{X^{n-\ell}Y^\ell}=(2\pi i)^{n+1}(-1)^\ell\binom{n}{\ell}\!\int_{\gamma^S}\! f\,\tau^\ell\,d\tau .
\end{equation}
The kernel realizes the correction: $\tk(\tau,\ell+1)$ is the antidifference
$\delta_T\tk(\tau,\ell+1)=-(\tau+1)^\ell+(-1)^\ell(\tau+1)^{n-\ell}$, whose two monomials are the
$\ell$- and $(n-\ell)$-components exchanged by $Y|_S=X$; so the $T$-segment integral against
$\tk$ is the coboundary $P$ inverting $\delta_T$, with
$[P|_S-P]_{X^{n-\ell}Y^\ell}=-(2\pi i)^{n+1}(-1)^\ell\binom{n}{\ell}\!\int_{\gamma^T}\! f\,\tk(\tau,\ell+1)$.
Therefore
\begin{equation}
[r_f]_{X^{n-\ell}Y^\ell}
=(2\pi i)^{n+1}(-1)^\ell\binom{n}{\ell}\Big[\int_{\gamma^S}\! f\,\tau^\ell+\int_{\gamma^T}\! f\,\tk(\tau,\ell+1)\Big]
=(2\pi i)^{n+1}(-1)^\ell\binom{n}{\ell}\,i^{\ell+1}L^*(f,\ell+1),
\end{equation}
using $L^*(f,\ell+1)=i^{-(\ell+1)}[\int_{\gamma^S}f\tau^\ell+\int_{\gamma^T}f\tk(\tau,\ell+1)]$.
\end{proof}

\subsection{Why a finite contour}\label{subsec:why}% TODO placeholder title

\begin{remark}\label{rem:canonical}
For a weak cusp form $f\in S^!_k$ the classical Mellin integral
$\int_0^{i\infty}f(\tau)(\tau/i)^s\tfrac{d\tau}{\tau}$ does not converge --- $f$ grows at
the cusp --- and the deformed-Mellin construction of~\cite{BFK2011} supplies a finite value
through a non-holomorphic regularization there. The functional of Definition~\ref{def:Lint}
reaches the same value with no regularization: written as its two segments, each diverges as
$\tau_0\to i\infty$, with opposite signs, and the two cancel at every finite $\tau_0$, so
$L^*(f,s)$ is finite throughout $\HH$ although the Mellin contour itself has no limit. By
Proposition~\ref{prop:indep} the value is independent of $\tau_0$ and of the paths; BFK's
regularized value is its evaluation at $\tau_0=i$ (Theorem~\ref{thm:weakL}), and its
integer-$s$ values are Brown's quasi-periods (Corollary~\ref{cor:quasi}). Two independent
finite computations --- Brown's cohomological periods and the BFK $L$-values, obtained
without reference to one another --- meet at this one contour, which fixes it as the
canonical object. Finiteness and $\tau_0$-independence are the analytic shadow of the
parabolic representative~\cite{Brown2017,BCD2018,BGKO}.
\end{remark}

\section{Meromorphic modular forms}\label{sec:mero}% TODO placeholder title

\begin{definition}[Polylog projection]\label{def:proj}
Let $f\in F_k$ have interior poles at the points $\tau_p$ of a set $\mathcal P_f$ of
$\Gamma$-inequivalent representatives, and write $\Li_s(z)=\sum_{n>0}z^n n^{-s}$. Since
$\Li_{1-m}\big(e^{2\pi i(\tau-\tau_p)}\big)$ has a pole of order $m$ at $\tau_p$, the principal part of
$f$ there is matched by a unique \emph{polylog projection}
\begin{equation}
\widehat P_{\tau_p}f(\tau):=\sum_{m>0}r^*_{f,\tau_p}(m)\,\Li_{1-m}\big(e^{2\pi i(\tau-\tau_p)}\big),
\end{equation}
its coefficients $r^*_{f,\tau_p}(m)$ being the principal-part data of $f$ at $\tau_p$. The
\emph{projection} $\widehat P:=\sum_{\im\tau_p\ge1}\widehat P_{\tau_p}$ removes the principal
part of every pole on or above the line $\im\tau=1$.
\end{definition}

% §5.14b — [Def] regularized cusp coefficients (def:cRf)
\begin{definition}[Regularized cusp coefficients]\label{def:cRf}
With $\widehat P$ as in Definition~\ref{def:proj},
\begin{equation}
c_{Rf}(n):=\big[(I-\widehat P)f\big]_{q^n}.
\end{equation}
\end{definition}

\begin{definition}[The space $F(\infty)$]\label{def:Finfty}
$F(\infty)$ is the space of meromorphic $g$ on $\HH$ with $g(\tau)=g(\tau+1)$, finitely many
poles in any compact subset of the strip $\{{\rm Im}\,\tau>0,\ |{\rm Re}\,\tau|\le\tfrac12\}$,
and $g(\tau)=O(e^{C\,{\rm Im}\,\tau})$ as ${\rm Im}\,\tau\to\infty$ for some $C$. Each such
$g$ has a Fourier expansion $g=\sum_n c_g(n)q^n$.
\end{definition}

Then $F_k\subset F(\infty)$, and $(I-\widehat P)f\in F(\infty)$ --- the polylog tails are
$T$-periodic but not modular, so the projection leaves $F_k$.

\begin{lemma}[$q$-series--integral commutation]\label{lem:commute}
Let $g\in F(\infty)$ have no pole on or above the line ${\rm Im}\,\tau=1$ (e.g.\
$g=(I-\widehat P)f$). Then $g=\sum_n c_g(n)q^n$ converges along the $T$-segment at $\tau_0=i$
and integrates term by term:
\begin{equation}
\int_{\gamma^T_i} g(\tau)\,\tk(\tau,s)\,d\tau
=\sum_n c_g(n)\int_{i-1}^{i} e^{2\pi in\tau}\,\tk(\tau,s)\,d\tau.
\end{equation}
A pole at ${\rm Im}\,\tau_p\ge1$ defeats this --- there the coefficients grow at least like
$n^{m-1}e^{2\pi n\,{\rm Im}\,\tau_p}$~\cite[Lem.~2.6]{1806}.
\end{lemma}

\begin{definition}[Polylog integral]\label{def:polylog}
For $N\ge0$ and $z\in\HH$, the \emph{polylog integral} is the $L$-integral of a single tail
$\Li_{-N}(e^{2\pi i(\tau-z)})$ along the $T$-segment at $\tau_0=i$ (where $\gamma^S$ degenerates, since
$-1/i=i$):
\begin{equation}
I_N(s,z):=i^{-s}\int_{i-1}^{i}\Li_{-N}\big(e^{2\pi i(\tau-z)}\big)\,\tk(\tau,s)\,d\tau.
\end{equation}
\end{definition}

\begin{lemma}[Closed form of the polylog integral]\label{lem:polylogcf}
Write $\mathcal G(s,m):=\dfrac{\Gamma(s,2\pi m)}{(2\pi m)^s}+i^k\dfrac{\Gamma(k-s,2\pi m)}{(2\pi m)^{k-s}}$,
the building block of Theorem~\ref{thm:weakL} continued to $m<0$. For the projected poles
${\rm Im}\,z>1$, where $|e(\tau-z)|\ge1$ on the contour, polylog inversion
$\Li_{-N}(w)=(-1)^{N+1}\Li_{-N}(1/w)$ resums the integral into the convergent series
\begin{equation}
I_N(s,z)=\delta_{N,0}\Big(\tfrac1s+\tfrac{i^k}{k-s}\Big)
+(-1)^{N+1}\sum_{n\ge1} n^N e^{2\pi inz}\,\mathcal G(s,-n).
\end{equation}
\end{lemma}

\begin{theorem}[$L$-function of a meromorphic modular form]\label{thm:mero}
For $f\in F_k$ with no pole on the edge ${\rm Im}\,\tau=1$, at $\tau_0=i$,
\begin{equation}
L^*(f,s)=-c_{Rf}(0)\Big(\tfrac1s+\tfrac{i^k}{k-s}\Big)
+\sum_{n\ne0}c_{Rf}(n)\,\mathcal G(s,n)
+\sum_{{\rm Im}\,\tau_p>1}\sum_{m>0}r^*_{f,\tau_p}(m)\,I_{m-1}(s,\tau_p),
\end{equation}
$\mathcal G(s,n)=\dfrac{\Gamma(s,2\pi n)}{(2\pi n)^s}+i^k\dfrac{\Gamma(k-s,2\pi n)}{(2\pi n)^{k-s}}$;
the regularized cusp sum (Definition~\ref{def:cRf}, Theorem~\ref{thm:weakL}) plus one polylog
integral (Lemma~\ref{lem:polylogcf}) per interior pole above the edge. Poles on the edge enter
by the principal value of Lemma~\ref{lem:onedge}.
\end{theorem}

\begin{lemma}[$L$-function, a pole on the edge]\label{lem:onedge}
Routing the $T$-path just below versus just above a pole $\tau_p$ of $f\in F_k$ at height
${\rm Im}\,\tau_p=1$ gives finite values differing by the residue,
\begin{equation}
L^*_{\rm below}(f,s)-L^*_{\rm above}(f,s)
=2\pi i\,i^{-s}\Res_{\tau=\tau_p}\!\big[f(\tau)\,\tk(\tau,s)\big].
\end{equation}
We define the edge value as the principal value $L^*=\tfrac12(L^*_{\rm above}+L^*_{\rm below})
=L^*_{\rm below}-\pi i\,i^{-s}\Res_{\tau=\tau_p}[f\,\tk]$. At the elliptic point $\tau_p=i$ the
pole lies on the $S$-segment instead; its principal value, reached as $\tau_0\to i$, is finite
and basepoint-independent.
\end{lemma}

\begin{corollary}[Wall-crossing]\label{cor:wall}
For $f\in F_k$, $L^*(f,s)$ is constant in $\tau_0$ within each chamber of
$\HH\setminus f^{-1}(\infty)$; crossing a wall, where the contour sweeps past an interior pole
$\tau_p$, changes it by
\begin{equation}
\Delta L^*(f,s)=2\pi i\,i^{-s}\Res_{\tau=\tau_p}\!\big[f(\tau)\,\tk(\tau,s)\big]
\end{equation}
when the $T$-segment crosses $\tau_p$ (and by the same with $\tau^{s-1}$ in place of $\tk$ when
the $S$-segment does). Thus $L^*$ is fixed by the contour up to the residues it crosses --- the
meromorphic counterpart of Proposition~\ref{prop:indep}, recovering its constancy on $M^!_k$.
\end{corollary}
